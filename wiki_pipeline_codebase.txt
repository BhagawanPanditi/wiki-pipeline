#!/usr/bin/env bash
# ==============================================================================
# End-to-End Wiki Knowledge Graph Pipeline Runner
# ==============================================================================
# Runs all 9 pipeline stages sequentially for a single domain or all 7 domains:
#   Stage 1: extract_concepts.py       (Concept Extraction from Chunks)
#   Stage 2: verify_extraction.py      (Phase 1 Verification Audit)
#   Stage 3: refine_concepts.py        (Second-Pass Concept Refinement)
#   Stage 4: verify_extraction.py      (Phase 2 Verification Delta Audit)
#   Stage 5: consolidate_documents.py (Document-Level Deduplication)
#   Stage 6: consolidate_global.py    (Corpus-Level Global Consolidation)
#   Stage 7: write_briefs_and_entities.py (Briefs & Vital Entities Extraction)
#   Stage 8: link_entities_jaccard.py (Semantic Jaccard Linking & LLM Relations)
#   Stage 9: final_consolidation.py   (Macro/Deep Verify & Hybrid Wiki Export)
#
# Supported Domains:
#   1. sustainable_living
#   2. biology
#   3. earth_science
#   4. economics
#   5. psychology
#   6. robotics
#   7. stackoverflow
#
# Usage:
#   ./pipeline.sh <domain | all> [options]
#
# Examples:
#   ./pipeline.sh sustainable_living
#   ./pipeline.sh all
#   ./pipeline.sh sustainable_living --from-stage 7
#   ./pipeline.sh all --data-dir /custom/data --wiki-base-dir /custom/wiki
# ==============================================================================

set -eo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
cd "$SCRIPT_DIR"

# Default Paths (Adjustable via CLI flags or environment variables)
DATA_DIR="${DATA_DIR:-/home/compiling-ganesh/24m0829/long_document/wiki_pipeline/data}"
WIKI_BASE_DIR="${WIKI_BASE_DIR:-/home/compiling-ganesh/24m0829/long_document/wiki}"

ALL_DOMAINS=(
    "sustainable_living"
    "biology"
    "earth_science"
    "economics"
    "psychology"
    "robotics"
    "stackoverflow"
)

TARGET_DOMAIN="${1:-sustainable_living}"
shift || true

FROM_STAGE=1
TO_STAGE=9
DISABLE_THINKING=true
EXTRA_ARGS=()

while [[ $# -gt 0 ]]; do
    case "$1" in
        --data-dir|--data_dir)
            DATA_DIR="$2"; shift 2 ;;
        --wiki-base-dir|--wiki_base_dir)
            WIKI_BASE_DIR="$2"; shift 2 ;;
        --from-stage|--from_stage|--start-stage|--start_stage)
            FROM_STAGE="$2"; shift 2 ;;
        --to-stage|--to_stage)
            TO_STAGE="$2"; shift 2 ;;
        --enable-thinking|--enable_thinking)
            DISABLE_THINKING=false; shift ;;
        --disable-thinking|--disable_thinking)
            DISABLE_THINKING=true; shift ;;
        *)
            EXTRA_ARGS+=("$1"); shift ;;
    esac
done

THINKING_FLAG="--disable_thinking"
if [ "$DISABLE_THINKING" = false ]; then
    THINKING_FLAG="--enable_thinking"
fi

# Print banner
echo "=============================================================================="
echo "          END-TO-END CONCEPT KNOWLEDGE GRAPH PIPELINE RUNNER                  "
echo "=============================================================================="
echo "Script Directory  : $SCRIPT_DIR"
echo "Data Directory    : $DATA_DIR"
echo "Wiki Base Dir     : $WIKI_BASE_DIR"
echo "Target Domain     : $TARGET_DOMAIN"
echo "Stage Range       : Stage $FROM_STAGE -> Stage $TO_STAGE"
echo "Thinking Mode     : $THINKING_FLAG"
echo "=============================================================================="
echo ""

run_stage() {
    local stage_num="$1"
    local stage_name="$2"
    shift 2
    local cmd=("$@")

    if [ "$stage_num" -lt "$FROM_STAGE" ] || [ "$stage_num" -gt "$TO_STAGE" ]; then
        echo ">>> [STAGE $stage_num: $stage_name] SKIPPED (Stage range: $FROM_STAGE-$TO_STAGE)"
        return 0
    fi

    echo ""
    echo "=============================================================================="
    echo ">>> [STAGE $stage_num: $stage_name] Starting at $(date '+%Y-%m-%d %H:%M:%S')"
    echo "Command: ${cmd[*]}"
    echo "=============================================================================="

    local start_time=$(date +%s)
    local retries=5
    local attempt=1

    # Pre-flight server check before launching stage
    if [ -f "check_servers.py" ]; then
        python3 check_servers.py || {
            echo ">>> [STAGE $stage_num ERROR] Pre-flight server check failed! Aborting." >&2
            exit 1
        }
    fi

    while [ $attempt -le $retries ]; do
        if "${cmd[@]}"; then
            break
        else
            if [ $attempt -lt $retries ]; then
                echo ">>> [STAGE $stage_num WARNING] Command failed on attempt $attempt/$retries. Retrying with --resume in 5s..." >&2
                if [ -f "check_servers.py" ]; then
                    python3 check_servers.py || true
                fi
                sleep 5
            else
                echo ">>> [STAGE $stage_num ERROR] Command failed after $retries attempts!" >&2
                exit 1
            fi
        fi
        attempt=$((attempt + 1))
    done
    local end_time=$(date +%s)
    local elapsed=$((end_time - start_time))

    echo ">>> [STAGE $stage_num: $stage_name] COMPLETED in ${elapsed}s ($(date '+%Y-%m-%d %H:%M:%S'))"
}

run_domain_pipeline() {
    local domain="$1"
    local input_corpus="$DATA_DIR/${domain}.jsonl"
    local wiki_path="$WIKI_BASE_DIR/${domain}"

    echo ""
    echo "##############################################################################"
    echo "##############################################################################"
    echo "  PROCESSING DOMAIN: [${domain^^}]"
    echo "  Input File : $input_corpus"
    echo "  Wiki Path  : $wiki_path"
    echo "##############################################################################"
    echo "##############################################################################"

    if [ ! -f "$input_corpus" ]; then
        # Check if .json exists
        if [ -f "$DATA_DIR/${domain}.json" ]; then
            input_corpus="$DATA_DIR/${domain}.json"
        else
            echo "ERROR: Input file $input_corpus does not exist!" >&2
            exit 1
        fi
    fi

    mkdir -p "$wiki_path"

    # Stage 1: Concept Extraction (Pass 1)
    run_stage 1 "Concept Extraction (Pass 1)" \
        python3 extract_concepts.py \
            --input_path "$input_corpus" \
            --wiki_path "$wiki_path" \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    # Stage 2: Audit Verification (Phase 1)
    run_stage 2 "Verification Audit (Phase 1)" \
        python3 verify_extraction.py \
            --wiki_path "$wiki_path" \
            --phase 1 \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    # Stage 3: Second-Pass Refinement
    run_stage 3 "Concept Refinement (Pass 2)" \
        python3 refine_concepts.py \
            --wiki_path "$wiki_path" \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    # Stage 4: Second-Pass Verification (Phase 2)
    run_stage 4 "Verification Delta (Phase 2)" \
        python3 verify_extraction.py \
            --wiki_path "$wiki_path" \
            --phase 2 \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    # Stage 5: Document Consolidation
    run_stage 5 "Document Consolidation" \
        python3 consolidate_documents.py \
            --wiki_path "$wiki_path" \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    # Stage 6: Global Consolidation
    run_stage 6 "Global Consolidation" \
        python3 consolidate_global.py \
            --wiki_path "$wiki_path" \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    # Stage 7: Briefs & Entities
    run_stage 7 "Briefs & Entities Extraction" \
        python3 write_briefs_and_entities.py \
            --wiki_path "$wiki_path" \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    # Stage 8: Jaccard Linking & LLM Relationship Synthesis
    run_stage 8 "Jaccard Linking & Relation Synthesis" \
        python3 link_entities_jaccard.py \
            --wiki_path "$wiki_path" \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    # Stage 9: Final Consolidation & Wiki Export
    run_stage 9 "Final Consolidation & Hybrid Wiki Export" \
        python3 final_consolidation.py \
            --wiki_path "$wiki_path" \
            $THINKING_FLAG \
            --resume "${EXTRA_ARGS[@]}"

    echo ""
    echo "=============================================================================="
    echo ">>> DOMAIN PIPELINE COMPLETED: [${domain^^}]"
    echo ">>> Wiki Knowledge Graph & Pages saved to: $wiki_path"
    echo "=============================================================================="
}

TOTAL_START=$(date +%s)

if [ "$TARGET_DOMAIN" = "all" ] || [ "$TARGET_DOMAIN" = "ALL" ]; then
    echo "Running complete pipeline for all 7 domains: ${ALL_DOMAINS[*]}"
    for dom in "${ALL_DOMAINS[@]}"; do
        run_domain_pipeline "$dom"
    done
else
    # Check if target domain is valid
    MATCHED=false
    for dom in "${ALL_DOMAINS[@]}"; do
        if [ "$dom" = "$TARGET_DOMAIN" ]; then
            MATCHED=true
            break
        fi
    done
    if [ "$MATCHED" = false ]; then
        echo "Warning: '$TARGET_DOMAIN' is not in standard list (${ALL_DOMAINS[*]}). Running anyway..."
    fi
    run_domain_pipeline "$TARGET_DOMAIN"
fi

TOTAL_END=$(date +%s)
TOTAL_ELAPSED=$((TOTAL_END - TOTAL_START))

echo ""
echo "=============================================================================="
echo "PIPELINE EXECUTION FINISHED SUCCESSFULLY!"
echo "Total Execution Time: ${TOTAL_ELAPSED}s ($((TOTAL_ELAPSED / 60)) minutes)"
echo "=============================================================================="

#!/usr/bin/env python3
"""Pipeline configuration: servers, models, defaults and per-stage overrides.

Reads ``config.json`` from the repository root when it exists, and falls back to
the defaults below when it does not. The file is git-ignored, so every machine
keeps its own.

``servers`` and ``model_name`` apply to every stage, to ``check_servers.py`` and
to the runner's preflight. A file that exists but is not valid JSON is an error,
not a silent fall-back to the defaults.

Example::

    {
      "servers": ["http://cn27-a40:8000/v1", "http://cn30-a40:8001/v1"],
      "model_name": "Qwen/Qwen3.8-27B",
      "stage_concurrency": {
        "extract-concepts": 8,
        "add-reverse-links": {"concurrency_per_server": 48, "attempts": 6}
      }
    }

``stage_concurrency`` overrides the built-in limits in ``utils/cli.py``
(``STAGE_PROFILES``) for one stage. A plain number sets that stage's concurrency.
An object may set ``concurrency_per_server``, ``attempts`` and
``budget_minutes``. Use it to match the KV-cache budget that vLLM reports at
startup ("Maximum concurrency for 32,768 tokens per request") without editing code.
"""

import json
from pathlib import Path
from typing import Any, Dict, List, Tuple

CONFIG_FILE = Path(__file__).resolve().parent / "config.json"


def _read_config() -> Dict[str, Any]:
    """Return the parsed config.json, or an empty mapping when there is no file.

    A file that exists but cannot be read, is not valid JSON, or is not a JSON
    object is an error. Falling back to the built-in cluster servers in that case
    would send every stage to the wrong machines without a word.

    Returns:
        The parsed object, or ``{}`` when config.json does not exist.

    Raises:
        ValueError: If config.json exists but is unreadable, not valid JSON, or
            not a JSON object.
    """
    if not CONFIG_FILE.exists():
        return {}
    try:
        with open(CONFIG_FILE, "r", encoding="utf-8") as f:
            data = json.load(f)
    except (OSError, ValueError) as exc:
        raise ValueError(f"{CONFIG_FILE} is not readable JSON: {exc}") from exc
    if not isinstance(data, dict):
        raise ValueError(f"{CONFIG_FILE} must contain a JSON object at the top level.")
    return data


_warned_config_keys: set = set()


def load_global_config() -> Tuple[List[str], str]:
    """Return the server list and model name from config.json.

    Returns:
        (servers, model_name) loaded from config.json.

    Raises:
        ValueError: If config.json is missing or unreadable.
    """
    data = _read_config()
    servers = data.get("servers") or []
    model = data.get("model_name") or data.get("model") or ""
    return servers, model


def load_stage_overrides() -> Dict[str, Any]:
    """Return the per-stage tuning overrides from config.json, keyed by stage name.

    Returns:
        The ``stage_concurrency`` object, or ``{}`` when config.json does not set it.

    Raises:
        ValueError: If config.json exists but ``stage_concurrency`` is not an object,
            so that a mistyped list is reported rather than silently ignored.
    """
    data = _read_config()
    if "stage_concurrency" not in data:
        return {}
    overrides = data["stage_concurrency"]
    if not isinstance(overrides, dict):
        raise ValueError(f"{CONFIG_FILE}: 'stage_concurrency' must be an object keyed by stage name.")
    return overrides

#!/usr/bin/env python3
"""
Pre-flight vLLM Server Health Check
-----------------------------------
Pings every vLLM server defined in config.json (or config.py defaults)
via GET /models to verify liveness and responsiveness before pipeline tasks launch.
"""

import sys
import httpx
from config import load_global_config

def main():
    """Probe every configured LLM server and report which are healthy.

    Reads ``config.json``, sends each server a short request, and prints the result
    per server. The runner calls this before every stage and again before each retry.

    Returns:
        None. Exits 2 when ``config.json`` is invalid and 1 when any server is
        unreachable, so a caller can tell the two apart."""
    try:
        raw_servers, model = load_global_config()
    except ValueError as exc:
        print(f"ERROR: {exc}")
        sys.exit(2)
    servers = list(raw_servers.keys()) if isinstance(raw_servers, dict) else raw_servers
    if not isinstance(servers, list) or not servers or any(not isinstance(s, str) or not s.strip() for s in servers):
        print("ERROR: config.json must contain a non-empty 'servers' list or dictionary of URLs.")
        sys.exit(2)

    print("=" * 65)
    print(f" PRE-FLIGHT CHECK: Checking {len(servers)} configured vLLM servers...")
    print(f" Expected Model: {model}")
    print("=" * 65)

    all_ok = True
    for s in servers:
        target = s.rstrip('/')
        url = f"{target}/models"
        try:
            r = httpx.get(url, timeout=7.0)
            if r.status_code == 200:
                print(f" [PASS] {target} is LIVE (HTTP 200)")
            else:
                print(f" [FAIL] {target} returned HTTP {r.status_code}")
                all_ok = False
        except Exception as e:
            print(f" [FAIL] {target} is UNREACHABLE ({type(e).__name__}: {e})")
            all_ok = False

    print("=" * 65)
    if not all_ok:
        print("ERROR: One or more vLLM servers are down or unreachable!")
        print("Please check your sbatch jobs / network or update config.json.")
        sys.exit(1)
    else:
        print(f"ALL {len(servers)} SERVERS ARE LIVE AND HEALTHY! Proceeding...")
        print("=" * 65)

if __name__ == "__main__":
    main()

"""Pipeline Error Handling & Classification.
--------------------------------------------
Classifies errors into transient (server network faults), recoverable (malformed/truncated
model outputs), and permanent (invalid requests/syntax) to manage intelligent retry loops.
"""

import asyncio
import json
import time
from typing import Optional

try:  # httpx is a hard runtime dependency, but keep unit tests importable.
    import httpx
except ImportError:  # pragma: no cover
    httpx = None

try:
    import openai
except ImportError:  # pragma: no cover
    openai = None

try:
    from pydantic import ValidationError
except ImportError:  # pragma: no cover
    ValidationError = None


TRANSIENT = "transient"
RECOVERABLE = "recoverable"
PERMANENT = "permanent"


class StageBudgetExceeded(RuntimeError):
    """The stage exhausted its wall-clock budget and must stop retrying."""


class PermanentFailure(RuntimeError):
    """An item failed for a reason that retrying cannot fix."""

    def __init__(self, message: str, *, cause: Optional[BaseException] = None):
        """Record a failure that must not be retried.

        Args:
            message: Human-readable reason.
            cause: The underlying exception, kept so the debug log and the stage's
                final error message can show what actually went wrong."""
        super().__init__(message)
        self.cause = cause


class TransientFailureExhausted(RuntimeError):
    """An item kept failing transiently and ran out of attempts."""

    def __init__(self, message: str, *, attempts: int, cause: Optional[BaseException] = None):
        """Record that every retry for one item was used up.

        Args:
            message: Human-readable reason.
            attempts: How many attempts were made before giving up.
            cause: The last underlying exception."""
        super().__init__(message)
        self.attempts = attempts
        self.cause = cause


class AllServersUnhealthy(RuntimeError):
    """Every configured vLLM server stopped accepting connections."""


# Our own control-flow exceptions must never be retried.
_CONTROL_EXCEPTIONS = (StageBudgetExceeded, PermanentFailure, TransientFailureExhausted,
                       AllServersUnhealthy)


def classify_error(exc: BaseException) -> str:
    """Return TRANSIENT, RECOVERABLE or PERMANENT for an exception."""
    if isinstance(exc, _CONTROL_EXCEPTIONS):
        return PERMANENT

    # ── asyncio / stdlib network faults ────────────────────────────────────
    if isinstance(exc, (asyncio.TimeoutError, TimeoutError, ConnectionError)):
        return TRANSIENT
    if isinstance(exc, OSError) and getattr(exc, "errno", None) is not None:
        # Socket-level faults (ECONNRESET, EPIPE, ETIMEDOUT, ...).
        return TRANSIENT

    # ── httpx transport faults ─────────────────────────────────────────────
    if httpx is not None:
        if isinstance(exc, httpx.TimeoutException):
            return TRANSIENT
        if isinstance(exc, (httpx.ConnectError, httpx.ReadError, httpx.WriteError,
                            httpx.RemoteProtocolError, httpx.NetworkError)):
            return TRANSIENT
        if isinstance(exc, httpx.HTTPStatusError):
            return _classify_status(exc.response.status_code)

    # ── openai SDK faults ──────────────────────────────────────────────────
    if openai is not None:
        if isinstance(exc, (openai.APITimeoutError, openai.APIConnectionError)):
            return TRANSIENT
        if isinstance(exc, openai.RateLimitError):
            return TRANSIENT
        if isinstance(exc, openai.InternalServerError):
            return TRANSIENT
        if isinstance(exc, openai.APIStatusError):
            return _classify_status(getattr(exc, "status_code", 500))

    # ── programming / invariant bugs (must never retry) ────────────────────
    if isinstance(exc, (NameError, AttributeError, ImportError, TypeError, AssertionError)):
        return PERMANENT

    # ── malformed / truncated model output ─────────────────────────────────
    if isinstance(exc, json.JSONDecodeError):
        return RECOVERABLE
    if ValidationError is not None and isinstance(exc, ValidationError):
        return RECOVERABLE
    if isinstance(exc, (ValueError, KeyError, IndexError,
                        StopIteration, UnicodeDecodeError)):
        # Includes recoverable parse failures and length truncation exceptions
        return RECOVERABLE

    return RECOVERABLE


def _classify_status(status_code: Optional[int]) -> str:
    """Map an HTTP status code onto a retry class."""
    if status_code is None:
        return TRANSIENT
    if status_code == 429 or status_code >= 500:
        return TRANSIENT
    if 400 <= status_code < 500:
        # 400/401/403/404/422 are deterministic: the request itself is wrong
        # (bad payload, max_tokens over context length, unknown model, auth).
        return PERMANENT
    return TRANSIENT


def describe_error(exc: BaseException) -> str:
    """Full, untruncated one-line description used in logs and error messages."""
    return f"{type(exc).__name__}: {exc}"


class StageBudget:
    """Wall-clock ceiling for one stage.

    A stage must survive slow servers without idling forever on backoff, so
    every retry path checks this budget. On expiry the stage raises
    StageBudgetExceeded, leaves its checkpoint intact, and is resumable.
    """

    def __init__(self, minutes: Optional[float]):
        """Start the wall-clock clock for one stage.

        Args:
            minutes: Budget in minutes. ``None`` or a non-positive value disables the
                budget entirely, in which case nothing ever expires and ``cap_sleep``
                passes delays through unchanged."""
        self.minutes = minutes
        self.started_at = time.monotonic()
        self.deadline: Optional[float] = None
        if minutes is not None and minutes > 0:
            self.deadline = self.started_at + (minutes * 60.0)

    @property
    def enabled(self) -> bool:
        """Whether a budget was actually configured.

        Returns:
            ``True`` when ``minutes`` was a positive number."""
        return self.deadline is not None

    def remaining(self) -> Optional[float]:
        """Seconds left before the budget expires.

        Returns:
            The remaining seconds, or ``None`` when no budget is configured. Negative
            once expired."""
        if self.deadline is None:
            return None
        return max(0.0, self.deadline - time.monotonic())

    def elapsed(self) -> float:
        """Seconds since this budget was created.

        Returns:
            Elapsed wall-clock seconds as a float."""
        return time.monotonic() - self.started_at

    def expired(self) -> bool:
        """Whether the budget has run out.

        Returns:
            ``False`` when no budget is configured, otherwise whether the remaining
            time has reached zero."""
        return self.deadline is not None and time.monotonic() >= self.deadline

    def check(self, context: str = "stage") -> None:
        """Raise StageBudgetExceeded once the budget is gone."""
        if self.expired():
            raise StageBudgetExceeded(
                f"{context} exhausted its {self.minutes:g}-minute budget after "
                f"{self.elapsed() / 60.0:.1f} min. Checkpoints are preserved; "
                f"rerun with --resume after addressing the cause, or raise "
                f"--stage_budget_minutes."
            )

    def cap_sleep(self, delay: float) -> float:
        """Clamp a backoff sleep so it cannot overshoot the deadline."""
        remaining = self.remaining()
        if remaining is None:
            return delay
        return max(0.0, min(delay, remaining))

"""Atomic I/O Utilities.
-----------------------
Provides crash-resilient file writers using sibling temporary files, fsync,
and atomic file renames to prevent partial/torn writes during preemption or shutdown.
"""

import json
import os
from pathlib import Path
from typing import Any, Union

PathLike = Union[str, Path]


def _fsync_dir(directory: Path) -> None:
    """Persist a rename so it survives power loss, where the OS allows it."""
    try:
        dir_fd = os.open(directory, os.O_RDONLY)
    except OSError:
        return
    try:
        os.fsync(dir_fd)
    except OSError:
        # Not supported on every filesystem; the rename is still atomic.
        pass
    finally:
        os.close(dir_fd)


import uuid

def atomic_write_json(path: PathLike, data: Any, *, indent: int = 2,
                      fsync_dir: bool = True) -> None:
    """Write JSON beside its destination, then atomically replace the old file."""
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(f".{path.name}.{os.getpid()}.{uuid.uuid4().hex}.tmp")
    try:
        with temporary.open("w", encoding="utf-8") as f:
            json.dump(data, f, indent=indent, ensure_ascii=False)
            f.flush()
            os.fsync(f.fileno())
        os.replace(temporary, path)
        if fsync_dir:
            _fsync_dir(path.parent)
    finally:
        try:
            temporary.unlink()
        except FileNotFoundError:
            pass


def atomic_write_text(path: PathLike, text: str, *, fsync_dir: bool = True) -> None:
    """Atomically replace a text file."""
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(f".{path.name}.{os.getpid()}.{uuid.uuid4().hex}.tmp")
    try:
        with temporary.open("w", encoding="utf-8") as f:
            f.write(text)
            f.flush()
            os.fsync(f.fileno())
        os.replace(temporary, path)
        if fsync_dir:
            _fsync_dir(path.parent)
    finally:
        try:
            temporary.unlink()
        except FileNotFoundError:
            pass


def read_json(path: PathLike, default: Any = None) -> Any:
    """Read JSON, returning ``default`` when the file is absent or corrupt.

    A truncated checkpoint must never be mistaken for a valid one, so parse
    failures degrade to the default rather than propagating.
    """
    path = Path(path)
    if not path.is_file():
        return default
    try:
        with path.open("r", encoding="utf-8") as f:
            return json.load(f)
    except (OSError, json.JSONDecodeError, ValueError):
        return default


def sanitize_filename(name: str) -> str:
    """Make a concept title safe to use as a filename.

    Three stages carried byte-identical copies of this. Kept exactly as they had
    it: alphanumerics plus space/dot/underscore/hyphen, trailing space stripped,
    spaces to underscores, capped at 100 characters.
    """
    keepchars = (' ', '.', '_', '-')
    sanitized = "".join(c for c in name if c.isalnum() or c in keepchars).rstrip()
    return sanitized.replace(" ", "_")[:100]

"""Pipeline Stage Manifest and Verification Gate.

Implements Audit v3 Section 3.4:
- write_stage_manifest: writes stages/<stage>.done.json containing expected and completed item counts.
- verify_stage_manifest: verifies that an upstream stage completed with zero failures before allowing the next stage to run.
"""

from datetime import datetime
from pathlib import Path
from typing import Any, Dict
from utils.atomic_io import atomic_write_json, read_json


def write_stage_manifest(wiki_path: Path, stage: str, expected: int, completed: int, failed: int = 0, **extra) -> Path:
    """Record a stage completion manifest atomically."""
    manifest_dir = wiki_path / "stages"
    manifest_dir.mkdir(parents=True, exist_ok=True)
    manifest_file = manifest_dir / f"{stage}.done.json"

    data = {
        "stage": stage,
        "expected": expected,
        "completed": completed,
        "failed": failed,
        "finished_at": datetime.now().isoformat(),
        **extra,
    }
    atomic_write_json(manifest_file, data)
    return manifest_file


def verify_stage_manifest(wiki_path: Path, required_stage: str) -> Dict[str, Any]:
    """Check that required upstream stage completed cleanly. Raises RuntimeError if missing or failed."""
    manifest_file = wiki_path / "stages" / f"{required_stage}.done.json"
    if not manifest_file.is_file():
        raise RuntimeError(
            f"Required upstream stage manifest missing: {manifest_file}. "
            f"Run '{required_stage}' first."
        )

    data = read_json(manifest_file)
    if not isinstance(data, dict):
        raise RuntimeError(f"Corrupt manifest file: {manifest_file}")

    expected = data.get("expected", 0)
    completed = data.get("completed", 0)
    failed = data.get("failed", 0)

    if failed > 0 or (expected > 0 and completed < expected):
        raise RuntimeError(
            f"Upstream stage '{required_stage}' had incomplete or failed items "
            f"(expected={expected}, completed={completed}, failed={failed}). "
            f"Rerun with --resume before proceeding."
        )

    return data

"""Shared head-noun normalization and title key generation.

Normalizes only the head noun (last word of noun phrase) deterministically
using spaCy (en_core_web_sm) with parser and NER disabled for high throughput.
No domain-specific stoplists or hardcoded noun lists.
"""

from functools import lru_cache

_nlp_model = None


def _nlp():
    """Load and cache the spaCy pipeline. Fails loud if missing."""
    global _nlp_model
    if _nlp_model is None:
        try:
            import spacy
            _nlp_model = spacy.load("en_core_web_sm", disable=["parser", "ner"])
        except Exception as e:
            raise RuntimeError(
                f"spaCy or the 'en_core_web_sm' model is missing: {e}. "
                "Install via: pip install spacy && python -m spacy download en_core_web_sm"
            ) from e
    return _nlp_model


@lru_cache(maxsize=100000)
def normalize_head_noun(term: str) -> str:
    """Normalize the head noun (last word) of a concept title or entity.

    Leaves leading modifiers untouched, lowercases, and lemmatizes the final token.
    Deterministic, domain-agnostic, and idempotent.
    """
    if not term:
        return ""

    words = " ".join(term.strip().split()).casefold().split()
    if not words:
        return ""

    nlp_pipeline = _nlp()
    # Prefix with 'the ' so the tagger reliably treats the final token as a noun
    doc = nlp_pipeline(f"the {words[-1]}")
    if len(doc) >= 2 and doc[1].lemma_:
        words[-1] = doc[1].lemma_.casefold()

    return " ".join(words)


@lru_cache(maxsize=100000)
def canonical_title_key(title: str) -> str:
    """Produce a canonical key for candidate grouping across titles."""
    return normalize_head_noun(title)


@lru_cache(maxsize=100000)
def normalize_entity(entity: str) -> str:
    """Produce a normalized entity string for overlap computation."""
    return normalize_head_noun(entity)

"""Disjoint Set / Union Find data structure with path compression and union by rank.

Implements F-36 shared union-find with exact match keying and chain guards.
"""

from typing import Dict, Iterable, List, Optional


class UnionFind:
    """Disjoint Set / Union-Find structure keyed by exact normalized string keys."""

    def __init__(self, elements: Optional[Iterable[str]] = None):
        self.parent: Dict[str, str] = {}
        self.rank: Dict[str, int] = {}
        self.size: Dict[str, int] = {}
        if elements:
            for elem in elements:
                self.add(elem)

    def add(self, x: str) -> None:
        if x not in self.parent:
            self.parent[x] = x
            self.rank[x] = 0
            self.size[x] = 1

    def find(self, x: str) -> str:
        if x not in self.parent:
            self.add(x)
            return x
        # Path compression
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x: str, y: str) -> bool:
        """Union two elements. Returns True if sets were merged, False if already in same set."""
        root_x = self.find(x)
        root_y = self.find(y)
        if root_x == root_y:
            return False

        # Union by rank
        if self.rank[root_x] < self.rank[root_y]:
            root_x, root_y = root_y, root_x

        self.parent[root_y] = root_x
        self.size[root_x] += self.size[root_y]
        if self.rank[root_x] == self.rank[root_y]:
            self.rank[root_x] += 1
        return True

    def get_groups(self) -> Dict[str, List[str]]:
        """Return dict of root -> list of elements."""
        groups: Dict[str, List[str]] = {}
        for x in self.parent:
            root = self.find(x)
            groups.setdefault(root, []).append(x)
        return groups

"""Checkpoint & State Logging Utilities.
--------------------------------------
Manages durable JSON completion logs and CheckpointLog state trackers for pipeline
stages, ensuring completed units are safely skipped on resume.
"""

import asyncio
import json
from pathlib import Path
from typing import Any, Callable, Dict, Iterable, List, Optional

from .atomic_io import atomic_write_json


def load_json_log(path: Path, *, corrupt_message: Optional[str] = None,
                  coerce_bool: bool = False) -> Dict[str, Any]:
    """Load a checkpoint dict, tolerating absence and corruption.

    ``corrupt_message`` reproduces the warning each stage used to print when its
    log existed but could not be parsed, so the console output is unchanged.
    ``coerce_bool`` is for the pure ``{key: bool}`` logs; stages that mix other
    value types into the same log (document consolidation records concept
    counts) must leave it off or those values would be flattened to booleans.
    """
    p = Path(path)
    if not p.is_file():
        return {}
    try:
        with open(p, "r", encoding="utf-8") as f:
            data = json.load(f)
    except Exception:
        if corrupt_message:
            print(f"Warning: {corrupt_message}")
        return {}
    if not isinstance(data, dict):
        return {}
    if coerce_bool:
        return {str(k): bool(v) for k, v in data.items()}
    return data


def load_bool_log(path: Path, *, corrupt_message: Optional[str] = None) -> Dict[str, bool]:
    """Load a ``{key: bool}`` completion log, tolerating absence and corruption."""
    return load_json_log(path, corrupt_message=corrupt_message, coerce_bool=True)


def save_log_sync(path: Path, data: Dict[str, Any]) -> None:
    """Atomic synchronous checkpoint write."""
    atomic_write_json(path, data)


async def save_log_async(path: Path, data: Dict[str, Any]) -> None:
    """Offload the atomic write to a thread so the event loop keeps streaming."""
    await asyncio.to_thread(save_log_sync, path, data)


class CheckpointLog:
    """Tracks per-item completion and flushes atomically every ``flush_every``.

    A crash loses at most ``flush_every`` marks, and because the write is
    atomic the log on disk is always parseable. Items are re-derived from the
    log on resume, so an unflushed mark costs a redo rather than corruption.
    """

    def __init__(self, path: Path, *, flush_every: int = 25):
        """Open a checkpoint log, loading any state already on disk.

        Args:
            path: JSON file holding ``{item_key: completed}``.
            flush_every: How many marks may accumulate before a flush is required. A
                crash loses at most this many marks, and because the write is atomic
                the file on disk is always parseable."""
        self.path = Path(path)
        self.flush_every = max(1, flush_every)
        self.data: Dict[str, bool] = load_bool_log(self.path)
        self._dirty = 0

    def is_done(self, key: str) -> bool:
        """Report whether one item is already complete.

        Args:
            key: Item key.

        Returns:
            ``True`` when the key is present and marked complete. A key recorded as
            ``False`` (started but not finished) counts as not done, so resume retries it."""
        return bool(self.data.get(key, False))

    def mark(self, key: str, completed: bool = True) -> None:
        """Record an item's completion state.

        Args:
            key: Item key.
            completed: ``True`` for a finished item. Passing ``False`` marks an item as
                started-but-incomplete, which :meth:`reset_incomplete` later clears.

        Returns:
            None. The write is deferred until :meth:`flush`."""
        self.data[key] = completed
        self._dirty += 1

    def needs_flush(self) -> bool:
        """Whether enough marks have accumulated to warrant a write.

        Returns:
            ``True`` once ``flush_every`` marks are pending."""
        return self._dirty >= self.flush_every

    def pending(self, keys: Iterable[str]) -> List[str]:
        """Keys that are not yet marked complete."""
        return [k for k in keys if not self.is_done(k)]

    def reset_incomplete(self, on_reset: Optional[Callable[[str], None]] = None) -> int:
        """Drop every non-complete entry so resume retries it.

        ``on_reset`` lets the caller clean the item's partial output while
        keeping its debug evidence, matching the extraction stages' behaviour.
        """
        removed = 0
        for key, completed in list(self.data.items()):
            if not completed:
                if on_reset is not None:
                    on_reset(key)
                del self.data[key]
                removed += 1
        return removed

    async def flush(self) -> None:
        """Atomically write pending marks without blocking the loop.

        Returns:
            None. Does nothing when no mark is pending, so calling it unconditionally
            at the end of a stage is cheap."""
        if self._dirty:
            await save_log_async(self.path, dict(self.data))
            self._dirty = 0

    def flush_sync(self) -> None:
        """Synchronous counterpart of :meth:`flush`.

        For use outside a coroutine, such as a ``finally`` block in synchronous code or
        an interpreter shutdown hook.

        Returns:
            None. Does nothing when no mark is pending."""
        if self._dirty:
            save_log_sync(self.path, dict(self.data))
            self._dirty = 0

"""Citation & Provenance Sanitization.
-------------------------------------
Utilities for stripping internal document/chunk citation tags from briefs
and standardizing document-level citations in markdown exports.
"""

import re


# Briefs are summaries, not source maps. Strip internal document/chunk references
# whether they came from the pipeline or were repeated by the language model.
_BRACKETED_DOC_CITATION = re.compile(
    r"\[\s*(?:doc_|canon_)[A-Za-z0-9_-]+(?:::[A-Za-z0-9_-]+)?\s*\]",
    re.IGNORECASE,
)
_BRACKETED_CHUNK_CITATION = re.compile(
    r"\[\s*[A-Za-z0-9_-]+::[A-Za-z0-9_-]+\s*\]"
)
_BRACKETED_CHUNK_ID = re.compile(r"\[\s*chunk_[A-Za-z0-9_-]+\s*\]", re.IGNORECASE)
_BARE_DOC_ID = re.compile(
    r"(?<![A-Za-z0-9_-])(?:doc_|canon_)[A-Za-z0-9_-]+(?:::[A-Za-z0-9_-]+)?(?![A-Za-z0-9_-])",
    re.IGNORECASE,
)

# Any suffix after :: is chunk/source-level detail. The exported wiki cites the
# document only, e.g. [doc_00001::chunk_001] -> [doc_00001].
_CHUNK_CITATION = re.compile(r"\[\s*([A-Za-z0-9_-]+)::[A-Za-z0-9_-]+\s*\]")


def clean_brief_text(text: str) -> str:
    """Remove internal provenance IDs and return a single-line brief."""
    if not text:
        return ""

    cleaned = _BRACKETED_DOC_CITATION.sub("", text)
    cleaned = _BRACKETED_CHUNK_CITATION.sub("", cleaned)
    cleaned = _BRACKETED_CHUNK_ID.sub("", cleaned)
    cleaned = _BARE_DOC_ID.sub("", cleaned)
    # A brief is emitted as a quoted, single-line YAML scalar. Collapse all
    # whitespace (including a single newline) so embedded paragraphs cannot
    # break the frontmatter layout.
    return re.sub(r"\s+", " ", cleaned).strip()


def clean_chunk_citations(text: str) -> str:
    """Convert chunk-level bracket citations to document-level citations and deduplicate adjacent ones.
    
    Example:
        [canon_013435abc85f::chunk_000][canon_013435abc85f::chunk_001]
        -> [canon_013435abc85f]
    """
    if not text:
        return ""

    def replace_citation_groups(match):
        group = match.group(0)
        canon_ids = re.findall(r"\[\s*([A-Za-z0-9_-]+)(?:::[A-Za-z0-9_-]+)?\s*\]", group)
        unique = []
        for cid in canon_ids:
            if cid not in unique:
                unique.append(cid)
        return " " + " ".join(f"[{cid}]" for cid in unique) + " "

    pattern = r"(?:\[\s*[A-Za-z0-9_-]+(?:::[A-Za-z0-9_-]+)?\s*\]\s*)+"
    cleaned = re.sub(pattern, replace_citation_groups, text)
    cleaned = re.sub(r" +", " ", cleaned)
    cleaned = re.sub(r"\s+([.,;:!?])", r"\1", cleaned)
    return cleaned.strip()

"""Corpus Processing, Deduplication & Chunking.
----------------------------------------------
Loads raw corpus files (.json or .jsonl), normalizes document records to doc_XXXXX
format, deduplicates identical documents, and builds token sliding-window units.
"""

import hashlib
import json
from pathlib import Path
from typing import Any, Dict, List, Optional, Tuple

# Accepted spellings, in priority order.
CONTENT_FIELDS = ("contents", "content", "text", "body")
ID_FIELDS = ("id", "doc_id", "document_id")

# Chunking defaults. These are part of the cache fingerprint, so changing them
# invalidates a cached chunking run rather than silently mixing window sizes.
DEFAULT_CHUNK_SIZE = 5000
DEFAULT_CHUNK_OVERLAP = 500

# Used as the output directory name when no domain hint is supplied.
DEFAULT_RUN_NAME = "default"

MANIFEST_VERSION = 2


class CorpusError(ValueError):
    """Raised when a corpus file cannot be read or is missing required fields.

    The message always names the file and, where applicable, the offending line
    or record index, so a bad corpus is diagnosable without a debugger.
    """


def short_hash(text: str) -> str:
    """Return the 12-character MD5 prefix used for canonical document ids.

    Args:
        text: Document content.

    Returns:
        The first 12 hex characters of the content's MD5 digest.

    Note:
        Citation ids are ``canon_<short_hash>``, so this algorithm is part of the
        on-disk format. Changing it would orphan every existing citation.
    """
    return hashlib.md5(text.encode("utf-8")).hexdigest()[:12]


def canonical_id(content: str) -> str:
    """Return the canonical document id for a piece of content.

    Args:
        content: Document content.

    Returns:
        ``canon_<hash>`` -- the id used in staging paths and citations.
    """
    return f"canon_{short_hash(content)}"


def fix_mojibake(text: str) -> str:
    """Repair ids that were decoded as cp437 when they were really UTF-8.

    Some source corpora arrive with ids whose non-ASCII characters were decoded
    with the wrong codec. This reverses that when it can, and otherwise returns
    the text unchanged rather than raising.

    Args:
        text: A raw identifier from the corpus.

    Returns:
        The repaired identifier, or the original if it was not mis-decoded.
    """
    try:
        return text.encode("cp437").decode("utf-8")
    except (UnicodeEncodeError, UnicodeDecodeError, AttributeError):
        return text


def file_digest(path: Path, block_size: int = 1024 * 1024) -> str:
    """Return the SHA-256 of a file, read in blocks so large corpora stream.

    Args:
        path: File to hash.
        block_size: Bytes read per iteration.

    Returns:
        Lowercase hex digest.
    """
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(block_size), b""):
            digest.update(block)
    return digest.hexdigest()


def detect_format(path: Path) -> str:
    """Decide whether a corpus file is a JSON array or JSON Lines.

    The file extension is trusted first because it is what the caller declared;
    if the extension is ambiguous the first non-whitespace byte decides.

    Args:
        path: Corpus file.

    Returns:
        ``"json_array"`` or ``"jsonl"``.

    Raises:
        CorpusError: If the file is empty or starts with neither ``[`` nor ``{``.
    """
    suffix = path.suffix.lower()
    if suffix == ".jsonl":
        return "jsonl"
    if suffix == ".json":
        return "json_array"

    with path.open("r", encoding="utf-8") as handle:
        head = handle.read(4096)
    first = next((c for c in head if not c.isspace()), "")
    if first == "[":
        return "json_array"
    if first == "{":
        return "jsonl"
    raise CorpusError(
        f"{path} does not look like a corpus: expected a JSON array starting "
        f"with '[' or JSON Lines starting with '{{', found {first!r}.")


def _normalise(record: Any, index: int, source: Path) -> Dict[str, str]:
    """Normalise one raw corpus record to ``{"id": ..., "content": ...}``.

    Args:
        record: A decoded JSON value from the corpus.
        index: Position of the record, used only in error messages.
        source: Corpus path, used only in error messages.

    Returns:
        A dict with exactly ``id`` (str) and ``content`` (str).

    Raises:
        CorpusError: If the record is not an object, has no content field, or
            has empty content.
    """
    if not isinstance(record, dict):
        raise CorpusError(f"{source} record #{index} is a {type(record).__name__}, "
                          f"expected a JSON object.")

    content = None
    for field in CONTENT_FIELDS:
        if field in record and record[field] is not None:
            content = record[field]
            break
    if content is None:
        raise CorpusError(
            f"{source} record #{index} has no content field. Looked for one of "
            f"{CONTENT_FIELDS}; keys present: {sorted(record)}.")
    if not isinstance(content, str):
        raise CorpusError(f"{source} record #{index} content is a "
                          f"{type(content).__name__}, expected a string.")
    if not content.strip():
        raise CorpusError(f"{source} record #{index} has empty content.")

    raw_id = None
    for field in ID_FIELDS:
        if field in record and record[field] is not None:
            raw_id = record[field]
            break
    if raw_id is None:
        # An id is not strictly required: fall back to the record's position so
        # the citation mapping still has a usable key.
        raw_id = f"doc_{index:06d}"

    return {"id": fix_mojibake(str(raw_id)), "content": content}


def load_corpus(path: Path) -> List[Dict[str, str]]:
    """Read a corpus file and normalise every record.

    Accepts a single JSON array or JSON Lines, auto-detected. Blank lines in
    JSONL input are skipped.

    Args:
        path: Corpus file (``.json`` or ``.jsonl``).

    Returns:
        A list of ``{"id": str, "content": str}`` dicts, in file order.

    Raises:
        CorpusError: If the file is missing, malformed, or a record lacks
            usable content. The message names the file and the record.
    """
    path = Path(path)
    if not path.is_file():
        raise CorpusError(f"Corpus file not found: {path}")

    fmt = detect_format(path)
    raw: List[Any] = []

    if fmt == "json_array":
        try:
            with path.open("r", encoding="utf-8") as handle:
                data = json.load(handle)
        except json.JSONDecodeError as exc:
            # A .json file that is really JSON Lines is a common mistake; say so.
            raise CorpusError(
                f"{path} is not valid JSON ({exc}). If it holds one JSON object "
                f"per line, rename it to .jsonl.") from exc
        if isinstance(data, dict):
            # Tolerate {"documents": [...]} / {"data": [...]} wrappers.
            for key in ("documents", "data", "docs", "records"):
                if isinstance(data.get(key), list):
                    data = data[key]
                    break
        if not isinstance(data, list):
            raise CorpusError(f"{path} holds a {type(data).__name__}, expected a JSON array.")
        raw = data
    else:
        with path.open("r", encoding="utf-8") as handle:
            for line_no, line in enumerate(handle, start=1):
                if not line.strip():
                    continue
                try:
                    raw.append(json.loads(line))
                except json.JSONDecodeError as exc:
                    raise CorpusError(f"{path} line {line_no} is not valid JSON: {exc}") from exc

    if not raw:
        raise CorpusError(f"{path} contains no records.")

    return [_normalise(record, index, path) for index, record in enumerate(raw)]


def deduplicate(docs: List[Dict[str, str]]) -> Tuple[Dict[str, str], Dict[str, str]]:
    """Index documents by their clean document id (doc_XXXXX), deduplicating if needed.

    Args:
        docs: Normalised records from :func:`load_corpus`.

    Returns:
        A 2-tuple of:
            * ``canon_docs`` -- ``{doc_id: content}``, one entry per unique
              document, in first-seen order.
            * ``doc_mapping`` -- ``{original_id: doc_id}`` for every record
              in the corpus. This is the citation mapping written to ``doc_mapping.json``.
    """
    canon_docs: Dict[str, str] = {}
    doc_mapping: Dict[str, str] = {}
    for idx, doc in enumerate(docs, start=1):
        raw_id = doc.get("id") or ""
        # If the input id is already a doc_XXXXX, keep it; otherwise assign doc_XXXXX format
        if raw_id.startswith("doc_"):
            doc_id = raw_id
        else:
            doc_id = f"doc_{idx:05d}"
        canon_docs.setdefault(doc_id, doc["content"])
        doc_mapping[raw_id if raw_id else doc_id] = doc_id
    return canon_docs, doc_mapping


def chunk_document(doc_id: str, content: str, tokenizer: Any,
                   chunk_size: int = DEFAULT_CHUNK_SIZE,
                   overlap: int = DEFAULT_CHUNK_OVERLAP) -> List[Dict[str, Any]]:
    """Split one document into overlapping token windows with surrounding context.

    Each unit carries the window itself plus the tokens immediately before and
    after it, so the extractor can resolve references that straddle a boundary.

    Args:
        doc_id: Canonical document id.
        content: Full document text.
        tokenizer: Anything exposing ``encode(str) -> list`` and
            ``decode(list, skip_special_tokens=bool) -> str``.
        chunk_size: Window size in tokens.
        overlap: Tokens repeated on each side of a window, also used as the
            context window.

    Returns:
        A list of unit dicts with keys ``doc_id``, ``chunk_id``,
        ``current_chunk``, ``context_before`` and ``context_after``. Empty for a
        document that tokenises to nothing.

    Raises:
        ValueError: If ``overlap >= chunk_size``, which would make the stride
            non-positive and loop forever.
    """
    if overlap >= chunk_size:
        raise ValueError(f"chunk overlap ({overlap}) must be smaller than "
                         f"chunk size ({chunk_size}).")

    tokens = tokenizer.encode(content)
    total_tokens = len(tokens)
    if total_tokens == 0:
        return []

    chunks: List[Dict[str, Any]] = []
    step = chunk_size - overlap
    chunk_idx = 0

    for start in range(0, total_tokens, step):
        end = min(start + chunk_size, total_tokens)
        chunk_tokens = tokens[start:end]
        before_tokens = tokens[max(0, start - overlap):start]
        after_tokens = tokens[end:min(total_tokens, end + overlap)]

        chunks.append({
            "doc_id": doc_id,
            "chunk_id": f"chunk_{chunk_idx:03d}",
            "current_chunk": tokenizer.decode(chunk_tokens, skip_special_tokens=True),
            "context_before": tokenizer.decode(before_tokens, skip_special_tokens=True),
            "context_after": tokenizer.decode(after_tokens, skip_special_tokens=True),
        })
        chunk_idx += 1
        if end >= total_tokens:
            break

    return chunks


def whole_document_unit(doc_id: str, content: str) -> Dict[str, Any]:
    """Return a single extraction unit holding an entire document.

    Used when chunking is disabled. Keeping the same key names as
    :func:`chunk_document` means no downstream stage has to know whether the
    corpus was chunked.

    Args:
        doc_id: Canonical document id.
        content: Full document text.

    Returns:
        A unit dict with the same shape as a chunk, with empty context.
    """
    return {
        "doc_id": doc_id,
        "chunk_id": "chunk_000",
        "current_chunk": content,
        "context_before": "",
        "context_after": "",
    }


def build_units(canon_docs: Dict[str, str], tokenizer: Optional[Any], *,
                chunking: bool,
                chunk_size: int = DEFAULT_CHUNK_SIZE,
                overlap: int = DEFAULT_CHUNK_OVERLAP) -> List[Dict[str, Any]]:
    """Turn canonical documents into the extraction units Pass 1 consumes.

    Args:
        canon_docs: ``{canonical_id: content}`` from :func:`deduplicate`.
        tokenizer: Required only when ``chunking`` is true. When chunking is off
            this may be ``None``, so a corpus can be processed without
            ``transformers`` installed.
        chunking: Whether to split documents into overlapping token windows.
        chunk_size: Window size in tokens (ignored when ``chunking`` is false).
        overlap: Context/overlap in tokens (ignored when ``chunking`` is false).

    Returns:
        A list of unit dicts, grouped by document in ``canon_docs`` order.

    Raises:
        ValueError: If chunking is requested without a tokenizer.
    """
    if chunking and tokenizer is None:
        raise ValueError("Chunking requires a tokenizer; pass one or disable chunking.")

    from tqdm import tqdm
    units: List[Dict[str, Any]] = []
    desc = "Tokenizing & chunking documents" if chunking else "Preparing document units"
    for doc_id, content in tqdm(canon_docs.items(), desc=desc):
        if chunking:
            units.extend(chunk_document(doc_id, content, tokenizer,
                                        chunk_size=chunk_size, overlap=overlap))
        else:
            units.append(whole_document_unit(doc_id, content))
    return units


def cache_manifest(source_path: Path, *, tokenizer_name: Optional[str],
                   chunking: bool, chunk_size: int, overlap: int) -> Dict[str, Any]:
    """Fingerprint everything a cached chunking depends on.

    A cache is reused only when this manifest matches exactly, so editing the
    corpus, switching tokenizer, or changing the window size forces a rebuild
    instead of silently mixing old and new units.

    Args:
        source_path: Corpus file the units were built from.
        tokenizer_name: Tokenizer/model name, or ``None`` when not chunking.
        chunking: Whether chunking was enabled.
        chunk_size: Window size in tokens.
        overlap: Overlap in tokens.

    Returns:
        A JSON-serialisable dict suitable for writing to
        ``chunks_cache_manifest.json``.
    """
    return {
        "format_version": MANIFEST_VERSION,
        "source_sha256": file_digest(source_path),
        "chunking": bool(chunking),
        "tokenizer": tokenizer_name if chunking else None,
        "chunk_size": chunk_size if chunking else None,
        "overlap": overlap if chunking else None,
    }


def resolve_corpus_path(args: Any) -> Path:
    """Work out which corpus file a stage should read from --input_path.

    Args:
        args: Parsed CLI namespace.

    Returns:
        The resolved corpus path.

    Raises:
        CorpusError: If --input_path is missing or not a file.
    """
    input_path = getattr(args, "input_path", None)
    if input_path:
        path = Path(input_path)
        if not path.is_file():
            raise CorpusError(f"Corpus file not found: {path}")
        return path

    raise CorpusError("No corpus specified. Pass --input_path /path/to/documents.json (or .jsonl).")

RUN_CONFIG_NAME = "run_config.json"


def save_run_config(run_dir: Path, config: Dict[str, Any]) -> Path:
    """Persist how this run's corpus was prepared.

    Args:
        run_dir: The run's output directory (``<wiki_base_dir>/<domain>/``).
        config: JSON-serialisable settings; at minimum ``corpus``, ``domain``,
            ``chunking``, ``chunk_size`` and ``chunk_overlap``.

    Returns:
        The path written, ``<run_dir>/run_config.json``.
    """
    from .atomic_io import atomic_write_json

    run_dir = Path(run_dir)
    run_dir.mkdir(parents=True, exist_ok=True)
    path = run_dir / RUN_CONFIG_NAME
    atomic_write_json(path, config)
    return path


def load_run_config(run_dir: Path) -> Dict[str, Any]:
    """Read a run's saved corpus configuration.

    Args:
        run_dir: The run's output directory.

    Returns:
        The saved config dict, or an empty dict when the file does not exist or
        cannot be decoded. Callers should treat a missing config as "unknown"
        rather than as an error, so stages still run on output trees produced
        before this file existed.
    """
    path = Path(run_dir) / RUN_CONFIG_NAME
    if not path.is_file():
        return {}
    try:
        with path.open("r", encoding="utf-8") as handle:
            data = json.load(handle)
    except (OSError, json.JSONDecodeError):
        return {}
    return data if isinstance(data, dict) else {}


def chunking_enabled(run_dir: Path, default: bool = True) -> bool:
    """Report whether the run that produced ``run_dir`` chunked its corpus.

    Args:
        run_dir: The run's output directory.
        default: Value to return when no run config is recorded. Defaults to
            ``True`` so pre-existing BRIGHT trees keep their current behaviour.

    Returns:
        ``True`` when documents were split into chunks, ``False`` when each
        document was a single unit.
    """
    config = load_run_config(run_dir)
    if "chunking" not in config:
        return default
    return bool(config["chunking"])

def load_cached_chunks(cache_dir: Path) -> Dict[str, Dict[str, Any]]:

    chunks_cache_file = Path(cache_dir) / "chunks_cache.json"

"""Distribution statistics and reporting helpers for pipeline concepts and links.

Implements F-36 shared stats helpers:

- print_link_distribution

from typing import Any, Dict, List


def print_concept_word_distribution(concepts: List[Dict[str, Any]], title_label: str = "GLOBAL CONCEPTS") -> None:
    """Compute and print word count distribution statistics for concept descriptions."""
    if not concepts:
        print(f"\n--- {title_label} WORD COUNT DISTRIBUTION --- (0 concepts)")
        return

    word_counts = []
    for c in concepts:
        desc = c.get("description", "")
        word_counts.append(len(desc.split()))

    word_counts.sort()
    total = len(word_counts)
    mean_wc = sum(word_counts) / total
    median_wc = word_counts[total // 2]
    min_wc = word_counts[0]
    max_wc = word_counts[-1]
    p10 = word_counts[int(total * 0.10)]
    p25 = word_counts[int(total * 0.25)]
    p75 = word_counts[int(total * 0.75)]
    p90 = word_counts[int(total * 0.90)]

    under_60 = sum(1 for w in word_counts if w < 60)
    under_100 = sum(1 for w in word_counts if w < 100)
    over_900 = sum(1 for w in word_counts if w > 900)

    print(f"\n{'='*70}")
    print(f"📊 {title_label} WORD COUNT DISTRIBUTION (N={total})")
    print(f"{'='*70}")
    print(f"  Min: {min_wc:<5} | P10: {p10:<5} | P25: {p25:<5} | Median: {median_wc:<5} | Mean: {mean_wc:.1f}")
    print(f"  P75: {p75:<5} | P90: {p90:<5} | Max: {max_wc:<5}")
    print(f"  Thin (<60 words):  {under_60} ({under_60/total*100:.1f}%)")
    print(f"  Brief (<100 words): {under_100} ({under_100/total*100:.1f}%)")
    print(f"  Large (>900 words): {over_900} ({over_900/total*100:.1f}%)")
    print(f"{'='*70}\n")


def print_link_distribution(concepts: List[Dict[str, Any]], title_label: str = "RELATED LINKS") -> None:
    """Compute and print histogram distribution of related links across all concepts."""
    if not concepts:
        print(f"\n[{title_label}] No concepts found.")
        return

    counts = [len(c.get("outbound_relations", [])) for c in concepts]
    total = len(counts)
    avg_links = sum(counts) / total if total else 0.0
    max_count = max(counts) if counts else 0

    bins = [("0 links", 0, 0)]
    step = 5
    curr = 1
    while curr <= max(max_count, 1):
        upper = curr + step - 1
        bins.append((f"{curr} - {upper} links", curr, upper))
        curr += step

    bin_counts = {label: 0 for label, _, _ in bins}
    for count in counts:
        for label, low, high in bins:
            if low <= count <= high:
                bin_counts[label] += 1
                break

    print(f"\n{'='*70}")
    print(f"🔗 {title_label} DISTRIBUTION (Concepts: {total})")
    print(f"  Average Links per Concept: {avg_links:.2f} | Max Links: {max_count} | 0 Links: {bin_counts['0 links']}")
    print(f"{'-'*70}")
    for label, _, _ in bins:
        c_count = bin_counts[label]
        pct = (c_count / total * 100) if total else 0.0
        bar = "█" * int(pct / 4)
        print(f"  {label:<20} {c_count:<8} {pct:>5.1f}%   {bar}")
    print(f"{'='*70}\n")

"""Shared concept manipulation operations.

Implements F-36:
- fuse_descriptions: deduplicating and combining sentences while preserving citation tags and paragraph structure.
- union_aliases: combining aliases without duplicates.
- generate_stable_slug: collision-free unique slugs (F-12).
"""

import hashlib
import re
from typing import Iterable, List, Optional, Set

from utils.atomic_io import sanitize_filename


def generate_stable_slug(title: str, existing_slugs: Optional[Set[str]] = None) -> str:
    """Generate a collision-free filesystem and markdown slug for a concept title (F-12)."""
    base = sanitize_filename(title) or "unnamed_concept"
    base = base[:80].strip("-_")
    if not base:
        base = "concept"

    if existing_slugs is None:
        return base

    if base not in existing_slugs:
        existing_slugs.add(base)
        return base

    # Append deterministic 6-character SHA1 hash of full title on collision
    h = hashlib.sha1(title.strip().casefold().encode("utf-8")).hexdigest()[:6]
    candidate = f"{base}-{h}"
    if candidate not in existing_slugs:
        existing_slugs.add(candidate)
        return candidate

    counter = 1
    while f"{candidate}-{counter}" in existing_slugs:
        counter += 1
    final_slug = f"{candidate}-{counter}"
    existing_slugs.add(final_slug)
    return final_slug


def fuse_descriptions(descriptions: List[str], max_tokens: int = 1500) -> str:
    """Fuse multiple concept descriptions, deduplicating sentences while retaining citations."""
    seen_sentences: Set[str] = set()
    fused_sentences: List[str] = []

    for desc in descriptions:
        if not desc:
            continue
        # Split into sentences preserving trailing delimiters
        sentences = re.split(r'(?<=[.!?])\s+', desc.strip())
        for s in sentences:
            s_clean = s.strip()
            if not s_clean:
                continue
            # Compare lowercase text without brackets/citations
            norm = re.sub(r"\[doc_\d+\]", "", s_clean).casefold().strip()
            norm = " ".join(norm.split())
            if norm and norm not in seen_sentences:
                seen_sentences.add(norm)
                fused_sentences.append(s_clean)

    fused = " ".join(fused_sentences)
    return fused


def union_aliases(alias_lists: Iterable[List[str]]) -> List[str]:
    """Union lists of aliases, deduplicating case-insensitively while preserving clean display."""
    seen = set()
    result = []
    for alist in alias_lists:
        for a in alist:
            norm = a.strip().casefold()
            if norm and norm not in seen:
                seen.add(norm)
                result.append(a.strip())
    return result

"""JSON Extraction & Fallback Repair.
------------------------------------
Extracts JSON payloads from model output blocks and repairs truncated JSON responses.
"""

import re

_FENCED_OBJECT = re.compile(r"```(?:json)?\s*(\{.*\})\s*```", re.DOTALL)
_FENCED_OBJECT_OR_ARRAY = re.compile(r"```(?:json)?\s*(\{.*\}|\[.*\])\s*```", re.DOTALL)
_BARE_OBJECT = re.compile(r"(\{.*\})", re.DOTALL)


def repair_truncated_json(text: str) -> str:
    """Attempt to repair JSON truncated by context or max_tokens limits.

    Finds the first '{' or '[', strips unclosed trailing string tokens, and
    closes any open brackets/braces.
    """
    if not text:
        return ""
    start = text.find("{")
    start_arr = text.find("[")
    if start == -1 or (start_arr != -1 and start_arr < start):
        start = start_arr
    if start == -1:
        return text

    snippet = text[start:].strip()
    # Strip trailing partial escapes or incomplete tokens
    snippet = re.sub(r'\\+$', '', snippet)
    # If ends in a dangling key/string like: "foo", "bar or "foo": "bar
    # Count unescaped quotes to check if inside a string
    quote_count = len(re.findall(r'(?<!\\)"', snippet))
    if quote_count % 2 != 0:
        snippet += '"'

    # Strip trailing comma
    snippet = re.sub(r',\s*$', '', snippet)

    # Balance brackets and braces
    stack = []
    in_string = False
    escape = False
    for ch in snippet:
        if escape:
            escape = False
            continue
        if ch == '\\':
            escape = True
            continue
        if ch == '"':
            in_string = not in_string
            continue
        if not in_string:
            if ch in ('{', '['):
                stack.append('}' if ch == '{' else ']')
            elif ch in ('}', ']'):
                if stack and stack[-1] == ch:
                    stack.pop()

    # Append needed closing brackets in reverse order
    while stack:
        snippet += stack.pop()

    return snippet


def clean_json_string(raw_response: str, *, allow_bare_object: bool = False,
                      allow_array: bool = False) -> str:
    """Extract a clean JSON string from a raw model response.

    Falls back to the stripped raw response when no JSON block is found; then
    attempts repair if truncated.
    """
    if not raw_response:
        return ""

    pattern = _FENCED_OBJECT_OR_ARRAY if allow_array else _FENCED_OBJECT
    match = pattern.search(raw_response)
    if match:
        return repair_truncated_json(match.group(1).strip())

    if allow_bare_object:
        bare = _BARE_OBJECT.search(raw_response)
        if bare:
            return repair_truncated_json(bare.group(1).strip())

    return repair_truncated_json(raw_response.strip())

"""Verification reporting and Pass 1 vs Pass 2 quality comparison metrics.

Aggregates extraction quality metrics (concept coverage, precision, citation accuracy)
between initial Pass 1 extraction and Pass 2 refinement. Writes verification summaries
and comparative reports to disk.
"""

from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Any, Dict, List, Optional

from .atomic_io import atomic_write_json, read_json

PASS1_RESULT = "verification_pass1.json"
PASS2_RESULT = "verification_pass2.json"
PASS1_LABEL = "PASS 1 (raw extraction)"
PASS2_LABEL = "PASS 2 (after refinement)"
SUMMARY_FILENAME = "verification_summary.json"


@dataclass
class VerificationMetrics:
    """Aggregate quality metrics for one verification run."""

    label: str
    result_filename: str
    total_chunks: int = 0
    average_coverage: float = 0.0
    sufficient_count: int = 0
    pass_rate: float = 0.0
    missed_concepts: int = 0
    missed_facts: int = 0
    synthetic_count: int = 0
    sample_missed_concepts: List[str] = None  # type: ignore[assignment]
    sample_missed_facts: List[str] = None  # type: ignore[assignment]

    def __post_init__(self) -> None:
        """Replace the mutable-default sentinels with empty lists.

        ``None`` is used as the field default because a dataclass cannot safely default
        to a shared list. This runs after ``__init__`` so callers may omit both sample
        fields and still get independent, mutable lists.

        Returns:
            None."""
        if self.sample_missed_concepts is None:
            self.sample_missed_concepts = []
        if self.sample_missed_facts is None:
            self.sample_missed_facts = []

    def to_dict(self) -> Dict[str, Any]:
        """Render the metrics as a plain dict for JSON output.

        Returns:
            Every field of this dataclass, via ``dataclasses.asdict``. Written into
            ``verification_summary.json`` so the two phases can be compared offline."""
        return asdict(self)


def collect_metrics(staging_dir: Path, *, label: str,
                    result_filename: str = PASS1_RESULT,
                    target_keys: Optional[set] = None) -> VerificationMetrics:
    """Aggregate every per-chunk verification result under ``staging_dir``."""
    metrics = VerificationMetrics(label=label, result_filename=result_filename)
    scores: List[float] = []
    missed_concepts: List[str] = []
    missed_facts: List[str] = []

    staging_dir = Path(staging_dir)
    if not staging_dir.is_dir():
        return metrics

    for chunk_dir in sorted(staging_dir.glob("*/*")):
        key = f"{chunk_dir.parent.name}::{chunk_dir.name}"
        if target_keys is not None and key not in target_keys:
            continue

        result_file = chunk_dir / result_filename
        if not result_file.is_file():
            # Legacy fallback for pass 1
            if result_filename == "verification_pass1.json":
                legacy_file = chunk_dir / "verification.json"
                if legacy_file.is_file():
                    result_file = legacy_file
                else:
                    continue
            else:
                continue

        data = read_json(result_file, default=None)
        if not isinstance(data, dict) or "coverage_score" not in data:
            continue
        if data.get("synthetic", False):
            metrics.synthetic_count += 1
            continue
        metrics.total_chunks += 1
        try:
            scores.append(float(data.get("coverage_score", 0.0)))
        except (TypeError, ValueError):
            scores.append(0.0)
        if data.get("is_sufficient", False):
            metrics.sufficient_count += 1
        chunk_concepts = data.get("missing_concepts", []) or []
        chunk_facts = data.get("missing_facts", []) or []
        missed_concepts.extend(str(c) for c in chunk_concepts)
        missed_facts.extend(str(f) for f in chunk_facts)

    if metrics.total_chunks:
        metrics.average_coverage = sum(scores) / len(scores)
        metrics.pass_rate = (metrics.sufficient_count / metrics.total_chunks) * 100.0
    metrics.missed_concepts = len(missed_concepts)
    metrics.missed_facts = len(missed_facts)
    metrics.sample_missed_concepts = missed_concepts[:5]
    metrics.sample_missed_facts = missed_facts[:5]
    return metrics


def format_report(domain: str, metrics: VerificationMetrics) -> str:
    """Render one run in the established summary format."""
    lines = [
        "=" * 60,
        f"VERIFICATION SUMMARY REPORT [{domain.upper()}] - {metrics.label}",
        "=" * 60,
    ]
    if metrics.total_chunks == 0:
        lines.append(f"No {metrics.result_filename} results found to summarize.")
        return "\n".join(lines)

    lines.extend([
        f"Total Chunks Verified : {metrics.total_chunks}",
        f"Average Coverage Score: {metrics.average_coverage:.3f} / 1.000",
        f"Pass Rate (Sufficient): {metrics.pass_rate:.1f}% "
        f"({metrics.sufficient_count}/{metrics.total_chunks})",
        f"Total Missed Concepts : {metrics.missed_concepts}",
        f"Total Missed Facts    : {metrics.missed_facts}",
    ])
    if metrics.sample_missed_concepts:
        lines.append("\nSample Missed Concepts:")
        lines.extend(f"  - {c}" for c in metrics.sample_missed_concepts)
    if metrics.sample_missed_facts:
        lines.append("\nSample Missed Facts:")
        lines.extend(f"  - {f}" for f in metrics.sample_missed_facts)
    return "\n".join(lines)


def _delta_pct(before: float, after: float) -> Optional[float]:
    """Express a change as a percentage of the earlier value.

    Args:
        before: The Pass 1 value.
        after: The Pass 2 value.

    Returns:
        The signed percentage change, or ``None`` when ``before`` is zero -- a
        division that would be meaningless, which the report renders as "n/a"
        rather than as an infinite improvement."""
    if before == 0:
        return None
    return ((after - before) / before) * 100.0


def _fmt_pct(value: Optional[float]) -> str:
    """Format a percentage delta for the improvement table.

    Args:
        value: A signed percentage, or ``None`` when the delta is undefined.

    Returns:
        ``"+12.3%"``-style text, or ``"n/a"`` for ``None``."""
    return "n/a" if value is None else f"{value:+.1f}%"


def build_improvement(before: VerificationMetrics, after: VerificationMetrics) -> Dict[str, Any]:
    """Compute the Pass 1 -> Pass 2 deltas."""
    return {
        "coverage_before": round(before.average_coverage, 4),
        "coverage_after": round(after.average_coverage, 4),
        "coverage_delta": round(after.average_coverage - before.average_coverage, 4),
        "pass_rate_before": round(before.pass_rate, 2),
        "pass_rate_after": round(after.pass_rate, 2),
        "pass_rate_delta_points": round(after.pass_rate - before.pass_rate, 2),
        "sufficient_before": before.sufficient_count,
        "sufficient_after": after.sufficient_count,
        "chunks_verified": after.total_chunks,
        "missed_concepts_before": before.missed_concepts,
        "missed_concepts_after": after.missed_concepts,
        "missed_concepts_delta": after.missed_concepts - before.missed_concepts,
        "missed_concepts_change_pct": _delta_pct(before.missed_concepts, after.missed_concepts),
        "missed_facts_before": before.missed_facts,
        "missed_facts_after": after.missed_facts,
        "missed_facts_delta": after.missed_facts - before.missed_facts,
        "missed_facts_change_pct": _delta_pct(before.missed_facts, after.missed_facts),
    }


def format_improvement(domain: str, improvement: Dict[str, Any]) -> str:
    """Render the side-by-side improvement block."""
    cov_b = improvement["coverage_before"]
    cov_a = improvement["coverage_after"]
    cov_d = improvement["coverage_delta"]
    pr_b = improvement["pass_rate_before"]
    pr_a = improvement["pass_rate_after"]
    pr_d = improvement["pass_rate_delta_points"]
    mc_b = improvement["missed_concepts_before"]
    mc_a = improvement["missed_concepts_after"]
    mf_b = improvement["missed_facts_before"]
    mf_a = improvement["missed_facts_after"]
    chunks = improvement["chunks_verified"]

    return "\n".join([
        "=" * 60,
        f"VERIFICATION IMPROVEMENT [{domain.upper()}]  PASS 1 -> PASS 2",
        "=" * 60,
        f"Chunks Compared       : {chunks}",
        f"{'Metric':<22}{'Before':>10}{'After':>10}{'Delta':>14}",
        "-" * 60,
        f"{'Average Coverage':<22}{cov_b:>10.3f}{cov_a:>10.3f}{cov_d:>+14.3f}",
        f"{'Pass Rate (%)':<22}{pr_b:>10.1f}{pr_a:>10.1f}{pr_d:>+13.1f}p",
        f"{'Sufficient Chunks':<22}{improvement['sufficient_before']:>10}"
        f"{improvement['sufficient_after']:>10}"
        f"{improvement['sufficient_after'] - improvement['sufficient_before']:>+14}",
        f"{'Missed Concepts':<22}{mc_b:>10}{mc_a:>10}"
        f"{improvement['missed_concepts_delta']:>+10} "
        f"{_fmt_pct(improvement['missed_concepts_change_pct']):>3}",
        f"{'Missed Facts':<22}{mf_b:>10}{mf_a:>10}"
        f"{improvement['missed_facts_delta']:>+10} "
        f"{_fmt_pct(improvement['missed_facts_change_pct']):>3}",
        "=" * 60,
    ])

def resolve_settings(args: argparse.Namespace, stage: str) -> StageSettings:
    """Resolve runtime settings ONLY from CLI flags or config.json.

    Precedence: CLI flag > config.json (stage-specific or top-level).
    There are NO hardcoded python fallbacks; all settings must be explicitly defined.
    """
    # Read config.json
    cfg = _read_config()
    stage_concurrency_cfg = cfg.get("stage_concurrency", {}) if isinstance(cfg.get("stage_concurrency"), dict) else {}
    stage_cfg = stage_concurrency_cfg.get(stage, {}) if isinstance(stage_concurrency_cfg, dict) else {}
    if not isinstance(stage_cfg, dict):
        stage_cfg = {"concurrency_per_server": int(stage_cfg)}

    default_profile = STAGE_PROFILES.get(stage)

    def get_setting(cli_val: Any, key: str, default_val: Any = None) -> Any:
        if cli_val not in (None, 0, 0.0, ""):
            return cli_val
        if key in stage_cfg:
            return stage_cfg[key]
        if key in cfg:
            return cfg[key]
        if default_val is not None:
            return default_val
        raise ValueError(
            f"Missing required parameter for stage '{stage}': '{key}'! "
            f"Specify via CLI argument or set '{key}' in config.json."
        )

    # 1. Servers (Accepts list of URLs, or dict of {url: concurrency})
    raw_servers = getattr(args, "servers", None) or cfg.get("servers")
    if not raw_servers:
        raise ValueError("Missing 'servers'! Specify via --servers or set 'servers' in config.json.")

    # 2. Model
    model = getattr(args, "model", None) or cfg.get("model_name") or cfg.get("model")
    if not model:
        raise ValueError("Missing 'model'! Specify via --model or set 'model_name' in config.json.")

    # 3. Determine Concurrency
    cli_concurrency = getattr(args, "concurrency_per_server", None)
    if cli_concurrency not in (None, 0, 0.0, ""):
        default_concurrency = int(cli_concurrency)
    elif "concurrency_per_server" in stage_cfg:
        default_concurrency = int(stage_cfg["concurrency_per_server"])
    elif isinstance(stage_cfg, dict) and stage_cfg.get("concurrency"):
        default_concurrency = int(stage_cfg["concurrency"])
    elif "concurrency_per_server" in cfg:
        default_concurrency = int(cfg["concurrency_per_server"])
    elif isinstance(raw_servers, dict) and raw_servers:
        default_concurrency = max(int(c) for c in raw_servers.values())
    elif default_profile:
        default_concurrency = default_profile.concurrency_per_server
    else:
        default_concurrency = 16

    # Build per-server concurrency mapping for this specific stage:
    # 1. stage_cfg has specific server URLs: {"http://cn-dgx...": 32, "http://cn-a40...": 16}
    # 2. raw_servers is a list of URLs: ["http://cn-a40:8000/v1", "http://cn-dgx:8000/v1"]
    # 3. raw_servers is a dict: {"http://cn-a40:8000/v1": 16, "http://cn-dgx:8000/v1": 32}
    if isinstance(stage_cfg, dict) and any(isinstance(v, (int, float)) and k.startswith("http") for k, v in stage_cfg.items()):
        server_list = list(raw_servers.keys()) if isinstance(raw_servers, dict) else list(raw_servers)
        servers = {s: int(stage_cfg.get(s, default_concurrency)) for s in server_list}
        concurrency = max(servers.values())
    elif isinstance(raw_servers, dict):
        if "concurrency_per_server" in stage_cfg or (isinstance(stage_cfg, dict) and stage_cfg.get("concurrency")):
            servers = {s: default_concurrency for s in raw_servers}
            concurrency = default_concurrency
        else:
            servers = raw_servers
            concurrency = default_concurrency
    else:
        servers = raw_servers
        concurrency = default_concurrency

    # 4. Max Tokens (Checks CLI -> stage_cfg -> cfg -> default STAGE_PROFILES)
    default_max_tokens = default_profile.max_tokens if default_profile else 4096
    max_tokens = int(get_setting(
        getattr(args, "max_tokens", None),
        "max_tokens",
        default_val=default_max_tokens
    ))

    # 5. Request Timeout (Checks CLI -> stage_cfg -> cfg -> default STAGE_PROFILES)
    default_timeout = default_profile.request_timeout if default_profile else 600.0
    timeout = float(get_setting(
        getattr(args, "request_timeout", None),
        "request_timeout",
        default_val=default_timeout
    ))

    # Optional / Tuning parameters (attempts & budget)
    attempts = getattr(args, "max_attempts", 0) or stage_cfg.get("attempts") or cfg.get("attempts") or (default_profile.attempts if default_profile else 4)
    budget = getattr(args, "stage_budget_minutes", 0.0) or stage_cfg.get("budget_minutes") or cfg.get("budget_minutes") or (default_profile.budget_minutes if default_profile else 300.0)

    profile = StageProfile(
        key=stage,
        concurrency_per_server=concurrency,
        request_timeout=timeout,
        connect_timeout=30.0,
        max_tokens=max_tokens,
        attempts=int(attempts),
        budget_minutes=float(budget),
    )

    return StageSettings(
        profile=profile,
        servers=servers,
        concurrency_per_server=max(1, concurrency),
        model_name=model,
        max_tokens=max_tokens,
        attempts=max(1, int(attempts)),
        budget_minutes=float(budget) if budget else None,
    )

"""Shared LLM calling helpers, sampling parameters, thinking-stripping, and growth-aware retries.

Implements Audit v3 Section 4:
- final_answer: strictly strips any <think>...</think> reasoning blocks.
  Fails loud if <think is opened but never closed.
- OutputTruncated: detects finish_reason == 'length' and doubles max_tokens (up to max_growth times)
  clamped to max_model_len before failing loud.
- get_llm_kwargs: task profiling ('generate' vs 'judge').
"""

from typing import Any, Dict, List, Optional
from utils.errors import PermanentFailure


class OutputTruncated(PermanentFailure):
    """Raised when an LLM output was truncated by max_tokens limit even after permitted growth."""
    pass


def final_answer(content: Optional[str]) -> str:
    """Return only text after the model's reasoning. Never returns thinking.

    Raises ValueError if an opened <think> block is never terminated.
    """
    text = content or ""
    if "</think>" in text:
        text = text.rsplit("</think>", 1)[1]
    if "</thought>" in text:
        text = text.rsplit("</thought>", 1)[1]

    if "<think" in text or "<thought" in text:
        raise ValueError("Unterminated <think>/<thought> block in model output")

    return text.strip()


def _resolve_thinking(args: Any, cfg: Optional[Dict[str, Any]] = None) -> bool:
    """Resolve thinking mode precedence: CLI flag > config.json > default False."""
    if args is not None:
        if getattr(args, "disable_thinking", False):
            return False
        if getattr(args, "enable_thinking", False):
            return True
    if cfg is not None and "enable_thinking" in cfg:
        return bool(cfg.get("enable_thinking", False))
    return False


def get_llm_kwargs(args: Any, cfg: Optional[Dict[str, Any]] = None, task: str = "generate") -> Dict[str, Any]:
    """Return sampling arguments tailored to task type and thinking mode."""
    thinking = _resolve_thinking(args, cfg)
    if thinking:
        return {
            "temperature": 0.6,
            "top_p": 0.95,
            "extra_body": {"chat_template_kwargs": {"enable_thinking": True}},
        }

    base = {"chat_template_kwargs": {"enable_thinking": False}, "top_k": 20}
    if task == "judge":
        return {
            "temperature": 0.1,
            "top_p": 0.8,
            "presence_penalty": 0.0,
            "extra_body": {**base, "repetition_penalty": 1.0},
        }
    return {
        "temperature": 0.3,
        "top_p": 0.8,
        "presence_penalty": 0.5,
        "extra_body": {**base, "repetition_penalty": 1.0},
    }


async def call_llm(
    client: Any,
    model: str,
    messages: List[Dict[str, str]],
    max_tokens: int,
    *,
    args: Any = None,
    cfg: Optional[Dict[str, Any]] = None,
    task: str = "generate",
    max_growth: int = 0,
    max_model_len: Optional[int] = None,
    **override_kwargs
) -> str:
    """Call LLM with automatic thinking-stripping, finish_reason check, and token-growth retries.

    Args:
        max_growth: 1 for large extraction/refine/doc-merge stages; 0 for small output stages.
        max_model_len: Server context limit (e.g. 65536 or 98304) to clamp token doubling.
    """
    model_kwargs = get_llm_kwargs(args, cfg, task=task)
    model_kwargs.update(override_kwargs)

    limit = max_tokens
    for attempt in range(max_growth + 1):
        resp = await client.chat.completions.create(
            model=model,
            messages=messages,
            max_tokens=limit,
            **model_kwargs
        )
        ch = resp.choices[0]
        finish_reason = getattr(ch, "finish_reason", None)

        if finish_reason == "stop":
            return final_answer(ch.message.content)

        if finish_reason == "length":
            prompt_tokens = getattr(resp.usage, "prompt_tokens", 0) if hasattr(resp, "usage") and resp.usage else 0
            room = (max_model_len - prompt_tokens - 64) if max_model_len and prompt_tokens else limit * 2
            nxt = min(limit * 2, room)
            if attempt == max_growth or nxt <= limit:
                raise OutputTruncated(
                    f"Hit max_tokens={limit} (finish_reason='length', prompt={prompt_tokens}); exhausted {max_growth} retries."
                )
            limit = nxt
            continue

        raise ValueError(f"Unexpected finish_reason={finish_reason!r} from LLM server")

    raise OutputTruncated(f"Exhausted {max_growth} attempts on length truncation.")

"""vLLM Distributed Server Pool & Worker Dispatch.
--------------------------------------------------
Evenly partitions LLM requests across a cluster of OpenAI-compatible vLLM servers
with per-server concurrency limits, automatic failover, and request timeouts.
"""

import asyncio
import re
from dataclasses import dataclass, field
from typing import Any, Awaitable, Callable, Dict, List, Optional, Sequence

import httpx
from openai import AsyncOpenAI
from tqdm import tqdm

                     TransientFailureExhausted, describe_error)

@dataclass
class PoolResult:
    """Outcome of a pooled run."""

    results: List[Any] = field(default_factory=list)
    failures: List[Dict[str, Any]] = field(default_factory=list)
    per_server: Dict[str, int] = field(default_factory=dict)
    attempted: int = 0
    completed: int = 0

    @property
    def failed_count(self) -> int:
        """Number of items that did not complete.

        Returns:
            The length of :attr:`failures`. Provided so callers can test for a clean
            run without touching the failure records themselves."""
        return len(self.failures)

    def raise_if_failed(self, label: str, *, examples: int = 10) -> None:
        """Fail the stage loudly, naming the items that did not complete."""
        if not self.failures:
            return
        sample = [f.get("key", f.get("index")) for f in self.failures[:examples]]
        first = self.failures[0].get("error", "unknown error")
        raise RuntimeError(
            f"{label} left {len(self.failures)} item(s) incomplete after retries. "
            f"Examples: {sample}. First error: {first}"
        )


class ServerPool:
    """Round-robin vLLM client pool with per-server concurrency caps.

    Work is partitioned deterministically (item ``i`` goes to server ``i % N``)
    and each server has its own queue and concurrency cap, so a slow tail cannot
    pile onto whichever worker happens to be free.

    Args:
        servers: vLLM OpenAI-compatible base URLs.
        concurrency_per_server: In-flight requests allowed per server.
        request_timeout: Per-request read timeout in seconds.
        connect_timeout: Optional separate connection timeout in seconds.
        max_connections_slack: Extra httpx connections above the concurrency cap,
            so a retry can start before the previous socket is released.
        label: Stage name, used in log messages and retry labels.
        failover: Whether to move work off a server that stops responding.
        unhealthy_threshold: Consecutive failures before a server is marked
            unhealthy.
        llm_log: Optional :class:`~utils.llm_log.LLMCallLog`. When given,
            every request issued through this pool is appended to it, and the
            log is closed when the pool closes.
    """

    def __init__(
        self,
        servers: Sequence[str],
        *,
        concurrency_per_server: int,
        request_timeout: float,
        connect_timeout: Optional[float] = None,
        max_connections_slack: int = 10,
        label: str = "stage",
        failover: bool = True,
        unhealthy_threshold: int = 3,
        llm_log: Optional[Any] = None,
    ):
        """Configure a pool; no connection is opened yet.

        Args:
            servers: vLLM OpenAI-compatible base URLs. Blank entries are dropped.
            concurrency_per_server: In-flight requests allowed per server.
            request_timeout: Per-request read timeout in seconds.
            connect_timeout: Optional separate connection timeout in seconds.
            max_connections_slack: Extra httpx connections above the concurrency cap.
            label: Stage name used in log messages and retry labels.
            failover: Whether to move work off a server that stops responding.
            unhealthy_threshold: Consecutive transient failures before a server is
                retired for the rest of this map.
            llm_log: Optional call log; every request is appended to it when given.

        Raises:
            RuntimeError: If no usable server URL was supplied.
            ValueError: If ``concurrency_per_server`` is below 1."""
        parsed_servers: List[str] = []
        parsed_concurrencies: List[int] = []

        if isinstance(servers, dict):
            for s_url, s_conc in servers.items():
                if isinstance(s_url, str) and s_url.strip():
                    parsed_servers.append(s_url.strip())
                    parsed_concurrencies.append(max(1, int(s_conc)))
        else:
            default_conc = int(concurrency_per_server) if isinstance(concurrency_per_server, (int, float)) and concurrency_per_server > 0 else 32
            for s in servers:
                if not isinstance(s, str) or not s.strip():
                    continue
                s_str = s.strip()
                # Check for inline concurrency URL:CONCURRENCY (e.g. http://host:8000/v1:64)
                m = re.search(r':(\d+)$', s_str)
                if m and not s_str.endswith('/v1'):
                    conc = int(m.group(1))
                    url = s_str[:m.start()]
                elif isinstance(concurrency_per_server, dict):
                    url = s_str
                    conc = int(concurrency_per_server.get(url, default_conc))
                else:
                    url = s_str
                    conc = default_conc
                parsed_servers.append(url)
                parsed_concurrencies.append(max(1, conc))

        if not parsed_servers:
            raise RuntimeError(f"{label}: no usable LLM servers configured.")

        self.servers = parsed_servers
        self.server_concurrencies = parsed_concurrencies
        self.concurrency_per_server = max(parsed_concurrencies)  # For legacy property access
        self.request_timeout = float(request_timeout)
        self.connect_timeout = connect_timeout
        self.max_connections_slack = max_connections_slack
        self.label = label
        self.failover = failover
        self.unhealthy_threshold = max(1, unhealthy_threshold)
        self.llm_log = llm_log

        self._http: List[httpx.AsyncClient] = []
        self._clients: List[AsyncOpenAI] = []
        self._consecutive_failures: List[int] = []
        self._healthy: List[bool] = []
        self._served: List[int] = []
        self._requeue_cursor = 0
        self._lock = asyncio.Lock()

    # ── lifecycle ──────────────────────────────────────────────────────────
    def _build_clients(self) -> None:
        """Create one httpx client and one SDK client per server with per-server limits."""
        timeout = (httpx.Timeout(self.request_timeout, connect=self.connect_timeout)
                   if self.connect_timeout else httpx.Timeout(self.request_timeout))
        for url, conc in zip(self.servers, self.server_concurrencies):
            limits = httpx.Limits(max_connections=conc + self.max_connections_slack)
            http_client = httpx.AsyncClient(limits=limits, timeout=timeout)
            self._http.append(http_client)
            client = AsyncOpenAI(base_url=url, api_key="EMPTY",
                                 http_client=http_client, max_retries=0)
            self._clients.append(client)
        self._consecutive_failures = [0] * len(self.servers)
        self._healthy = [True] * len(self.servers)
        self._served = [0] * len(self.servers)

    async def __aenter__(self) -> "ServerPool":
        """Build the clients and return the pool for use in ``async with``.

        Returns:
            This pool, ready to :meth:`map`."""
        self._build_clients()
        return self

    async def __aexit__(self, *exc_info: Any) -> None:
        """Close every connection when leaving an ``async with`` block.

        Args:
            *exc_info: Ignored exception context.

        Returns:
            None. Never suppresses an in-flight exception."""
        await self.aclose()

    async def aclose(self) -> None:
        """Release all httpx connections and flush the call log.

        Safe to call more than once. A failure to close one client is swallowed so
        teardown cannot mask the error that ended the stage.

        Returns:
            None."""
        for client in self._http:
            try:
                await client.aclose()
            except Exception:  # pragma: no cover - teardown must not raise
                pass
        self._http.clear()
        self._clients.clear()
        if self.llm_log is not None:
            # Flushing here guarantees the trail survives a stage that dies
            # after the pool closes. Re-opening is cheap if a later pool in the
            # same process reuses the log.
            self.llm_log.close()

    @property
    def healthy_servers(self) -> List[str]:
        """Server URLs still considered usable.

        Returns:
            The subset of :attr:`servers` not retired by repeated transient failures."""
        return [s for s, ok in zip(self.servers, self._healthy) if ok]

    @property
    def per_server_counts(self) -> Dict[str, int]:
        """Requests actually issued to each server.

        Returns:
            ``{server_url: request_count}`` for every configured server, including
            those that served nothing. Counts attempts, not items, so a retried item
            is counted on each server that tried it."""
        return dict(zip(self.servers, self._served))

    def _mark_failure(self, index: int, classification: str, error: Optional[BaseException] = None) -> None:
        """Track connection-level failures and retire a server when it dies."""
        # Only true network connectivity drops should retire a server, not model output validation errors
        is_network_failure = False
        if error is not None:
            err_str = str(error).lower()
            if any(term in err_str for term in ["connect", "connection refused", "remoteprotocolerror", "unreachable"]):
                is_network_failure = True

        if not is_network_failure or not self.failover:
            return

        self._consecutive_failures[index] += 1
        if self._consecutive_failures[index] >= self.unhealthy_threshold and self._healthy[index]:
            print(f"\n[WARNING] Server {self.servers[index]} retired due to consecutive connection failures: {error}")
            self._healthy[index] = False

    def _next_healthy(self) -> Optional[int]:
        """Pick the next server to receive requeued work.

        Rotates through healthy servers so failover does not dump everything onto the
        first one in the list.

        Returns:
            The index of a healthy server, or ``None`` when every server has been
            retired -- which the caller turns into an ``AllServersUnhealthy`` failure."""
        healthy = [i for i, ok in enumerate(self._healthy) if ok]
        if not healthy:
            return None
        self._requeue_cursor = (self._requeue_cursor + 1) % len(healthy)
        return healthy[self._requeue_cursor]

    # ── dispatch ───────────────────────────────────────────────────────────
    async def map(
        self,
        items: Sequence[Any],
        handler: Callable[[AsyncOpenAI, Any, int], Awaitable[Any]],
        *,
        attempts: int = 4,
        desc: Optional[str] = None,
        key_fn: Optional[Callable[[Any], str]] = None,
        budget: Optional[StageBudget] = None,
        debug: Optional[Any] = None,
        attempt_timeout: Optional[float] = None,
        on_result: Optional[Callable[[Any, Any], Awaitable[None]]] = None,
        show_progress: bool = True,
    ) -> PoolResult:
        """Process ``items`` spread evenly across servers.

        ``handler(client, item, server_index)`` performs one logical unit of
        work and returns its result, or raises. Retries, classification,
        budget enforcement, progress streaming and failure collection are all
        handled here so stages never reimplement them.

        ``on_result(item, result)`` is awaited as each item completes, which is
        how stages checkpoint and stream partial output to disk.
        """
        result = PoolResult(per_server={s: 0 for s in self.servers})
        total = len(items)
        if total == 0:
            return result

        server_count = len(self.servers)
        # Shared dynamic queue: workers pull the next item as soon as their server
        # is free, preventing idle GPUs and balancing unequal workloads.
        shared_queue: asyncio.Queue = asyncio.Queue()
        for index, item in enumerate(items):
            shared_queue.put_nowait((index, item))

        outstanding = total
        pbar = tqdm(total=total, desc=desc or self.label, disable=not show_progress,
                    smoothing=0.1, mininterval=0.5)

        async def run_item(server_index: int, index: int, item: Any) -> None:
            """Drive one item to completion, with retries and accounting.

            Applies ``retry_async`` around :func:`call`, then records the outcome: on
            success the value is appended and ``on_result`` is awaited; on exhaustion the
            failure is recorded against the item's index so the caller can recover the item
            from its own list.

            Args:
                server_index: Index of the server owning this item's queue.
                index: Position of the item in the caller's sequence.
                item: The work unit itself.

            Returns:
                None. Mutates ``result`` and the progress bar; never raises for an ordinary
                item failure.

            Raises:
                AllServersUnhealthy: If every server was retired while this item was in
                flight, which aborts the whole map rather than losing the remaining work."""
            nonlocal outstanding
            key = key_fn(item) if key_fn is not None else str(index)
            client = self._clients[server_index]
            url = self.servers[server_index]

            attempts_made = {"n": 0}

            async def call() -> Any:
                """Issue one attempt at this item and publish its identity.

                Increments the server's request counter, then sets the ``CURRENT_CALL``
                contextvar so a wrapped client can attribute the request to this item, server
                and attempt number in the call log. A response the handler rejects is recorded
                separately, because the HTTP exchange itself may well have succeeded.

                Returns:
                    Whatever the handler returns.

                Raises:
                    Whatever the handler raises, after logging a rejection if an ``llm_log``
                    is attached. ``retry_async`` decides whether that is retryable."""
                self._served[server_index] += 1
                attempts_made["n"] += 1
                return await handler(client, item, server_index)

            try:
                value = await retry_async(
                    call,
                    attempts=attempts,
                    label=f"{self.label}[{key}]",
                    budget=budget,
                    debug=debug,
                    attempt_timeout=attempt_timeout,
                    server=url,
                    context={"key": key, "index": index},
                )
                async with self._lock:
                    self._consecutive_failures[server_index] = 0
                result.results.append(value)
                result.completed += 1
                result.per_server[url] = result.per_server.get(url, 0) + 1
                if on_result is not None:
                    await on_result(item, value)
            except (PermanentFailure, TransientFailureExhausted) as exc:
                classification = "permanent" if isinstance(exc, PermanentFailure) else "transient"
                cause = getattr(exc, "cause", None)
                self._mark_failure(server_index, classification, error=cause or exc)
                result.failures.append({
                    "key": key, "index": index, "server": url,
                    "classification": classification,
                    "error": describe_error(cause) if cause else str(exc),
                })
            except AllServersUnhealthy as exc:
                result.failures.append({"key": key, "index": index, "server": url,
                                        "classification": "permanent", "error": str(exc)})
                raise
            finally:
                outstanding -= 1
                pbar.update(1)

        async def worker(server_index: int) -> None:
            """Consume items dynamically as this server becomes ready."""
            while True:
                if budget is not None:
                    budget.check(self.label)
                if not self._healthy[server_index]:
                    return
                try:
                    entry = await asyncio.wait_for(shared_queue.get(), timeout=IDLE_POLL_SECONDS)
                except asyncio.TimeoutError:
                    if outstanding <= 0 and shared_queue.empty():
                        return
                    continue

                index, item = entry
                if not self._healthy[server_index]:
                    # Server retired; return item to shared_queue for other healthy servers
                    shared_queue.put_nowait((index, item))
                    shared_queue.task_done()
                    return

                await run_item(server_index, index, item)
                shared_queue.task_done()

        workers = [asyncio.create_task(worker(s))
                   for s in range(server_count)
                   for _ in range(self.server_concurrencies[s])]
        try:
            await asyncio.gather(*workers)
        finally:
            pbar.close()
            for task in workers:
                if not task.done():
                    task.cancel()

        if not any(self._healthy):
            raise AllServersUnhealthy(
                f"{self.label}: every configured vLLM server became unreachable "
                f"({', '.join(self.servers)}). Check the serving jobs, then rerun "
                f"with --resume."
            )
        return result


def pool_for_stage(args: Any, settings: Any, stage: str) -> ServerPool:
    """Build the ServerPool a stage sends its requests through.

    Every stage configures its pool the same way: the resolved servers,
    concurrency and timeouts, its call log, and whether to fail over. Building it
    here keeps those choices in one place.

    Args:
        args: Parsed CLI namespace. Supplies the call-log path and ``--no_failover``.
        settings: The stage's resolved settings from ``resolve_settings``.
        stage: Stage key, used as the pool's label and as its log name.

    Returns:
        An unopened ServerPool. Enter it with ``async with``.
    """
    return ServerPool(
        settings.servers,
        concurrency_per_server=settings.concurrency_per_server,
        request_timeout=settings.profile.request_timeout,
        connect_timeout=settings.profile.connect_timeout,
        max_connections_slack=20,
        label=stage,
        failover=not getattr(args, "no_failover", False),
    )

#!/usr/bin/env python3
"""
Downloads BRIGHT long_documents subsets from HuggingFace and writes them
into a local `data/` folder as .jsonl files.
"""

import json
from pathlib import Path
from datasets import load_dataset

DATASET_NAME = "xlangai/BRIGHT"
SUBSET = "long_documents"
BASE_DIR = Path(__file__).resolve().parent
OUTPUT_DIR = BASE_DIR / "data"

TARGET_DOMAINS = {
    "biology",
    "earth_science",
    "economics",
    "psychology",
    "sustainable_living",
    "robotics",
    "stackoverflow",
}

OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

print(f"Loading dataset: {DATASET_NAME} ({SUBSET})...")
dataset = load_dataset(DATASET_NAME, SUBSET)

file_handles = {}
for domain in TARGET_DOMAINS:
    file_path = OUTPUT_DIR / f"{domain}.jsonl"
    file_handles[domain] = open(file_path, "w", encoding="utf-8")

counts = {d: 0 for d in TARGET_DOMAINS}

try:
    for split_name, split_data in dataset.items():
        domain = split_name.strip().lower()
        if domain in TARGET_DOMAINS:
            print(f"Processing split: {domain} ({len(split_data)} records)...")
            for row in split_data:
                # Add domain field to row if not present
                if "domain" not in row:
                    row["domain"] = domain
                file_handles[domain].write(json.dumps(row, ensure_ascii=False) + "\n")
                counts[domain] += 1
        else:
            print(f"Skipping split: {split_name} (not in target domains)")
finally:
    for f in file_handles.values():
        f.close()

print("\nDownload complete! Files saved to:", OUTPUT_DIR)
print("Records per domain:")
for d, count in sorted(counts.items()):
    print(f"  {d:<22}: {count} records")

# ── XML Parsing Helper ────────────────────────────────────────────────────────

def parse_xml_concepts(text: str) -> List[Concept]:
    """Parse all completed <concept>...</concept> blocks directly from raw LLM output."""
    if not text:
        return []

    concepts = []
    blocks = re.findall(r"<concept>((?:(?!<concept>).)*?)</concept>", text, re.DOTALL | re.IGNORECASE)
    for block in blocks:
        title_m = re.search(r"<title>(.*?)</title>", block, re.DOTALL | re.IGNORECASE)
        if not title_m:
            continue
        title = title_m.group(1).strip()
        if not title:
            continue

        desc_m = re.search(r"<description>(.*?)</description>", block, re.DOTALL | re.IGNORECASE)
        description = desc_m.group(1).strip() if desc_m else ""

        # 3. Clean up in case description contains residual nested tags
        if "<concept>" in description or "<title>" in description:
            inner_desc_m = re.findall(r"<description>(.*?)(?:</description>|$)", description, re.DOTALL | re.IGNORECASE)
            if inner_desc_m:
                description = inner_desc_m[-1].strip()
            description = re.sub(r"<.*?>", "", description).strip()

        aliases = []
        aliases_m = re.search(r"<aliases>(.*?)</aliases>", block, re.DOTALL | re.IGNORECASE)
        if aliases_m:
            raw_aliases = aliases_m.group(1).strip()
            aliases = [a.strip() for a in re.split(r"[;,]", raw_aliases) if a.strip()]

        subtopics = []
        subtopics_m = re.search(r"<subtopics>(.*?)</subtopics>", block, re.DOTALL | re.IGNORECASE)
        if subtopics_m:
            raw_subs = subtopics_m.group(1).strip()
            subtopics = [s.strip() for s in re.split(r"[;,]", raw_subs) if s.strip()]

        concepts.append(Concept(
            title=title,
            aliases=aliases,
            subtopics=subtopics,
            description=description,
        ))

    return concepts

# ── Pipeline Implementation ──────────────────────────────────────────────────

class WikiExtractorPipeline:
    def __init__(self, args):
        self.args = args
        wiki_path = getattr(args, "wiki_path", None)
        if not wiki_path:
            raise ValueError("Missing required --wiki_path argument. Specify where to save wiki outputs.")

        self.base_wiki_dir = Path(wiki_path)
        self.base_wiki_dir.mkdir(parents=True, exist_ok=True)
        self.staging_dir = self.base_wiki_dir / "staging"
        self.cache_dir = self.base_wiki_dir / "cache"
        self.log_file = self.base_wiki_dir / "extraction_log.json"

        self.staging_dir.mkdir(parents=True, exist_ok=True)
        self.cache_dir.mkdir(parents=True, exist_ok=True)

        self.extraction_log = load_bool_log(
            self.log_file,
            corrupt_message="Log file corrupted. Initializing fresh extraction log.")
        self.cfg = _read_config()

    def cleanup_incomplete_states(self):
        """Remove partial concept files for incomplete chunks."""
        purged = 0
        for key, completed in list(self.extraction_log.items()):
            if not completed:
                doc_id, chunk_id = key.split("::", 1)
                partial_concepts = self.staging_dir / doc_id / chunk_id / "concepts.json"
                if partial_concepts.exists():
                    partial_concepts.unlink()
                    purged += 1
                del self.extraction_log[key]
        save_log_sync(self.log_file, self.extraction_log)
        if purged > 0:
            print(f"Cleared {purged} partial concept files.")

    def chunk_is_complete(self, chunk: Dict[str, Any]) -> bool:
        key = f"{chunk['doc_id']}::{chunk['chunk_id']}"
        if not self.extraction_log.get(key, False):
            return False

        concepts_file = self.staging_dir / chunk["doc_id"] / chunk["chunk_id"] / "concepts.json"
        try:
            with concepts_file.open("r", encoding="utf-8") as f:
                data = json.load(f)
            return isinstance(data.get("concepts"), list)
        except Exception:
            self.extraction_log[key] = False
            return False

    def prepare_chunks(self) -> List[Dict[str, Any]]:
        chunks_cache_file = self.cache_dir / "chunks_cache.json"
        cache_manifest_file = self.cache_dir / "chunks_cache_manifest.json"
        mapping_file = self.base_wiki_dir / "doc_mapping.json"

        corpus_path = resolve_corpus_path(self.args)
        chunking = bool(getattr(self.args, "chunking", True))
        chunk_size = int(getattr(self.args, "chunk_size", DEFAULT_CHUNK_SIZE))
        overlap = int(getattr(self.args, "chunk_overlap", DEFAULT_CHUNK_OVERLAP))

        expected_manifest = cache_manifest(
            corpus_path,
            tokenizer_name=self.settings.model_name if chunking else None,
            chunking=chunking,
            chunk_size=chunk_size,
            overlap=overlap)

        if chunks_cache_file.exists() and cache_manifest_file.exists() and mapping_file.exists():
            try:
                with cache_manifest_file.open("r", encoding="utf-8") as f:
                    cached_manifest = json.load(f)
                with chunks_cache_file.open("r", encoding="utf-8") as f:
                    cached_chunks = json.load(f)
                required_fields = {
                    "doc_id", "chunk_id", "current_chunk", "context_before", "context_after"
                }
                chunks_are_valid = isinstance(cached_chunks, list) and all(
                    isinstance(chunk, dict) and required_fields.issubset(chunk)
                    for chunk in cached_chunks
                )
                if cached_manifest == expected_manifest and chunks_are_valid:
                    print(f"Loading validated cached chunks from {chunks_cache_file}...")
                    return cached_chunks
            except (OSError, json.JSONDecodeError, TypeError):
                pass
            print("Cached chunks are stale or invalid; rebuilding them from current input.")

        print(f"Loading corpus from {corpus_path}...")
        docs = load_corpus(corpus_path)

        print("Deduplicating documents based on content hash...")
        canon_docs, doc_mapping = deduplicate(docs)
        atomic_write_json(mapping_file, doc_mapping)
        print(f"Deduplicated {len(docs)} docs down to {len(canon_docs)} unique canonical docs.")

        save_run_config(self.base_wiki_dir, {
            "corpus": str(corpus_path),
            "corpus_sha256": expected_manifest["source_sha256"],
            "wiki_path": str(self.base_wiki_dir),
            "model_name": self.settings.model_name,
            "chunking": chunking,
            "chunk_size": chunk_size if chunking else None,
            "chunk_overlap": overlap if chunking else None,
            "documents": len(docs),
            "unique_documents": len(canon_docs),
        })

        tokenizer = None
        if chunking:
            print(f"Loading instant offline tiktoken (cl100k_base) for chunking ({chunk_size} tokens, {overlap} overlap)...")
            import tiktoken
            tok_arg = getattr(self.args, "tokenizer", None)
            enc_name = "o200k_base" if tok_arg == "o200k_base" else "cl100k_base"
            enc = tiktoken.get_encoding(enc_name)

            class TiktokenWrapper:
                def encode(self, text: str) -> list:
                    return enc.encode(text, disallowed_special=())

                def decode(self, tokens: list, skip_special_tokens: bool = True) -> str:
                    return enc.decode(tokens)

            tokenizer = TiktokenWrapper()
        else:
            print("Chunking disabled: treating each document as a single unit.")

        all_chunks = build_units(canon_docs, tokenizer, chunking=chunking,
                                 chunk_size=chunk_size, overlap=overlap)

        atomic_write_json(chunks_cache_file, all_chunks)
        atomic_write_json(cache_manifest_file, expected_manifest)
        print(f"Cached {len(all_chunks)} unit(s) to disk.")
        return all_chunks

    async def extract_chunk(self, client, chunk_data: Dict[str, Any], server_index: int) -> ChunkExtractionSchema:
        user_prompt = f"""\
CONTEXT BEFORE:
{chunk_data['context_before']}

CURRENT CHUNK:
{chunk_data['current_chunk']}

CONTEXT AFTER:
{chunk_data['context_after']}

Extract the substantive concepts from CURRENT CHUNK now according to the system instructions.
"""
        max_model_len = getattr(self.args, "max_model_len", None) or (self.cfg.get("max_model_len") if hasattr(self, "cfg") else None)
        raw_text = await call_llm(
            client=client,
            model=self.settings.model_name,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": user_prompt}
            ],
            max_tokens=self.settings.max_tokens,
            args=self.args,
            cfg=getattr(self, "cfg", None),
            task="generate",
            max_growth=1,
            max_model_len=max_model_len,
        )

        completed_concepts = parse_xml_concepts(raw_text)

        # F-04: Disambiguate empty output from failure
        # If output has neither <concept> nor sentinel <no_concepts/>, and chunk has substance, raise recoverable error to retry
        if not completed_concepts and "<no_concepts" not in raw_text.lower():
            chunk_text = chunk_data.get("current_chunk", "")
            chunk_word_count = len(chunk_text.split())
            if chunk_word_count > 150:
                raise ValueError(
                    f"LLM produced no concepts and missing <no_concepts/> sentinel for non-empty chunk "
                    f"({chunk_word_count} words). Retrying."
                )

        return ChunkExtractionSchema(concepts=completed_concepts)

    async def persist_extraction(self, chunk_data: Dict[str, Any], extracted: ChunkExtractionSchema) -> None:
        doc_id = chunk_data["doc_id"]
        chunk_id = chunk_data["chunk_id"]
        key = f"{doc_id}::{chunk_id}"

        subfolder = self.staging_dir / doc_id / chunk_id
        await asyncio.to_thread(subfolder.mkdir, parents=True, exist_ok=True)

        concepts_file = subfolder / "concepts.json"
        json_payload = extracted.model_dump_json(indent=2)
        await asyncio.to_thread(concepts_file.write_text, json_payload, encoding="utf-8")

        self.extraction_log[key] = True
        self.flush_count += 1
        if self.flush_count >= 50:
            await save_log_async(self.log_file, self.extraction_log)
            self.flush_count = 0

    async def run(self):
        self.settings = resolve_settings(self.args, STAGE)
        self.budget = StageBudget(self.settings.budget_minutes)
        self.flush_count = 0

        if self.args.resume:
            self.cleanup_incomplete_states()

        all_chunks = self.prepare_chunks()
        pending_chunks = [c for c in all_chunks if not self.chunk_is_complete(c)]

        settings = self.settings
        pool = ServerPool(
            settings.servers,
            concurrency_per_server=settings.concurrency_per_server,
            request_timeout=settings.profile.request_timeout,
            connect_timeout=settings.profile.connect_timeout,
            label=STAGE,
            llm_log=None,
            failover=not getattr(self.args, "no_failover", False),
        )

        total_in_flight = sum(pool.server_concurrencies)
        print(f"Total chunks: {len(all_chunks)} | Pending chunks: {len(pending_chunks)}")
        print(f"Active Server Pool ({len(pool.servers)} servers, {total_in_flight} total concurrency):")
        for url, conc in zip(pool.servers, pool.server_concurrencies):
            print(f"  {url:<38} -> {conc} concurrent workers")

        if not pending_chunks:
            print("All chunks are already processed!")
            return

        for chunk in pending_chunks:
            key = f"{chunk['doc_id']}::{chunk['chunk_id']}"
            self.extraction_log.setdefault(key, False)
        async with pool:
            result = await pool.map(
                pending_chunks,
                self.extract_chunk,
                attempts=settings.attempts,
                desc=f"Extracting [{self.base_wiki_dir.name}]",
                key_fn=lambda c: f"{c['doc_id']}::{c['chunk_id']}",
                budget=self.budget,
                debug=None,
                attempt_timeout=settings.profile.attempt_timeout,
                on_result=self.persist_extraction,
            )

        await save_log_async(self.log_file, self.extraction_log)
        result.raise_if_failed(STAGE)

        print("Requests served per server:")
        for url, count in sorted(result.per_server.items(), key=lambda kv: -kv[1]):
            print(f"  {url:<40} {count}")

        failed_chunks = [c for c in all_chunks if not self.chunk_is_complete(c)]
        if failed_chunks:
            failed_keys = [f"{c['doc_id']}::{c['chunk_id']}" for c in failed_chunks]
            print(f"[ERROR] {len(failed_chunks)} chunk(s) incomplete after retries. Examples: {failed_keys[:10]}")
            raise RuntimeError(f"Pass 1 left {len(failed_chunks)} chunk(s) incomplete; rerun with --resume.")

        write_stage_manifest(
            wiki_path=self.base_wiki_dir,
            stage=STAGE,
            expected=len(all_chunks),
            completed=len(all_chunks) - len(failed_chunks),
            failed=len(failed_chunks),
        )
        print(f"\nSuccessfully finished extraction for: {self.base_wiki_dir.name}! All {len(all_chunks)} chunks are valid.")

# ── CLI Entrypoint ────────────────────────────────────────────────────────────

def main() -> None:
    parser = build_parser(
        stage=STAGE,
        description="High-throughput concept extractor: corpus -> per-chunk concepts",
        include_corpus=True,
        include_resume=True)
    args = parser.parse_args()
    pipeline = WikiExtractorPipeline(args)
    asyncio.run(pipeline.run())

if __name__ == "__main__":
    main()

# ── Pydantic Data Schemas for Verification ────────────────────────────────────

# The report is feedback, not a copy of the source. Keep it small enough to
# finish reliably even when the input chunk and concepts.json are both large.
MAX_MISSING_CONCEPTS = 8
MAX_MISSING_FACTS = 20
MAX_HALLUCINATIONS = 8


class VerificationResult(BaseModel):
    coverage_score: float = Field(
        ge=0.0, le=1.0,
        description="Score between 0.0 and 1.0 representing how thoroughly concepts and facts were retained."
    )
    is_sufficient: bool = Field(
        description="True if all primary concepts and essential metrics/facts were captured without critical omissions."
    )
    missing_concepts: List[str] = Field(
        max_length=MAX_MISSING_CONCEPTS,
        description="Primary thematic concepts present in the text chunk that were completely omitted from the extraction."
    )
    missing_facts: List[str] = Field(
        max_length=MAX_MISSING_FACTS,
        description="Important metrics, parameters, findings, tools, or dates present in text but absent from descriptions."
    )
    hallucinations: List[str] = Field(
        max_length=MAX_HALLUCINATIONS,
        description="Facts or statements in the extracted JSON that contradict or are absent from the source chunk."
    )
    critique: str = Field(
        min_length=1,
        description="1-2 sentences summarizing the completeness, factual accuracy, and granularity of the extraction."
    )
    synthetic: bool = Field(
        default=False,
        description="True if this verification result was synthesized as a fallback rather than evaluated by the LLM."
    )

    @model_validator(mode="after")
    def check_verdict(self) -> "VerificationResult":
        if self.is_sufficient and (self.coverage_score < 0.85 or self.missing_concepts):
            raise ValueError("is_sufficient cannot be true below 0.85 or with missing primary concepts")
        for field in ("missing_concepts", "missing_facts", "hallucinations"):
            values = getattr(self, field)
            normalized = [value.strip().casefold() for value in values]
            if any(not value for value in normalized) or len(set(normalized)) != len(values):
                raise ValueError(f"{field} must contain only nonempty, unique items")
        return self

# ── Verifier Pipeline ────────────────────────────────────────────────────────

class WikiVerifierPipeline:
    """Audits extracted concepts for one phase and reports aggregate quality."""

    def __init__(self, args):
        """Prepare one verification phase. Phase 1 checks Pass 1; phase 2 checks Pass 2.

        Args:
            args: Parsed CLI namespace. ``args.phase`` must be 1 or 2.

        Sets:
            The stage name, settings, paths and the result and log files for this phase.

        Raises:
            ValueError: If ``--phase`` is not 1 or 2."""
        self.args = args
        self.phase = int(args.phase)
        if self.phase not in (1, 2):
            raise ValueError("--phase must be 1 or 2")

        wiki_path = getattr(args, "wiki_path", None)
        if wiki_path:
            self.base_wiki_dir = Path(wiki_path)
            self.domain = self.base_wiki_dir.name
        else:
            self.domain = getattr(args, "domain", "default")
            wiki_base = getattr(args, "wiki_base_dir", None) or "wiki"
            self.base_wiki_dir = Path(wiki_base) / self.domain

        self.cfg = _read_config()
        self.stage = "verify-pass1" if self.phase == 1 else "verify-pass2"
        self.settings = resolve_settings(args, self.stage)

        self.staging_dir = self.base_wiki_dir / "staging"
        self.cache_dir = self.base_wiki_dir / "cache"

        # Phase-specific outputs: phase 2 must not overwrite phase 1
        self.result_filename = PASS1_RESULT if self.phase == 1 else PASS2_RESULT
        log_name = "verification_pass1_log.json" if self.phase == 1 else "verification_pass2_log.json"
        self.verification_log_file = self.base_wiki_dir / log_name

        upstream = "extract-concepts" if self.phase == 1 else "refine-concepts"
        verify_stage_manifest(self.base_wiki_dir, upstream)

        if not self.staging_dir.exists():
            raise FileNotFoundError(
                f"Staging directory missing at {self.staging_dir}. Run extraction first."
            )

        self.checkpoint = CheckpointLog(self.verification_log_file, flush_every=50)
        self.budget = StageBudget(self.settings.budget_minutes)
        self.verification_failures: Dict[str, str] = {}

        if getattr(args, "resume", False):
            cleared = self.checkpoint.reset_incomplete()
            if cleared:
                self.checkpoint.flush_sync()
                print(f"Cleared {cleared} incomplete verification marks for resume.")

    # ── item preparation ───────────────────────────────────────────────────
    def prepare_verification_items(self) -> List[Dict[str, Any]]:
        """Collect chunks whose extracted concepts are present and parseable, with deterministic sampling."""
        chunks_map = load_cached_chunks(self.cache_dir)

        print("Locating extracted concepts in staging directory...")
        verification_items: List[Dict[str, Any]] = []
        missing_extractions: List[str] = []

        for key, chunk_data in chunks_map.items():
            doc_id, chunk_id = key.split("::", 1)
            chunk_dir = self.staging_dir / doc_id / chunk_id
            concepts_file = chunk_dir / "concepts.json"

            if not concepts_file.exists():
                missing_extractions.append(key)
                continue

            try:
                with concepts_file.open("r", encoding="utf-8") as f:
                    concepts_data = json.load(f)
                if not isinstance(concepts_data.get("concepts"), list):
                    raise ValueError("concepts.json does not contain a concepts list")
            except Exception as exc:
                missing_extractions.append(f"{key} ({type(exc).__name__}: {exc})")
                continue

            # A result file that is missing or torn invalidates the mark, so a
            # resumed run re-audits it instead of trusting a hollow checkpoint.
            result_file = chunk_dir / self.result_filename
            if self.checkpoint.is_done(key):
                try:
                    with result_file.open("r", encoding="utf-8") as f:
                        prior = json.load(f)
                    VerificationResult.model_validate(prior)
                except Exception:
                    self.checkpoint.mark(key, False)

            verification_items.append({
                "doc_id": doc_id,
                "chunk_id": chunk_id,
                "key": key,
                "current_chunk": chunk_data["current_chunk"],
                "concepts_file": concepts_file,
                "result_file": result_file,
            })

        if missing_extractions:
            stage_hint = "Pass 1" if self.phase == 1 else "Pass 2"
            raise RuntimeError(
                f"{len(missing_extractions)} cached chunk(s) do not have valid extracted concepts. "
                f"Examples: {missing_extractions[:10]}. Complete {stage_hint} before verification."
            )

        # Deterministic sorting so sampling is identical between Phase 1 and Phase 2
        verification_items.sort(key=lambda x: x["key"])

        sample_size = getattr(self.args, "sample_size", 50)
        seed = getattr(self.args, "seed", 42)
        total_found = len(verification_items)
        if sample_size and sample_size > 0 and total_found > sample_size:
            import random
            rng = random.Random(seed)
            sampled = rng.sample(verification_items, sample_size)
            sampled_keys = {item["key"] for item in sampled}
            verification_items = [item for item in verification_items if item["key"] in sampled_keys]
            print(f"Sampled {len(verification_items)} random chunks for audit (seed={seed}, total available: {total_found}).")
        else:
            print(f"Found {len(verification_items)} extracted chunks ready for verification.")

        return verification_items

    def format_concepts_for_prompt(self, concepts_json_raw: str) -> str:
        """Format concepts into clean, readable text."""
        try:
            data = json.loads(concepts_json_raw)
            concepts = data.get("concepts", [])
        except Exception:
            return concepts_json_raw

        if not concepts:
            return "No concepts were extracted from this chunk (empty extraction)."

        lines = []
        for i, c in enumerate(concepts, 1):
            title = c.get("title", "Untitled")
            aliases = ", ".join(c.get("aliases", [])) if isinstance(c.get("aliases"), list) else ""
            desc = c.get("description", "").strip()

            block = [f"CONCEPT {i}: {title}"]
            if aliases:
                block.append(f"  Aliases: {aliases}")
            if desc:
                block.append(f"  Description: {desc}")
            lines.append("\n".join(block))

        return "\n\n".join(lines)

    # ── single LLM interaction ─────────────────────────────────────────────
    def build_prompt(self, item: Dict[str, Any], concepts_json_raw: str) -> str:
        """Build the auditor prompt for one unit."""
        concepts_formatted = self.format_concepts_for_prompt(concepts_json_raw)

        return f"""\
RAW SOURCE CHUNK:
{item['current_chunk']}

CURRENT EXTRACTED CONCEPTS TO AUDIT:
{concepts_formatted}

AUDITOR INSTRUCTION:
Compare the RAW SOURCE CHUNK against the EXTRACTED CONCEPTS above.
Evaluate whether the extraction accurately captured the core concepts and substantive facts from the text.
Identify any genuinely missing primary concepts (max {MAX_MISSING_CONCEPTS}), missing facts (max {MAX_MISSING_FACTS}), or hallucinations (max {MAX_HALLUCINATIONS}).
Output ONLY the <verification>...</verification> block.
"""

    def parse_verification(self, raw_text: Optional[str], key: str) -> VerificationResult:
        """Parse an audit response from XML (with JSON fallback) into VerificationResult."""
        if not raw_text:
            raise ValueError("Empty verification response")

        # 1. Check for XML <verification> block
        if "<verification>" in raw_text:
            block_m = re.search(r"<verification>((?:(?!<verification>).)*?)</verification>", raw_text, re.DOTALL | re.IGNORECASE)
            block = block_m.group(1) if block_m else raw_text

            score_m = re.search(r"<coverage_score>(.*?)</coverage_score>", block, re.DOTALL | re.IGNORECASE)
            coverage_score = None
            if score_m:
                val_str = re.search(r"([0-9]*\.?[0-9]+)", score_m.group(1).strip())
                if val_str:
                    try:
                        coverage_score = float(val_str.group(1))
                        if coverage_score > 1.0 and coverage_score <= 100.0:
                            coverage_score /= 100.0
                        coverage_score = max(0.0, min(1.0, coverage_score))
                    except ValueError:
                        coverage_score = None

            if coverage_score is None:
                raise ValueError("Missing or invalid coverage_score in verification output")

            def extract_list(tag: str) -> List[str]:
                m = re.search(rf"<{tag}>(.*?)</{tag}>", block, re.DOTALL | re.IGNORECASE)
                if not m:
                    return []
                items = []
                for line in m.group(1).strip().splitlines():
                    line = line.strip().lstrip("-*•0123456789.) ").strip()
                    if line and line.lower() not in ("none", "n/a", "[]", "none."):
                        items.append(line)
                return items

            missing_concepts = extract_list("missing_concepts")[:MAX_MISSING_CONCEPTS]
            missing_facts = extract_list("missing_facts")[:MAX_MISSING_FACTS]
            hallucinations = extract_list("hallucinations")[:MAX_HALLUCINATIONS]

            suff_m = re.search(r"<is_sufficient>(.*?)</is_sufficient>", block, re.DOTALL | re.IGNORECASE)
            if suff_m:
                raw_suff = suff_m.group(1).strip().lower()
                is_sufficient = "true" in raw_suff or "yes" in raw_suff
            else:
                is_sufficient = (coverage_score >= 0.85 and len(missing_concepts) == 0)

            if missing_concepts or coverage_score < 0.85:
                is_sufficient = False

            crit_m = re.search(r"<critique>(.*?)</critique>", block, re.DOTALL | re.IGNORECASE)
            critique = crit_m.group(1).strip() if crit_m else "Verification audit complete."
            if not critique:
                critique = "Verification audit complete."

            def dedupe(arr: List[str]) -> List[str]:
                seen = set()
                res = []
                for a in arr:
                    norm = a.strip().casefold()
                    if norm and norm not in seen:
                        seen.add(norm)
                        res.append(a.strip())
                return res

            parsed_data = {
                "coverage_score": coverage_score,
                "is_sufficient": is_sufficient,
                "missing_concepts": dedupe(missing_concepts),
                "missing_facts": dedupe(missing_facts),
                "hallucinations": dedupe(hallucinations),
                "critique": critique,
            }
            try:
                return VerificationResult.model_validate(parsed_data)
            except Exception as e:
                raise ValueError(f"Could not validate verification output for {key}: {e}")

        raise ValueError(f"Could not parse verification XML for {key}")

    async def verify_chunk(self, client: Any, item: Dict[str, Any],
                           server_index: int) -> VerificationResult:
        """One audit call using call_llm (task='judge'). Retries are owned by the server pool."""
        concepts_json_raw = await asyncio.to_thread(
            item["concepts_file"].read_text, encoding="utf-8"
        )
        user_prompt = self.build_prompt(item, concepts_json_raw)

        max_model_len = getattr(self.settings, "max_model_len", 65536)
        content = await call_llm(
            client=client,
            model=self.settings.model_name,
            messages=[
                {"role": "system", "content": VERIFICATION_SYSTEM_PROMPT},
                {"role": "user", "content": user_prompt}
            ],
            max_tokens=self.settings.max_tokens or 4096,
            args=self.args,
            cfg=self.cfg,
            task="judge",
            max_growth=0,
            max_model_len=max_model_len,
        )
        return self.parse_verification(content, item["key"])

    # ── streaming persistence ──────────────────────────────────────────────
    async def persist_result(self, item: Dict[str, Any],
                             result: VerificationResult) -> None:
        """Write one chunk's audit result and advance the checkpoint."""
        payload = result.model_dump_json(indent=2)
        await asyncio.to_thread(atomic_write_text, item["result_file"], payload)
        self.checkpoint.mark(item["key"], True)
        if self.checkpoint.needs_flush():
            await self.checkpoint.flush()

    # ── reporting ──────────────────────────────────────────────────────────
    def generate_summary_report(self) -> None:
        """Print both runs and, for phase 2, the measured improvement on the evaluated sample."""
        items = self.prepare_verification_items()
        target_keys = {i["key"] for i in items}

        before = collect_metrics(self.staging_dir, label=PASS1_LABEL,
                                 result_filename=PASS1_RESULT,
                                 target_keys=target_keys)
        print("\n" + format_report(self.domain, before) + "\n")

        if self.phase == 1:
            summary_path = write_summary(self.base_wiki_dir, self.domain, before)
            print(f"Saved Pass 1 verification summary to: {summary_path}")
            print("Phase 2 (after Pass 2 refinement) will report the improvement.\n")
            return

        after = collect_metrics(self.staging_dir, label=PASS2_LABEL,
                                result_filename=PASS2_RESULT,
                                target_keys=target_keys)
        print(format_report(self.domain, after) + "\n")

        if before.total_chunks == 0 or after.total_chunks == 0:
            print("WARNING: cannot compute improvement - one of the two "
                  "verification runs has no results on disk.\n")
            summary_path = write_summary(self.base_wiki_dir, self.domain, before,
                                         after if after.total_chunks else None)
            print(f"Saved verification summary to: {summary_path}\n")
            return

        improvement = build_improvement(before, after)
        print(format_improvement(self.domain, improvement) + "\n")
        summary_path = write_summary(self.base_wiki_dir, self.domain, before, after)
        print(f"Saved both verification runs + improvement to: {summary_path}\n")

    # ── orchestration ──────────────────────────────────────────────────────
    async def run(self) -> None:
        """Audit every pending unit, then write the phase's summary report.

        Each unit's verdict goes to its own result file and is checkpointed. When every
        unit is done the report is printed. Phase 2 also prints the improvement over
        phase 1 and writes ``verification_summary.json``.

        Returns:
            None.

        Raises:
            RuntimeError: If units remain without a verdict after retries, or if cached
                chunks lack the extractions they need."""
        items = self.prepare_verification_items()
        pending = [i for i in items if not self.checkpoint.is_done(i["key"])]

        print(f"Verification phase {self.phase} for domain '{self.domain}'.")
        print(f"Total extractions found: {len(items)} | Pending verification: {len(pending)}")
        print(f"Spreading work across {len(self.settings.servers)} server(s) at "
              f"{self.settings.concurrency_per_server} concurrent request(s) each "
              f"(item i -> server i % N).")
        print(f"Per-item attempts: {self.settings.attempts} | "
              f"Stage budget: {self.settings.budget_minutes:g} min")

        if not pending:
            print("All available extractions are already verified for this phase!")
            self.generate_summary_report()
            return

        pool = pool_for_stage(self.args, self.settings, self.stage)

        try:
            async with pool:
                result = await pool.map(
                    pending,
                    self.verify_chunk,
                    attempts=self.settings.attempts,
                    desc=f"Verifying phase {self.phase} [{self.domain}]",
                    key_fn=lambda item: item["key"],
                    budget=self.budget,
                    debug=None,
                    attempt_timeout=self.settings.profile.attempt_timeout,
                    on_result=self.persist_result,
                )
        finally:
            await self.checkpoint.flush()

        print("Requests served per server:")
        for url, count in sorted(result.per_server.items(), key=lambda kv: -kv[1]):
            print(f"  {url:<40} {count}")

        for failure in result.failures:
            self.verification_failures[failure["key"]] = failure["error"]

        result.raise_if_failed(self.stage)
        failed_items = [i["key"] for i in items if not self.checkpoint.is_done(i["key"])]
        if failed_items:
            self.generate_summary_report()
            print(f"[ERROR] {len(failed_items)} chunk(s) remain unverified. "
                  f"Examples: {failed_items[:10]}")
            first = next(iter(self.verification_failures.values()), "unknown error")
            raise RuntimeError(
                f"Verification phase {self.phase} left {len(failed_items)} chunk(s) "
                f"incomplete; rerun the stage with --resume to retry them. "
                f"First error: {first}"
            )

        self.generate_summary_report()
        write_stage_manifest(self.base_wiki_dir, self.stage, expected=len(items), completed=len(items), failed=0)

# ── CLI Entrypoint ────────────────────────────────────────────────────────────

def main():
    """Parse the command line and run one verification phase.

    Returns:
        None."""
    parser = build_parser(
        stage="verify-pass1",
        description="Knowledge Graph extraction verifier & auditor (runs per phase)",
        include_max_tokens=True,
        include_resume=True,
    )
    parser.add_argument("--phase", type=int, default=1, choices=(1, 2),
                        help="1 = audit raw Pass 1 output (default); "

    main()

# ── Pydantic Data Schemas for Second Pass ─────────────────────────────────────

class ConceptUpdate(BaseModel):
    title: str = Field(description="Exact title of the EXISTING concept being updated.")
    additional_aliases: List[str] = Field(
        default_factory=list, 
        description="New acronyms or alternative canonical names to add."
    )
    additional_description: str = Field(
        default="",
        description="Dense factual sentences to append to existing description paragraph."
    )

class Concept(BaseModel):
    title: str = Field(description="Exact canonical title (1-6 words, singular).")
    aliases: List[str] = Field(default_factory=list, description="Explicit acronyms or alternative names.")
    description: str = Field(description="Single dense factual paragraph with all raw metrics, parameters, and details.")

class Pass2ExtractionSchema(BaseModel):
    new_concepts: List[Concept] = Field(
        default_factory=list, 
        description="Brand new substantive concepts missed entirely in Pass 1."
    )
    concept_updates: List[ConceptUpdate] = Field(
        default_factory=list, 
        description="Updates and factual patches for concepts extracted in Pass 1."
    )

# ── Second-Pass System Prompt ────────────────────────────────────────────────

SYSTEM_PROMPT_PASS2 = """\
You are an expert knowledge auditor performing a SECOND-PASS EXTRACTION & REFINEMENT on a Knowledge Wiki.

YOUR MISSION:
Perform an exhaustive comparison between the RAW CHUNK text and the EXISTING EXTRACTED CONCEPTS.
Identify any missing concepts or omitted factual details, and generate XML patch updates.

INPUT & NOISE CLEANING:
- Silently ignore HTML tags, scraping artifacts, navigation menus, headers/footers, cookie notices, and boilerplate.
- If the chunk contains no substantive informational content, output nothing.

CONTEXT BOUNDARIES:
- CURRENT CHUNK is your source for facts. Use CONTEXT BEFORE / CONTEXT AFTER ONLY to resolve pronouns (e.g. "this system", "they") or acronyms to standard canonical titles.
- Never extract facts that appear exclusively in CONTEXT BEFORE or CONTEXT AFTER.
- Never invent facts or add outside knowledge.

TITLES AND ALIASES:
- Title: 1-6 words, Title Case, the full standard name (expand acronyms), singular unless the standard name is plural. Use the canonical name any other document would use for the same subject, ensuring concepts merge cleanly across sources.
- Aliases: Semicolon-separated list of acronyms, alternate designations, or specific minor sub-topics/parameters absorbed into this concept. If none, leave empty.

WHAT IS A CONCEPT (UNIVERSAL DEFINITION):
A concept is a canonical, self-contained, semantically coherent unit of knowledge that captures a meaningful subject or body of information, along with its relevant facts, properties, and relationships. It is independent of document boundaries, can be consolidated and enriched across multiple sources, and is not merely a keyword or arbitrary text chunk.
- Scope: Concepts may represent any meaningful subject across domains (e.g., architectures, software libraries, algorithms, financial entities, scientific phenomena, operational procedures, policies, historical events, or core definitions).
- Granularity & Standalone Subtopics: Any distinct subtopic or specialized theme that contains substantial information (typically 300 to 600+ words/tokens with its own details, properties, or comparisons) SHOULD be extracted as an independent <new_concept>.
- Avoid Trivial Fragmentation: Do not create thin 1-2 sentence stub concepts for trivial sub-steps or isolated numbers; absorb those facts directly into the description of their primary subject (prefer <update_concept> over creating thin new concepts) and record specific names in <aliases>.
- Do not extract meta-labels ("Introduction", "Chapter 1", "Overview") or whole-chunk summaries.

COMPREHENSIVE INFORMATION RETENTION (ZERO FACT LOSS):
- Capture all substantive information present in the chunk inside the appropriate concept descriptions (either updating an existing concept via <update_concept> or adding a missed concept via <new_concept>). Leave no meaningful informational content unabsorbed.
- Descriptions must preserve exact concrete details rather than high-level generalizations: retain specific names, identifiers, terms, numbers, metrics, operational conditions, rules, relationships, chronology, and attributions.
- All substantive information must be expressed in complete, grammatically sound sentences. Descriptions must be dense, self-contained paragraphs that explicitly name the subject and stand entirely on their own.

OUTPUT FORMAT:
Output ONLY completed XML blocks with no introductory or concluding text.

Output brand-new concepts using <new_concept>...</new_concept> blocks:
<new_concept>
<title>Exact Canonical Title</title>
<aliases>Acronym or Alt Name 1; Alt Name 2</aliases>
<description>A single comprehensive, fact-dense paragraph detailing what the concept is along with all substantive facts, properties, and specific details from this chunk.</description>
</new_concept>

Output factual updates to existing Pass 1 concepts using <update_concept>...</update_concept> blocks:
<update_concept>
<title>Exact Canonical Title of Existing Concept</title>
<additional_aliases>New Alt Name 1; New Alt Name 2</additional_aliases>
<additional_description>Dense factual sentences containing newly extracted facts or details to append to the existing description.</additional_description>
</update_concept>

If the existing extraction is already complete and no updates or new concepts are needed, output strictly:
<no_changes/>
"""

# ── XML Parsing Helper for Refinement ────────────────────────────────────────

def parse_xml_refinement(text: str) -> Pass2ExtractionSchema:
    if not text:
        raise ValueError("Empty refinement response")

    # Explicit empty form contract (Audit v3 Section 3.3)
    if "<no_changes/>" in text or "<no_changes />" in text:
        return Pass2ExtractionSchema()

    new_concepts: List[Concept] = []
    # Match <new_concept> blocks
    new_blocks = re.findall(
        r"<new_concept>((?:(?!<new_concept>).)*?)</new_concept>",
        text, re.DOTALL | re.IGNORECASE
    )
    for block in new_blocks:
        title_m = re.search(r"<title>(.*?)</title>", block, re.DOTALL | re.IGNORECASE)
        if not title_m or not title_m.group(1).strip():
            continue
        title = title_m.group(1).strip()

        desc_m = re.search(r"<description>(.*?)</description>", block, re.DOTALL | re.IGNORECASE)
        description = desc_m.group(1).strip() if desc_m else ""

        aliases = []
        aliases_m = re.search(r"<aliases>(.*?)</aliases>", block, re.DOTALL | re.IGNORECASE)
        if aliases_m:
            raw_aliases = aliases_m.group(1).strip()
            aliases = [a.strip() for a in re.split(r"[;,]", raw_aliases) if a.strip()]

        new_concepts.append(Concept(title=title, aliases=aliases, description=description))

    concept_updates: List[ConceptUpdate] = []
    # Match <update_concept> blocks
    update_blocks = re.findall(
        r"<update_concept>((?:(?!<update_concept>).)*?)</update_concept>",
        text, re.DOTALL | re.IGNORECASE
    )
    for block in update_blocks:
        title_m = re.search(r"<title>(.*?)</title>", block, re.DOTALL | re.IGNORECASE)
        if not title_m or not title_m.group(1).strip():
            continue
        title = title_m.group(1).strip()

        desc_m = re.search(r"<(?:additional_description|description)>(.*?)</(?:additional_description|description)>", block, re.DOTALL | re.IGNORECASE)
        additional_desc = desc_m.group(1).strip() if desc_m else ""

        aliases = []
        aliases_m = re.search(r"<(?:additional_aliases|aliases)>(.*?)</(?:additional_aliases|aliases)>", block, re.DOTALL | re.IGNORECASE)
        if aliases_m:
            raw_aliases = aliases_m.group(1).strip()
            aliases = [a.strip() for a in re.split(r"[;,]", raw_aliases) if a.strip()]

        concept_updates.append(ConceptUpdate(
            title=title,
            additional_aliases=aliases,
            additional_description=additional_desc
        ))

    if not new_concepts and not concept_updates:
        raise ValueError("Refinement response contained neither valid XML updates/new concepts nor <no_changes/>")

    return Pass2ExtractionSchema(new_concepts=new_concepts, concept_updates=concept_updates)

# ── Pipeline Implementation ──────────────────────────────────────────────────

class WikiSecondPassPipeline:
    def __init__(self, args):
        """Prepare Pass 2 refinement. Pass 1 output must already exist.

        Args:
            args: Parsed CLI namespace.

        Raises:
            FileNotFoundError: If the staging folder from Pass 1 is missing."""
        self.args = args
        wiki_path = getattr(args, "wiki_path", None)
        if wiki_path:
            self.base_wiki_dir = Path(wiki_path)
            self.domain = self.base_wiki_dir.name
        else:
            self.domain = getattr(args, "domain", "default")
            wiki_base = getattr(args, "wiki_base_dir", None) or "wiki"
            self.base_wiki_dir = Path(wiki_base) / self.domain

        self.cfg = _read_config()
        self.staging_dir = self.base_wiki_dir / "staging"
        self.cache_dir = self.base_wiki_dir / "cache"
        self.pass2_log_file = self.base_wiki_dir / "pass2_log.json"

        verify_stage_manifest(self.base_wiki_dir, "extract-concepts")

        if not self.staging_dir.exists():
            raise FileNotFoundError(f"Staging directory missing at {self.staging_dir}. Run Pass 1 extraction first.")

        self.pass2_log = load_bool_log(
            self.pass2_log_file,
            corrupt_message="Pass 2 log file corrupted. Re-initializing.")

    def cleanup_incomplete_states(self):
        """Purges pass2_log entries marked False and restores backups upon resume."""
        print("Sanitizing incomplete Pass 2 extraction states...")
        purged = 0
        for key, completed in list(self.pass2_log.items()):
            if not completed:
                doc_id, chunk_id = key.split("::")
                chunk_dir = self.staging_dir / doc_id / chunk_id

                pass1_backup = chunk_dir / "concepts_pass1_backup.json"
                concepts_file = chunk_dir / "concepts.json"

                # If backup exists, restore it to fix any corrupted merges
                if pass1_backup.exists():
                    shutil.copy(pass1_backup, concepts_file)

                del self.pass2_log[key]
                purged += 1

        save_log_sync(self.pass2_log_file, self.pass2_log)
        print(f"Purged {purged} incomplete Pass 2 chunk entries.")

    def prepare_pass2_items(self) -> List[Dict[str, Any]]:
        """Prepares items containing raw chunks, Pass 1 concepts, and audit verification if present."""
        print("Loading cached chunks index...")
        chunks_map = load_cached_chunks(self.cache_dir)

        items = []
        missing_or_invalid = []
        for key, chunk_data in chunks_map.items():
            doc_id, chunk_id = key.split("::")
            chunk_dir = self.staging_dir / doc_id / chunk_id
            concepts_file = chunk_dir / "concepts.json"

            if not concepts_file.exists():
                missing_or_invalid.append(key)
                continue

            # Load Pass 1 concepts
            try:
                with open(concepts_file, "r", encoding="utf-8") as f:
                    pass1_concepts = json.load(f)
                if not isinstance(pass1_concepts.get("concepts"), list):
                    raise ValueError("concepts.json does not contain a concepts list")
            except Exception as exc:
                missing_or_invalid.append(f"{key} ({type(exc).__name__}: {exc})")
                continue

            items.append({
                "doc_id": doc_id,
                "chunk_id": chunk_id,
                "key": key,
                "current_chunk": chunk_data["current_chunk"],
                "context_before": chunk_data.get("context_before", ""),
                "context_after": chunk_data.get("context_after", ""),
                "pass1_concepts": pass1_concepts.get("concepts", []) if isinstance(pass1_concepts, dict) else (pass1_concepts if isinstance(pass1_concepts, list) else []),
                "chunk_dir": chunk_dir
            })

        if missing_or_invalid:
            raise RuntimeError(
                f"{len(missing_or_invalid)} cached chunk(s) are missing valid Pass 1 concepts. "
                f"Examples: {missing_or_invalid[:10]}. Complete/resume Pass 1 before Pass 2."
            )

        print(f"Loaded {len(items)} chunks ready for Second Pass refinement.")
        return items

    async def extract_pass2_refinement(self, client, item: Dict[str, Any],
                                       server_index: int) -> Pass2ExtractionSchema:
        """Queries the LLM for second-pass deltas and missing facts."""
        pass1_lines = []
        concepts_list = item["pass1_concepts"]
        if isinstance(concepts_list, dict):
            concepts_list = concepts_list.get("concepts", [])
        if isinstance(concepts_list, list):
            for i, c in enumerate(concepts_list, 1):
                if not isinstance(c, dict):
                    continue
                title = c.get("title", "")
                aliases = ", ".join(c.get("aliases", [])) if isinstance(c.get("aliases"), list) else ""
                desc = c.get("description", "").strip() if isinstance(c.get("description"), str) else ""
                block = [f"CONCEPT {i}: {title}"]
                if aliases:
                    block.append(f"  Aliases: {aliases}")
                if desc:
                    block.append(f"  Description: {desc}")
                pass1_lines.append("\n".join(block))

        pass1_text = "\n\n".join(pass1_lines) if pass1_lines else "None (empty extraction)"

        user_prompt = f"""\
CONTEXT BEFORE:
{item['context_before']}

CURRENT CHUNK:
{item['current_chunk']}

CONTEXT AFTER:
{item['context_after']}

EXISTING EXTRACTED CONCEPTS (PASS 1):
{pass1_text}

Compare the CURRENT CHUNK against EXISTING EXTRACTED CONCEPTS.
If any primary concepts were missed, output them as <new_concept>.
If any quantitative metrics, bounds, or key facts were missed from existing concepts, output <update_concept> with the exact factual sentences to append.
If the existing extraction is already factually complete, output strictly <no_changes/>.
"""
        max_model_len = getattr(self.settings, "max_model_len", 65536)
        raw_text = await call_llm(
            client=client,
            model=self.settings.model_name,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT_PASS2},
                {"role": "user", "content": user_prompt}
            ],
            max_tokens=self.settings.max_tokens or 16000,
            args=self.args,
            cfg=self.cfg,
            task="generate",
            max_growth=1,
            max_model_len=max_model_len,
        )
        return parse_xml_refinement(raw_text)

    async def persist_refinement(self, item: Dict[str, Any],
                                 delta: Pass2ExtractionSchema) -> None:
        """SUCCESS: back up Pass 1, then merge into concepts.json."""
        key = item["key"]
        chunk_dir = item["chunk_dir"]

        # 1. Backup Pass 1 concepts.json if backup does not exist
        pass1_backup = chunk_dir / "concepts_pass1_backup.json"
        concepts_file = chunk_dir / "concepts.json"
        if not pass1_backup.exists() and concepts_file.exists():
            await asyncio.to_thread(shutil.copy, concepts_file, pass1_backup)

        # 2. Apply updates & merged output to concepts.json
        merged_data = self.apply_merges(item["pass1_concepts"], delta)
        merged_payload = json.dumps(merged_data, indent=2)
        await asyncio.to_thread(concepts_file.write_text, merged_payload, encoding="utf-8")

        self.pass2_log[key] = True
        self.flush_count += 1
        if self.flush_count >= 50:
            await save_log_async(self.pass2_log_file, self.pass2_log.copy())
            self.flush_count = 0

    def apply_merges(
        self, pass1_data: Any, pass2_delta: Pass2ExtractionSchema
    ) -> Dict[str, Any]:
        """Merges Pass 2 updates cleanly into existing Pass 1 concepts JSON."""
        if isinstance(pass1_data, list):
            concepts_list = pass1_data
        elif isinstance(pass1_data, dict):
            concepts_list = pass1_data.get("concepts", [])
        else:
            concepts_list = []
        concept_map = {c["title"].strip().lower(): c for c in concepts_list if isinstance(c, dict) and "title" in c}

        # 1. Apply concept updates
        for update in pass2_delta.concept_updates:
            target_key = update.title.strip().lower()
            if target_key in concept_map:
                c = concept_map[target_key]

                # Append aliases (deduplicated)
                existing_aliases = set(c.get("aliases", []))
                for alias in update.additional_aliases:
                    if alias not in existing_aliases:
                        c.setdefault("aliases", []).append(alias)
                        existing_aliases.add(alias)

                # Append factual sentences to description
                if update.additional_description.strip():
                    existing_desc = c.get("description", "").strip()
                    c["description"] = f"{existing_desc} {update.additional_description.strip()}".strip()
            else:
                if update.additional_description.strip():
                    new_item = {
                        "title": update.title,
                        "aliases": update.additional_aliases,
                        "description": update.additional_description.strip(),
                    }
                    concepts_list.append(new_item)
                    concept_map[target_key] = new_item

        # 2. Add brand-new concepts
        for new_c in pass2_delta.new_concepts:
            new_key = new_c.title.strip().lower()
            if new_key not in concept_map:
                c_dict = {
                    "title": new_c.title,
                    "aliases": new_c.aliases,
                    "description": new_c.description,
                }
                concepts_list.append(c_dict)
                concept_map[new_key] = c_dict
            else:
                c = concept_map[new_key]
                existing_aliases = set(c.get("aliases", []))
                for a in new_c.aliases:
                    if a not in existing_aliases:
                        c.setdefault("aliases", []).append(a)
                        existing_aliases.add(a)
                if new_c.description.strip():
                    existing_desc = c.get("description", "").strip()
                    c["description"] = f"{existing_desc} {new_c.description.strip()}".strip()

        return {"concepts": concepts_list}

    async def run(self):
        """Refine every chunk that verification flagged, merging the LLM's additions into its Pass 1 concepts.

        Before a chunk's concepts are changed, its Pass 1 version is backed up, so the
        original is never lost. Each refinement is checkpointed as it completes.

        Returns:
            None. Updates each chunk's ``concepts.json`` and writes ``pass2_log.json``.

        Raises:
            RuntimeError: If any chunk is still incomplete after retries."""
        self.settings = resolve_settings(self.args, STAGE)
        self.budget = StageBudget(self.settings.budget_minutes)
        self.flush_count = 0

        # Trigger cleanup if the resume flag is passed
        if hasattr(self.args, 'resume') and self.args.resume:
            self.cleanup_incomplete_states()

        items = self.prepare_pass2_items()
        pending = [
            i for i in items 
            if not self.pass2_log.get(i["key"], False)
        ]

        settings = self.settings
        total_pending = len(pending)
        print(f"Total chunks: {len(items)} | Pending Pass 2 refinement: {total_pending}")
        print(f"Spreading work across {len(settings.servers)} server(s) at "
              f"{settings.concurrency_per_server} concurrent request(s) each "
              f"(item i -> server i % N).")
        print(f"Per-item attempts: {settings.attempts} | Stage budget: "
              f"{settings.budget_minutes:g} min")

        if not pending:
            print("All chunks are already refined in Pass 2!")
            return

        # Mark every pending chunk in-progress
        for item in pending:
            self.pass2_log.setdefault(item["key"], False)

        pool = pool_for_stage(self.args, settings, STAGE)
        try:
            async with pool:
                result = await pool.map(
                    pending,
                    self.extract_pass2_refinement,
                    attempts=settings.attempts,
                    desc=f"Pass 2 Refinement [{self.domain}]",
                    key_fn=lambda i: i["key"],
                    budget=self.budget,
                    debug=None,
                    attempt_timeout=settings.profile.attempt_timeout,
                    on_result=self.persist_refinement,
                )
        finally:
            await save_log_async(self.pass2_log_file, self.pass2_log)

        print("Requests served per server:")
        for url, count in sorted(result.per_server.items(), key=lambda kv: -kv[1]):
            print(f"  {url:<40} {count}")

        result.raise_if_failed(STAGE)

        failed_items = [i["key"] for i in items if not self.pass2_log.get(i["key"], False)]
        if failed_items:
            print(f"[ERROR] {len(failed_items)} chunk(s) remain incomplete after Pass 2 retries. Examples: {failed_items[:10]}")
            raise RuntimeError(
                f"Pass 2 left {len(failed_items)} chunk(s) incomplete; rerun with --resume after checking server health."
            )

        write_stage_manifest(self.base_wiki_dir, STAGE, expected=len(items), completed=len(items), failed=0)
        print(f"\nPass 2 refinement successfully finished for domain: {self.domain}! All {len(items)} chunks are complete.")

class MergeCluster(BaseModel):
    primary_title: str = Field(description="The canonical primary title for this concept.")
    merged_titles: List[str] = Field(
        default_factory=list,
        description="Exact titles of other concepts in the input list that represent the same entity and should be merged into primary_title."

class DocumentConsolidationSchema(BaseModel):

SYSTEM_PROMPT_CONSOLIDATE = """\
You are an expert technical knowledge graph consolidator.
You are given a list of candidate concepts extracted across multiple chunks of the SAME document.

YOUR MISSION:
Identify groups of concepts that refer to the exact same core concept/subject within this document and should be merged together.

3. PRIMARY TITLE SELECTION: Pick the most standard, canonical, explicit title as `primary_title`.
4. SCOPE: Return only clusters of two or more titles. Titles in `<merged_titles>` must be copied EXACTLY as written in the candidate list.

# ── XML Parsing Helper ────────────────────────────────────────────────────────

def parse_xml_merge_clusters(text: str) -> List[MergeCluster]:
    """Parse <merge_clusters> blocks into MergeCluster objects directly."""
    if not text:
        return []

    clusters: List[MergeCluster] = []
    # Match each <cluster> block
    cluster_blocks = re.findall(r"<cluster>(.*?)</cluster>", text, re.DOTALL | re.IGNORECASE)
    for block in cluster_blocks:
        title_m = re.search(r"<primary_title>(.*?)</primary_title>", block, re.DOTALL | re.IGNORECASE)
        if not title_m or not title_m.group(1).strip():
            continue
        primary_title = title_m.group(1).strip()

        merged_titles = []
        merged_m = re.search(r"<merged_titles>(.*?)</merged_titles>", block, re.DOTALL | re.IGNORECASE)
        if merged_m:
            lines = merged_m.group(1).strip().splitlines()
            for line in lines:
                cleaned = re.sub(r"^[-*•\d\.\s]+", "", line).strip()
                if cleaned and cleaned.lower() != primary_title.lower():
                    merged_titles.append(cleaned)

        clusters.append(MergeCluster(primary_title=primary_title, merged_titles=merged_titles))

    # JSON fallback in case model emitted JSON
    if not clusters and "{" in text:
        try:
            m = re.search(r"\{.*\}", text, re.DOTALL)
            if m:
                payload = json.loads(m.group(0))
                if "merge_clusters" in payload:
                    schema = DocumentConsolidationSchema.model_validate(payload)
                    return schema.merge_clusters
        except Exception:
            pass

    return clusters

# ── Word Count Thin Concept Statistics Helper ─────────────────────────────────

def print_concept_word_distribution(concepts: List[Dict[str, Any]], title_label: str = "CONCEPTS"):
    """Compute and print word count distribution statistics for concept descriptions."""
    if not concepts:
        print(f"\n[{title_label}] No concepts found.")
        return

    word_counts = []
    for c in concepts:
        desc = c.get("description", "")
        # Count words
        words = len(re.findall(r"\b\w+\b", desc))
        word_counts.append(words)

    bins = [
        ("0 - 10 words", lambda w: 0 <= w <= 10),
        ("11 - 20 words", lambda w: 11 <= w <= 20),
        ("21 - 30 words", lambda w: 21 <= w <= 30),
        ("31 - 50 words", lambda w: 31 <= w <= 50),
        ("51 - 100 words", lambda w: 51 <= w <= 100),
        ("101 - 200 words", lambda w: 101 <= w <= 200),
        ("201+ words", lambda w: w > 200),
    ]

    bin_counts = {label: 0 for label, _ in bins}
    for count in word_counts:
        for label, fn in bins:
            if fn(count):
                bin_counts[label] += 1
                break

    total = len(word_counts)
    avg_words = sum(word_counts) / total if total else 0
    thin_30 = sum(1 for w in word_counts if w <= 30)

    print("\n============================================================")
    print(f"WORD-COUNT DISTRIBUTION [{title_label}] (Total Concepts: {total})")
    print(f"Average Words per Concept: {avg_words:.1f} | Thin Concepts (<= 30 words): {thin_30} ({thin_30 / total * 100:.1f}%)")
    print("------------------------------------------------------------")
    print(f"{'Word Count Range':<20} {'Count':<10} {'Percentage':<12}")
    print("------------------------------------------------------------")
    for label, _ in bins:
        count = bin_counts[label]
        pct = (count / total * 100) if total else 0.0
        bar = "█" * int(pct / 4)
        print(f"{label:<20} {count:<10} {pct:>5.1f}%   {bar}")
    print("============================================================\n")

# ── Batching Utilities ────────────────────────────────────────────────────────

def make_merge_batches(concepts: List[Dict[str, Any]], overlap: int = 10) -> List[List[Dict[str, Any]]]:
    """Group concept title-stubs into token-bounded batches with overlap, sorted by lemma key."""
    # F-13: Sort by lemma key then title so near-duplicates land in the same or adjacent batches
    ordered = sorted(concepts, key=lambda c: (canonical_title_key(str(c.get("title") or "")), str(c.get("title") or "").casefold()))
    batches: List[List[Dict[str, Any]]] = []
    
    i = 0
    while i < len(ordered):
        batch: List[Dict[str, Any]] = []
        size = 2
        j = i
        while j < len(ordered):
            concept = ordered[j]
            item_size = len(json.dumps(concept, ensure_ascii=False, separators=(",", ":"))) + 1
            if batch and (len(batch) >= MAX_MERGE_BATCH_CONCEPTS or size + item_size > MAX_MERGE_BATCH_CHARS):
                break
            batch.append(concept)
            size += item_size
            j += 1
        
        batches.append(batch)
        if j >= len(ordered):
            break
        # Advance with overlap so boundary items are compared together
        step = max(1, len(batch) - overlap)
        i += step

    return batches

# ── Pipeline Implementation ──────────────────────────────────────────────────

class WikiDocumentConsolidatorPipeline:
    def __init__(self, args):
        self.args = args
        wiki_path = getattr(args, "wiki_path", None)
        if not wiki_path:
            raise ValueError("Missing required --wiki_path argument.")

        self.base_wiki_dir = Path(wiki_path)
        verify_stage_manifest(self.base_wiki_dir, "refine-concepts")
        self.staging_dir = self.base_wiki_dir / "staging"
        self.consolidated_dir = self.base_wiki_dir / "consolidated"
        self.log_file = self.base_wiki_dir / "doc_consolidation_log.json"

        if not self.staging_dir.exists():
            raise FileNotFoundError(f"Staging directory missing at {self.staging_dir}. Run extraction first.")

        self.consolidated_dir.mkdir(parents=True, exist_ok=True)
        self.log_data = load_bool_log(self.log_file, corrupt_message="Log corrupted. Initializing fresh.")
        self.settings = resolve_settings(args, STAGE)
        self.budget = StageBudget(self.settings.budget_minutes)
        self.flush_count = 0
        self.log_lock = asyncio.Lock()

    def prepare_document_items(self) -> List[Dict[str, Any]]:
        """Collect all canonical documents and their raw chunk concepts."""
        print(f"Scanning staging directory: {self.staging_dir}...")
        doc_dirs = sorted([d for d in self.staging_dir.iterdir() if d.is_dir() and not d.name.startswith(".")])
        doc_items = []

        for doc_dir in doc_dirs:
            doc_id = doc_dir.name
            chunk_dirs = sorted([c for c in doc_dir.iterdir() if c.is_dir() and c.name.startswith("chunk_")])
            raw_concepts = []

            for chunk_dir in chunk_dirs:
                concepts_file = chunk_dir / "concepts.json"
                chunk_id = chunk_dir.name

                if concepts_file.exists():
                    try:
                        with open(concepts_file, "r", encoding="utf-8") as f:
                            data = json.load(f)
                            for concept in data.get("concepts", []):
                                if isinstance(concept, dict) and concept.get("title"):
                                    c_copy = dict(concept)
                                    c_copy["source_chunk"] = f"{doc_id}::{chunk_id}"
                                    raw_concepts.append(c_copy)
                    except Exception:
                        continue

            if raw_concepts:
                doc_items.append({
                    "doc_id": doc_id,
                    "raw_concepts": raw_concepts,
                    "output_dir": self.consolidated_dir / doc_id,
                })

        return doc_items

    async def get_merge_clusters(self, client: Any, item: Dict[str, Any], server_index: int) -> DocumentConsolidationSchema:
        doc_id = item["doc_id"]
        raw_concepts = item["raw_concepts"]

        lightweight_concepts = []
        for c in raw_concepts:
            desc = c.get("description", "")
            words = desc.split()[:30]
            lightweight_concepts.append({
                "title": c.get("title"),
                "aliases": c.get("aliases", [])[:8],
                "description_hint": " ".join(words)[:400],
            })

        batches = make_merge_batches(lightweight_concepts)
        # F-13: Limit concurrency to 1 within a document so pool concurrency limits are respected
        semaphore = asyncio.Semaphore(1)
        batch_pbar = None
        if len(batches) > 1:
            batch_pbar = tqdm(total=len(batches), desc=f"  [doc {doc_id[:12]}] {len(raw_concepts)} concepts", leave=False)

        def format_batch_for_prompt(batch_concepts: List[Dict[str, Any]]) -> str:
            lines = []
            for i, c in enumerate(batch_concepts, 1):
                title = c.get("title", "")
                aliases = ", ".join(c.get("aliases", []))
                hint = c.get("description_hint", "").strip()
                item_lines = [f"{i}. Title: \"{title}\""]
                if aliases:
                    item_lines.append(f"   Aliases: {aliases}")
                if hint:
                    item_lines.append(f"   Context: {hint}")
                lines.append("\n".join(item_lines))
            return "\n\n".join(lines)

        async def merge_batch(batch: List[Dict[str, Any]]) -> List[MergeCluster]:
            async with semaphore:
                batch_text = format_batch_for_prompt(batch)
                user_prompt = (
                    f"DOCUMENT ID: {doc_id}\nTOTAL EXTRACTED CONCEPTS ACROSS CHUNKS: "
                    f"{len(lightweight_concepts)}\nCONCEPTS IN THIS BATCH: {len(batch)}\n\n"
                    "CANDIDATE CONCEPTS LIST (only merge titles present in this list):\n\n"
                    f"{batch_text}\n"
                )

                try:
                    raw_text = await call_llm(
                        client=client,
                        model=self.settings.model_name,
                        messages=[
                            {"role": "system", "content": SYSTEM_PROMPT_CONSOLIDATE},
                            {"role": "user", "content": user_prompt}
                        ],
                        max_tokens=self.settings.max_tokens or 4096,
                        args=self.args,
                        cfg=getattr(self, "cfg", None),
                        task="judge",
                        check_truncation=True,
                    )
                    clusters = parse_xml_merge_clusters(raw_text)
                    if batch_pbar is not None:
                        batch_pbar.update(1)
                    return clusters
                except Exception as exc:
                    too_long = ("hit max_tokens" in str(exc).lower() or
                                isinstance(exc, BadRequestError) and
                                ("maximum context length" in str(exc).lower() or "input_tokens" in str(exc).lower()))
                    if not too_long or len(batch) <= 1:
                        raise
                    if batch_pbar is not None:
                        batch_pbar.total += 1
                        batch_pbar.refresh()

            mid = len(batch) // 2
            left, right = await asyncio.gather(merge_batch(batch[:mid]), merge_batch(batch[mid:]))
            return left + right

        batch_results = await asyncio.gather(*(merge_batch(batch) for batch in batches))
        if batch_pbar is not None:
            batch_pbar.close()

        return DocumentConsolidationSchema(
            merge_clusters=[cluster for clusters in batch_results for cluster in clusters]
        )

    def execute_programmatic_merge(self, raw_concepts: List[Dict[str, Any]], delta: DocumentConsolidationSchema) -> List[Dict[str, Any]]:
        """Group identical concepts and fuse their aliases, descriptions, and chunks using UnionFind."""
        # F-13: Resolve conflicting or overlapping clusters via UnionFind instead of dict overwrites
        uf = UnionFind()
        primary_titles: Dict[str, str] = {}

        for cluster in delta.merge_clusters:
            primary = cluster.primary_title.strip()
            if not primary:
                continue
            norm_primary = primary.casefold()
            uf.add(norm_primary)
            primary_titles.setdefault(norm_primary, primary)

            for alt in cluster.merged_titles:
                alt_clean = alt.strip()
                if alt_clean:
                    norm_alt = alt_clean.casefold()
                    uf.add(norm_alt)
                    uf.union(norm_primary, norm_alt)

        # Map each raw concept to its cluster root
        grouped_concepts: Dict[str, List[Dict[str, Any]]] = {}
        for c in raw_concepts:
            raw_title = c.get("title", "").strip()
            if not raw_title:
                continue
            norm_title = raw_title.casefold()
            root = uf.find(norm_title)
            grouped_concepts.setdefault(root, []).append(c)

        consolidated_concepts = []
        for root, concept_group in grouped_concepts.items():
            # Pick canonical display title: LLM primary title > most frequent title
            candidate_primaries = [primary_titles[item] for item in uf.get_groups().get(root, []) if item in primary_titles]
            if candidate_primaries:
                chosen_title = candidate_primaries[0]
            else:
                chosen_title = concept_group[0].get("title", "").strip()

            aliases_list = []
            subtopics_list = []
            source_chunks_set = set()

            for c in concept_group:
                for alias in c.get("aliases", []):
                    if alias.strip().casefold() != chosen_title.casefold():
                        aliases_list.append([alias.strip()])
                for sub in c.get("subtopics", []):
                    if sub.strip().casefold() != chosen_title.casefold():
                        subtopics_list.append([sub.strip()])
                raw_title = c.get("title", "").strip()
                if raw_title.casefold() != chosen_title.casefold():
                    aliases_list.append([raw_title])

                src_chunk = c.get("source_chunk")
                if src_chunk:
                    source_chunks_set.add(src_chunk)

            fused_desc = fuse_descriptions([c.get("description", "") for c in concept_group])

            consolidated_concepts.append({
                "title": chosen_title,
                "aliases": union_aliases(aliases_list),
                "subtopics": union_aliases(subtopics_list),
                "description": fused_desc,
                "source_chunks": sorted(list(source_chunks_set)),
            })

        return consolidated_concepts

    async def persist_consolidation(self, item: Dict[str, Any], delta: DocumentConsolidationSchema) -> None:
        doc_id = item["doc_id"]
        output_dir: Path = item["output_dir"]
        await asyncio.to_thread(output_dir.mkdir, parents=True, exist_ok=True)

        consolidated = self.execute_programmatic_merge(item["raw_concepts"], delta)
        out_payload = json.dumps({"doc_id": doc_id, "concepts": consolidated}, indent=2)
        await asyncio.to_thread(atomic_write_text, output_dir / "concepts.json", out_payload)

        async with self.log_lock:
            self.log_data[doc_id] = True
            self.flush_count += 1
            if self.flush_count >= 10:
                await save_log_async(self.log_file, self.log_data)
                self.flush_count = 0

    async def pass_through(self, pending: List[Dict[str, Any]]) -> None:
        no_merges = DocumentConsolidationSchema()
        for item in pending:
            await self.persist_consolidation(item, no_merges)
        if pending:
            await save_log_async(self.log_file, self.log_data)
            self.flush_count = 0

    async def run(self):
        doc_items = self.prepare_document_items()
        all_raw_concepts = [c for item in doc_items for c in item["raw_concepts"]]
        total_concepts_before = len(all_raw_concepts)

        pending = [d for d in doc_items if not self.log_data.get(d["doc_id"], False)]

        print(f"Total documents: {len(doc_items)} | Pending consolidation: {len(pending)}")
        print(f"Total raw concepts before consolidation: {total_concepts_before}")

        # Display Initial Word Distribution before consolidation
        print_concept_word_distribution(all_raw_concepts, title_label="STAGE 5: BEFORE DOCUMENT CONSOLIDATION")

        settings = self.settings
        if not chunking_enabled(self.base_wiki_dir):
            print("Chunking disabled: passing documents through without LLM calls.")
            await self.pass_through(pending)
        elif pending:
            print(f"Distributing {len(pending)} document(s) across {len(settings.servers)} server(s).")
            print(f"Per-item attempts: {settings.attempts} | Stage budget: {settings.budget_minutes:g} min")

            for item in pending:
                self.log_data.setdefault(item["doc_id"], False)

            pool = pool_for_stage(self.args, settings, STAGE)
            async with pool:
                res = await pool.map(
                    pending,
                    self.get_merge_clusters,
                    attempts=settings.attempts,
                    desc="Consolidating Documents",
                    key_fn=lambda it: it["doc_id"],
                    budget=self.budget,
                    attempt_timeout=settings.profile.attempt_timeout,
                    on_result=self.persist_consolidation,
                )
                res.raise_if_failed(STAGE)
        else:
            print("All documents are already consolidated!")

        await save_log_async(self.log_file, self.log_data)

        # Collect Consolidated Concepts and Display Final Word Distribution
        all_consolidated_concepts = []
        for doc_dir in self.consolidated_dir.iterdir():
            if doc_dir.is_dir():
                concepts_file = doc_dir / "concepts.json"
                if concepts_file.exists():
                    try:
                        with open(concepts_file, "r", encoding="utf-8") as f:
                            data = json.load(f)
                            all_consolidated_concepts.extend(data.get("concepts", []))
                    except Exception:
                        pass

        total_concepts_after = len(all_consolidated_concepts)
        print_concept_word_distribution(all_consolidated_concepts, title_label="STAGE 5: AFTER DOCUMENT CONSOLIDATION")

        write_stage_manifest(
            wiki_path=self.base_wiki_dir,
            stage=STAGE,
            expected=len(doc_items),
            completed=len(doc_items),
            failed=0,
            concepts_before=total_concepts_before,
            concepts_after=total_concepts_after,
        )

        print("============================================================")
        print("DOCUMENT CONSOLIDATION COMPLETE")
        print(f"Documents Processed:         {len(doc_items)}")
        print(f"Concepts Before Merge:       {total_concepts_before}")
        print(f"Concepts After Merge:        {total_concepts_after} (fused {total_concepts_before - total_concepts_after} duplicates)")
        print("============================================================\n")

def main():
    parser = build_parser(stage=STAGE, description="Consolidate concepts across chunks within each document", include_resume=True)
    args = parser.parse_args()

    pipeline = WikiDocumentConsolidatorPipeline(args)
    asyncio.run(pipeline.run())

if __name__ == "__main__":
    main()

def get_context_hint(desc: str, max_words: int = 40) -> str:
    """Extracts first ~40 words from description as rich context for LLM comparison."""
    if not desc:
        return "No description available."
    words = desc.strip().split()
    snippet = " ".join(words[:max_words])
    if len(words) > max_words:
        snippet += "..."
    return snippet

# ── XML System Prompt for LLM Concept Merge Verification ──────────────────────

SYSTEM_PROMPT_MERGE_VERIFY = """\
You are an expert knowledge consolidator evaluating candidate concepts across documents.
You are given candidate pairs of concepts that share identical titles, plural/spelling variants, or title-to-alias alignments.
Each concept is presented with its Canonical Title, Aliases, and a ~40-word Context Snippet from its description.

Your task is to determine whether both concepts refer to the EXACT SAME subject, entity, item, process, condition, or topic.

CRITICAL RULES:
1. MERGE (is_same_entity: true): Only if both concepts denote the identical subject (e.g., direct synonymous naming or alternative formal titles for the same entity).
2. KEEP SEPARATE (is_same_entity: false): Do NOT merge if one is an acronym that is polysemous, or if one is a sub-component/subtype/variant of the other.
3. When in doubt, prefer FALSE. Keeping concepts separate preserves integrity.

OUTPUT FORMAT:
Output ONLY the final <verdicts>...</verdicts> XML block with no introductory or concluding text:
<verdicts>
<verdict>
<pair_index>1</pair_index>
<is_same_entity>true</is_same_entity>
<chosen_canonical_title>Exact Canonical Title</chosen_canonical_title>
<reason>Direct synonymous reference.</reason>
</verdict>
</verdicts>
"""

class MergeVerdict(BaseModel):
    pair_index: int = 1
    is_same_entity: bool = False
    chosen_canonical_title: str = ""
    reason: str = ""

def parse_xml_merge_verdicts(text: str) -> List[MergeVerdict]:
    """Parse <verdicts> blocks into MergeVerdict objects."""
    if not text:
        return []

    verdicts: List[MergeVerdict] = []
    blocks = re.findall(r"<verdict>(.*?)</verdict>", text, re.DOTALL | re.IGNORECASE)
    for block in blocks:
        idx_m = re.search(r"<pair_index>(\d+)</pair_index>", block, re.IGNORECASE)
        idx = int(idx_m.group(1)) if idx_m else len(verdicts) + 1

        same_m = re.search(r"<is_same_entity>(.*?)</is_same_entity>", block, re.IGNORECASE)
        is_same = False
        if same_m:
            is_same = same_m.group(1).strip().lower() in ("true", "1", "yes")

        canon_m = re.search(r"<chosen_canonical_title>(.*?)</chosen_canonical_title>", block, re.DOTALL | re.IGNORECASE)
        canon = canon_m.group(1).strip() if canon_m else ""

        reason_m = re.search(r"<reason>(.*?)</reason>", block, re.DOTALL | re.IGNORECASE)
        reason = reason_m.group(1).strip() if reason_m else ""

        verdicts.append(MergeVerdict(
            pair_index=idx,
            is_same_entity=is_same,
            chosen_canonical_title=canon,
            reason=reason
        ))

    # JSON fallback
    if not verdicts and "{" in text:
        try:
            m = re.search(r"\{.*\}", text, re.DOTALL)
            if m:
                payload = json.loads(m.group(0))
                for item in payload.get("pair_verdicts", []):
                    verdicts.append(MergeVerdict(
                        pair_index=item.get("pair_index", len(verdicts) + 1),
                        is_same_entity=bool(item.get("is_same_entity", False)),
                        chosen_canonical_title=item.get("chosen_canonical_title", ""),
                        reason=item.get("reason", "")
                    ))
        except Exception:
            pass

    return verdicts

def make_candidate_pairs_prompt(pairs: List[Tuple[Dict[str, Any], Dict[str, Any]]]) -> str:
    blocks = []
    for i, (c1, c2) in enumerate(pairs, 1):
        c1_context = get_context_hint(c1.get("description", ""), max_words=40)
        c2_context = get_context_hint(c2.get("description", ""), max_words=40)
        c1_aliases = ", ".join(c1.get("aliases", [])) or "None"
        c2_aliases = ", ".join(c2.get("aliases", [])) or "None"

        block = (
            f"PAIR #{i}:\n"
            f"  Concept 1: \"{c1.get('title')}\"\n"
            f"    Aliases: {c1_aliases}\n"
            f"    Context (~40 words): {c1_context}\n"
            f"  Concept 2: \"{c2.get('title')}\"\n"
            f"    Aliases: {c2_aliases}\n"
            f"    Context (~40 words): {c2_context}"
        )
        blocks.append(block)
    return "VERIFY THE FOLLOWING CONCEPT PAIRS (Decide if they denote the EXACT SAME entity):\n\n" + "\n\n".join(blocks)

async def verify_merge_batch(client: Any, batch: List[Tuple[Dict[str, Any], Dict[str, Any]]], settings: Any, args: Any) -> List[MergeVerdict]:
    user_prompt = make_candidate_pairs_prompt(batch)
    raw_text = await call_llm(
        client=client,
        model=settings.model_name,
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT_MERGE_VERIFY},
            {"role": "user", "content": user_prompt}
        ],
        max_tokens=settings.max_tokens or 2048,
        args=args,
        task="judge",
        check_truncation=True,
    )
    return parse_xml_merge_verdicts(raw_text)

# ── Main Pipeline Execution ───────────────────────────────────────────────────

def global_consolidate(domain_dir: Path, args: Any):
    consolidated_dir = domain_dir / "consolidated"
    concepts_dir = domain_dir / "concepts"
    log_file = domain_dir / "global_consolidation.json"

    if not consolidated_dir.exists():
        raise FileNotFoundError(f"No consolidated directory found at {consolidated_dir}. Run document consolidation first.")

    verify_stage_manifest(domain_dir, "consolidate-documents")

    # 1. Load All Consolidated Concepts from per-document folders
    all_raw_concepts = []
    doc_files = sorted(list(consolidated_dir.glob("*/concepts.json")))
    if not doc_files:
        raise RuntimeError(f"No consolidated concept files found under {consolidated_dir}")

    print(f"Loading concepts from {len(doc_files)} document folders...")
    for doc_file in doc_files:
        with open(doc_file, "r", encoding="utf-8") as f:
            data = json.load(f)
            for c in data.get("concepts", []):
                t = c.get("title", "").strip()
                if not t:
                    continue
                all_raw_concepts.append(c)

    raw_concepts_count = len(all_raw_concepts)
    print(f"Total concepts loaded: {raw_concepts_count}")

    # Display initial word-count distribution before global merge
    print_concept_word_distribution(all_raw_concepts, title_label="STAGE 6: BEFORE GLOBAL CONSOLIDATION")

    # 2. Gather Candidate Merge Pairs (Exact Title, Lemma/Plural Variants, and Title-to-Alias alignments)
    # Group by canonical lemma key to find candidate exact and lemma variants across documents
    lemma_groups: Dict[str, List[Dict[str, Any]]] = {}
    for c in all_raw_concepts:
        t = c.get("title", "").strip()
        if not t:
            continue
        key = canonical_title_key(t)
        lemma_groups.setdefault(key, []).append(c)

    candidate_pairs: List[Tuple[Dict[str, Any], Dict[str, Any]]] = []
    seen_pair_keys: Set[Tuple[str, str]] = set()

    # (A) Candidate pairs from exact/lemma match across different documents/instances
    for key, group in lemma_groups.items():
        if len(group) > 1:
            for i in range(len(group)):
                for j in range(i + 1, min(i + 5, len(group))):  # pair across distinct concepts in group
                    c1, c2 = group[i], group[j]
                    pk = tuple(sorted([c1.get("title", ""), c2.get("title", "")]))
                    if pk[0].lower() != pk[1].lower() and pk not in seen_pair_keys:
                        seen_pair_keys.add(pk)
                        candidate_pairs.append((c1, c2))

    # (B) Candidate pairs from Title-to-Alias alignments
    alias_map: Dict[str, List[Dict[str, Any]]] = {}
    for c in all_raw_concepts:
        for alias in c.get("aliases", []):
            clean_al = canonical_title_key(alias)
            if len(clean_al) > 2:
                alias_map.setdefault(clean_al, []).append(c)

    for c in all_raw_concepts:
        norm_title = canonical_title_key(c.get("title", ""))
        if norm_title in alias_map:
            for match in alias_map[norm_title]:
                if canonical_title_key(match.get("title", "")) != norm_title:
                    pk = tuple(sorted([c.get("title", ""), match.get("title", "")]))
                    if pk not in seen_pair_keys:
                        seen_pair_keys.add(pk)
                        candidate_pairs.append((c, match))

    print(f"Identified {len(candidate_pairs)} candidate merge pair(s) (exact/lemma/alias) with contextual descriptions.")

    # 3. LLM Verification of Candidate Pairs
    # F-09: Union-find is keyed strictly by exact normalized title (strip().casefold())
    # Lemma keys are used purely to generate candidate pairs; union happens ONLY on LLM approval
    uf = UnionFind()
    # N-10: Key canonical choices by member title rather than ephemeral union root
    title_to_chosen_canonical: Dict[str, str] = {}
    for c in all_raw_concepts:
        t = c.get("title", "").strip()
        if t:
            uf.add(t.casefold())

    approved_merges = 0
    if candidate_pairs and not getattr(args, "skip_alias_verification", False):
        settings = resolve_settings(args, STAGE)
        batch_size = 10
        batches = [candidate_pairs[i:i + batch_size] for i in range(0, len(candidate_pairs), batch_size)]

        async def run_verification():
            pool = pool_for_stage(args, settings, STAGE)

            async def handler(client, batch, server_index):
                return await verify_merge_batch(client, batch, settings, args)

            async def on_batch_result(batch, verdicts):
                nonlocal approved_merges
                if verdicts:
                    for verdict in verdicts:
                        idx = verdict.pair_index - 1
                        if 0 <= idx < len(batch):
                            c1, c2 = batch[idx]
                            if verdict.is_same_entity:
                                t1, t2 = c1["title"].strip(), c2["title"].strip()
                                uf.union(t1.casefold(), t2.casefold())
                                approved_merges += 1
                                # F-10 & N-10: Record LLM's chosen_canonical_title for both members
                                if verdict.chosen_canonical_title.strip():
                                    title_to_chosen_canonical[t1.casefold()] = verdict.chosen_canonical_title.strip()
                                    title_to_chosen_canonical[t2.casefold()] = verdict.chosen_canonical_title.strip()

            async with pool:
                res = await pool.map(
                    batches,
                    handler,
                    attempts=settings.attempts,
                    desc="LLM Verifying Candidate Merge Pairs",
                    budget=None,
                    attempt_timeout=settings.profile.attempt_timeout,
                    on_result=on_batch_result,
                )
                res.raise_if_failed(STAGE)

        asyncio.run(run_verification())
        print(f"LLM confirmed and approved {approved_merges} concept merges.")
    else:
        print("Skipping LLM candidate pair verification (or no candidates found).")

    # 4. Final Attribute, Provenance & Description Fusion
    final_groups: Dict[str, List[Dict[str, Any]]] = {}
    for c in all_raw_concepts:
        title = c.get("title", "").strip()
        if not title:
            continue
        root = uf.find(title.casefold())
        final_groups.setdefault(root, []).append(c)

    global_concepts = []
    used_slugs: Set[str] = set()

    for root, concepts in tqdm(final_groups.items(), desc="Fusing global concept attributes"):
        # F-10 & N-10: Prefer LLM chosen canonical title from any group member > frequency > shortest
        chosen_title = None
        for c in concepts:
            c_norm = c.get("title", "").strip().casefold()
            if c_norm in title_to_chosen_canonical:
                chosen_title = title_to_chosen_canonical[c_norm]
                break

        if not chosen_title:
            title_counter = Counter(c.get("title", "").strip() for c in concepts)
            # Sort by frequency descending, then length ascending
            sorted_candidates = sorted(title_counter.items(), key=lambda kv: (-kv[1], len(kv[0])))
            chosen_title = sorted_candidates[0][0]

        # F-12: Assign collision-free stable slug once
        slug = generate_stable_slug(chosen_title, used_slugs)

        merged_aliases_set = set()
        merged_subtopics_set = set()
        global_source_chunks = set()

        for c in concepts:
            merged_aliases_set.add(c.get("title", "").strip())
            for a in c.get("aliases", []):
                if a.strip():
                    merged_aliases_set.add(a.strip())
            for s in c.get("subtopics", []):
                if s.strip():
                    merged_subtopics_set.add(s.strip())

            for sc in c.get("source_chunks", []):
                if sc.strip():
                    global_source_chunks.add(sc.strip())

        fused_desc = fuse_descriptions([c.get("description", "") for c in concepts])
        final_aliases = sorted([a for a in merged_aliases_set if a.casefold() != chosen_title.casefold()])
        final_subtopics = sorted([s for s in merged_subtopics_set if s.casefold() != chosen_title.casefold()])

        global_concepts.append({
            "title": chosen_title,
            "slug": slug,
            "aliases": final_aliases,
            "subtopics": final_subtopics,
            "description": fused_desc,
            "source_chunks": sorted(list(global_source_chunks)),
        })

    # Save summary and global concepts
    log_data = {
        "summary": {
            "canonical_docs_evaluated": len(doc_files),
            "raw_input_concepts": raw_concepts_count,
            "candidate_pairs_checked": len(candidate_pairs),
            "merges_approved_by_llm": approved_merges,
            "final_global_concepts": len(global_concepts),
        }
    }
    atomic_write_json(log_file, log_data)

    concepts_dir.mkdir(parents=True, exist_ok=True)
    print(f"Saving global_concepts.json to {domain_dir}...")
    atomic_write_json(domain_dir / "global_concepts.json", {"concepts": global_concepts})

    # F-11 & F-12: Write individual concept files using deterministic, collision-free slugs
    # Clean previous files to prevent stale _1/_2 duplicates on re-runs
    print("Writing concepts to concepts/ by unique slug...")
    for old_f in concepts_dir.glob("*.json"):
        try:
            old_f.unlink()
        except OSError:
            pass

    for gc in global_concepts:
        slug = gc.get("slug")
        if not slug:
            slug = generate_stable_slug(gc["title"], used_slugs)
            gc["slug"] = slug
        file_path = concepts_dir / f"{slug}.json"
        atomic_write_json(file_path, gc)

    assert len(set(c["slug"] for c in global_concepts)) == len(global_concepts), "Slug collision detected!"

    write_stage_manifest(
        wiki_path=domain_dir,
        stage=STAGE,
        expected=raw_concepts_count,
        completed=len(global_concepts),
        failed=0,
        approved_merges=approved_merges,
    )

    # Print final word count distribution across global concepts
    print_concept_word_distribution(global_concepts, title_label="STAGE 6: AFTER GLOBAL CONSOLIDATION")

    print("\n============================================================")
    print("GLOBAL CONSOLIDATION COMPLETE")
    print("============================================================")
    print(f"Total Canonical Docs Merged:  {len(doc_files)}")
    print(f"Raw Input Concepts Loaded:    {raw_concepts_count}")
    print(f"Candidate Pairs Audited:      {len(candidate_pairs)}")
    print(f"Approved LLM Merges:          {approved_merges}")
    print(f"Final Global Concepts:        {len(global_concepts)} (fused {raw_concepts_count - len(global_concepts)} duplicates)")
    print("============================================================\n")

def main():
    parser = build_parser(stage=STAGE, description="Consolidate concepts across documents using LLM-verified matches", include_resume=True)
    parser.add_argument("--skip_alias_verification", action="store_true",
                        help="Skip LLM verification of candidate pairs.")
    args = parser.parse_args()

    wiki_path = getattr(args, "wiki_path", None)
    if not wiki_path:
        raise ValueError("Missing required --wiki_path argument.")

    domain_dir = Path(wiki_path)
    global_consolidate(domain_dir, args)

if __name__ == "__main__":
    main()

"""Stage 7: Concept Brief & Key Entity Generation.
------------------------------------------------
Generates a 2-3 sentence overview brief and extracts important, domain-salient
entities for each consolidated concept in a single direct LLM call.

Usage:
    python3 write_briefs_and_entities.py --wiki_path /path/to/wiki
"""


import asyncio
import json
import re
from pathlib import Path
from typing import Any, Dict, List

from pydantic import BaseModel, Field

from utils.atomic_io import atomic_write_json, sanitize_filename
from utils.checkpoint import load_bool_log, save_log_sync

from utils.manifest import verify_stage_manifest, write_stage_manifest

STAGE = "write-briefs"

# ── XML System Prompt ─────────────────────────────────────────────────────────

SYSTEM_PROMPT_BRIEF_ENTITIES = """\
You are an expert knowledge agent.
Your task is to analyze a comprehensive concept description and produce two vital outputs in ONE pass:
1. A condensed BRIEF (1-3 sentences, ~25-60 words).
2. A list of 5 to 12 IMPORTANT ENTITIES mentioned in the text.

RULES FOR BRIEF:
- 1-3 complete sentences, about 25-60 words. If the description is under ~60 words, write ONE accurate sentence; never pad or infer.
- Keep exact numbers, metrics, and identifiers. Silently omit internal provenance IDs like [doc_...].
- No noise: do not add conversational filler.

RULES FOR ENTITIES (5-12):
- Specific named things in the text: proper names, standard terms of art, identifiers, quantities with units, named methods, or rules.
- Use the standard singular, unqualified surface form.
- Exclude words so vague they would fit almost any concept in the corpus.
- Each entity 1-4 words in Title Case.

OUTPUT FORMAT:
Output ONLY the <brief>...</brief> and <entities>...</entities> XML blocks:
<brief>
First dense sentence defining the core concept and its primary role. Second sentence explaining the primary mechanism, rule, or parameter.
</brief>
<entities>
- Primary Entity A
- Method or Rule B
- Tool or System C
- Component or Metric D
</entities>
"""

class BriefAndEntities(BaseModel):
    brief: str = Field(default="")
    entities: List[str] = Field(default_factory=list)

# ── XML Parsing Helper ────────────────────────────────────────────────────────

def parse_xml_brief_and_entities(text: str) -> BriefAndEntities:
    """Parse <brief> and <entities> XML blocks directly. Fails loud if tags are missing."""
    if not text:
        raise ValueError("Empty response received from LLM")

    brief_m = re.search(r"<brief>(.*?)</brief>", text, re.DOTALL | re.IGNORECASE)
    if not brief_m or not brief_m.group(1).strip():
        raise ValueError("Missing or empty <brief> block in LLM response")

    raw_brief = brief_m.group(1).strip()
    # Strip citation brackets like [doc_...::chunk_...]
    raw_brief = re.sub(r"\[(?:doc_|canon_)[^\]]+\]", "", raw_brief)
    brief = " ".join(raw_brief.split())

    entities = []
    entities_m = re.search(r"<entities>(.*?)</entities>", text, re.DOTALL | re.IGNORECASE)
    if not entities_m:
        raise ValueError("Missing <entities> block in LLM response")

    raw_entities = entities_m.group(1).strip().splitlines()
    for line in raw_entities:
        cleaned = re.sub(r"^[-*•\d\.\s]+", "", line).strip()
        if cleaned and len(cleaned) > 2 and len(cleaned) < 80:
            entities.append(cleaned)

    if not entities:
        raise ValueError("Zero valid entities found in <entities> block")

    return BriefAndEntities(brief=brief, entities=entities)

# ── Pipeline Implementation ──────────────────────────────────────────────────

class BriefAndEntitiesPipeline:
    def __init__(self, args):
        self.args = args
        wiki_path = getattr(args, "wiki_path", None)
        if not wiki_path:
            raise ValueError("Missing required --wiki_path argument.")

        self.domain_dir = Path(wiki_path)
        self.global_concepts_file = self.domain_dir / "global_concepts.json"
        self.concepts_dir = self.domain_dir / "concepts"
        self.log_file = self.domain_dir / "briefs_entities_log.json"

        if not self.global_concepts_file.exists():
            raise FileNotFoundError(
                f"Missing {self.global_concepts_file}. Complete Stage 6 (consolidate_global.py) first."
            )

        verify_stage_manifest(self.domain_dir, "consolidate-global")

        self.concepts_dir.mkdir(parents=True, exist_ok=True)
        if getattr(args, "resume", False):
            self.checkpoint = load_bool_log(self.log_file, corrupt_message="Log corrupted. Initializing fresh.")
            # Also populate checkpoint from disk if concepts/ already has enriched files
            for c_file in self.concepts_dir.glob("*.json"):
                try:
                    with open(c_file, "r", encoding="utf-8") as cf:
                        disk_data = json.load(cf)
                        t = disk_data.get("title")
                        if t and disk_data.get("brief") and "entities" in disk_data:
                            self.checkpoint[t] = True
                except Exception:
                    pass
        else:
            self.checkpoint = {}
        self.settings = resolve_settings(args, STAGE)
        self.budget = StageBudget(self.settings.budget_minutes)
        self.flush_count = 0
        self.log_lock = asyncio.Lock()

    async def generate_brief_and_entities(self, client: Any, item: Dict[str, Any], server_index: int) -> BriefAndEntities:
        title = item.get("title", "")
        aliases = ", ".join(item.get("aliases", []))
        description = item.get("description", "")

        user_prompt = f"CONCEPT TITLE: \"{title}\"\n"
        if aliases:
            user_prompt += f"ALIASES: {aliases}\n"
        user_prompt += f"\nFULL DESCRIPTION:\n{description}\n"

        max_model_len = getattr(self.args, "max_model_len", None) or (self.cfg.get("max_model_len") if hasattr(self, "cfg") else None)
        raw_text = await call_llm(
            client=client,
            model=self.settings.model_name,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT_BRIEF_ENTITIES},
                {"role": "user", "content": user_prompt}
            ],
            max_tokens=self.settings.max_tokens or 1024,
            args=self.args,
            cfg=getattr(self, "cfg", None),
            task="generate",
            max_growth=0,
            max_model_len=max_model_len,
        )
        return parse_xml_brief_and_entities(raw_text)

    async def persist_concept_brief(self, item: Dict[str, Any], res: BriefAndEntities):
        item["brief"] = res.brief
        item["entities"] = res.entities

        # Update individual concept JSON in concepts/ using stable slug (F-12)
        slug = item.get("slug")
        if not slug:
            slug = sanitize_filename(item["title"]) or "unnamed_concept"
        file_path = self.concepts_dir / f"{slug}.json"
        atomic_write_json(file_path, item)

        async with self.log_lock:
            self.checkpoint[item["title"]] = True
            self.flush_count += 1
            if self.flush_count >= 20:
                save_log_sync(self.log_file, self.checkpoint)
                self.flush_count = 0

    def assemble_and_save(self, concepts: List[Dict[str, Any]]):
        """Assemble enriched concepts from disk and update global_concepts.json atomically.
        Fails loud if any concept is missing its LLM-generated brief.
        """
        print(f"Assembling enriched concepts from {self.concepts_dir}...")
        assembled_concepts = []
        missing_briefs = []

        for c in concepts:
            slug = c.get("slug")
            if not slug:
                slug = sanitize_filename(c.get("title", "")) or "unnamed_concept"
            c_file = self.concepts_dir / f"{slug}.json"
            if c_file.exists():
                try:
                    with open(c_file, "r", encoding="utf-8") as cf:
                        disk_data = json.load(cf)
                        if disk_data.get("brief"):
                            c["brief"] = disk_data["brief"]
                        if disk_data.get("entities"):
                            c["entities"] = disk_data["entities"]
                except Exception:
                    pass

            if not c.get("brief"):
                missing_briefs.append(c.get("title", slug))
            assembled_concepts.append(c)

        if missing_briefs:
            raise RuntimeError(
                f"Stage 7 incomplete: {len(missing_briefs)} concept(s) missing LLM brief. "
                f"Examples: {missing_briefs[:10]}. Not writing global_concepts.json."
            )

        print(f"Updating {self.global_concepts_file} with briefs and entities...")
        atomic_write_json(self.global_concepts_file, {"concepts": assembled_concepts})

        write_stage_manifest(
            wiki_path=self.domain_dir,
            stage=STAGE,
            expected=len(concepts),
            completed=len(assembled_concepts),
            failed=0,
        )

    async def run(self):
        print(f"Loading concepts from {self.global_concepts_file}...")
        with open(self.global_concepts_file, "r", encoding="utf-8") as f:
            data = json.load(f)
            concepts = data.get("concepts", [])

        total = len(concepts)
        pending = [c for c in concepts if not self.checkpoint.get(c.get("title", ""), False)]

        print(f"Total concepts: {total} | Pending brief & entity extraction: {len(pending)}")
        settings = self.settings

        try:
            if pending:
                print(f"Distributing across {len(settings.servers)} server(s).")
                print(f"Per-item attempts: {settings.attempts} | Stage budget: {settings.budget_minutes:g} min")

                pool = pool_for_stage(self.args, settings, STAGE)
                async with pool:
                    res = await pool.map(
                        pending,
                        self.generate_brief_and_entities,
                        attempts=settings.attempts,
                        desc="Generating Briefs & Entities",
                        key_fn=lambda c: c.get("title", ""),
                        budget=self.budget,
                        attempt_timeout=settings.profile.attempt_timeout,
                        on_result=self.persist_concept_brief,
                    )
                    res.raise_if_failed(STAGE)
        finally:
            save_log_sync(self.log_file, self.checkpoint)
            self.assemble_and_save(concepts)

        print("\n============================================================")
        print("STAGE 7: BRIEF & IMPORTANT ENTITIES EXTRACTION COMPLETE")
        print(f"Total Concepts Processed: {total}")
        print(f"Output saved directly to {self.global_concepts_file}")
        print("============================================================\n")

#!/usr/bin/env python3
"""Stage 8: Semantic Jaccard Overlap Linker & Relation Characterization.
---------------------------------------------------------------------
Establishes bidirectional links between concepts sharing salient named entities:
1. Calculates Jaccard similarity across shared entities between concept pairs.
2. Characterizes direct relationships with concise LLM relation tags (<= 10 words).
3. Orders outbound relations by Jaccard similarity.

Usage:
    python3 link_entities_jaccard.py --wiki_path /path/to/wiki
"""


import asyncio
import json
import re
import math
from collections import defaultdict
from pathlib import Path
from typing import Any, Dict, List, Set, Tuple

import networkx as nx
from tqdm import tqdm

from utils.atomic_io import atomic_write_json, sanitize_filename
from utils.checkpoint import load_json_log, save_log_sync
from utils.cli import build_parser, resolve_settings
from utils.lemma import normalize_entity
from utils.llm_params import call_llm
from utils.llm_pool import pool_for_stage
from utils.manifest import verify_stage_manifest, write_stage_manifest
from utils.stats import print_link_distribution

STAGE = "link-entities"

SYSTEM_PROMPT_RELATION_SYNTHESIS = """\
You are an expert knowledge specialist.
You are given Concept 1 (the SUBJECT) and Concept 2 (the OBJECT), with their briefs and shared entities.
Write a phrase P such that "Concept 1 P Concept 2" is true, specific, and directional.

GUIDELINES:
- Output a concise phrase (3 to 10 words, no trailing punctuation).
- Prefer specific functional or hierarchical roles:
  * "is a subtype of"
  * "implements the protocol of"
  * "depends on components of"
  * "is evaluated against"
  * "serves as a competing alternative to"
  * "is an operational requirement for"
- Avoid vague tags like "related to", "connected with", "used with".
- If only co-occurrence is supported, output: "co-occurs with via <shared entity>".

OUTPUT FORMAT:
Output ONLY the <relation>...</relation> XML block:
<relation>
Concise relationship phrase
</relation>
"""

def parse_xml_relation(text: str) -> str:
    if not text:
        return ""
    rel_m = re.search(r"<relation>(.*?)</relation>", text, re.DOTALL | re.IGNORECASE)
    if rel_m:
        return rel_m.group(1).strip()
    return text.strip().strip('"').strip("'")

async def enrich_graph_via_jaccard(
    args: Any,
    wiki_path: Path,
    min_shared_entities: int = 1,
    min_jaccard: float = 0.05,
    max_hub_threshold: int = 150,
    specific_entity_threshold: int = 50,
    dry_run: bool = False
):
    global_concepts_file = wiki_path / "global_concepts.json"
    concepts_dir = wiki_path / "concepts"

    if not global_concepts_file.exists():
        raise FileNotFoundError(f"Missing {global_concepts_file}. Complete Stage 7 first.")

    verify_stage_manifest(wiki_path, "write-briefs")

    print(f"Loading concepts from {global_concepts_file}...")
    with open(global_concepts_file, "r", encoding="utf-8") as f:
        data = json.load(f)
        concepts = data.get("concepts", [])

    print(f"Loaded {len(concepts)} concepts. Indexing vital entities...")

    # Initial Link Distribution
    print_link_distribution(concepts, title_label="STAGE 8: BEFORE JACCARD LINKING")

    entity_to_concepts: Dict[str, Set[str]] = defaultdict(set)
    concept_entities: Dict[str, Set[str]] = {}
    concept_by_title: Dict[str, Dict[str, Any]] = {}
    title_lookup: Dict[str, str] = {}

    for c in concepts:
        title = c["title"]
        concept_by_title[title] = c
        norm_t = normalize_entity(title)
        if len(norm_t) > 2:
            title_lookup[norm_t] = title
        for alias in c.get("aliases", []):
            norm_a = normalize_entity(alias)
            if len(norm_a) > 2:
                title_lookup[norm_a] = title

        raw_entities = c.get("entities", [])
        norm_set = set()
        for ent in raw_entities:
            norm_e = normalize_entity(ent)
            if len(norm_e) > 2:
                norm_set.add(norm_e)
                entity_to_concepts[norm_e].add(title)
        concept_entities[title] = norm_set

    # Initial graph connectivity
    G = nx.Graph()
    for c in concepts:
        G.add_node(c["title"])
        for rel in c.get("outbound_relations", []):
            tgt = rel.get("target_concept", "").strip()
            if tgt and tgt in concept_by_title:
                G.add_edge(c["title"], tgt)

    initial_components = list(nx.connected_components(G))
    initial_components.sort(key=len, reverse=True)
    giant_size = len(initial_components[0]) if initial_components else 0
    print(f"Initial Graph Connectivity: {len(initial_components)} components. Giant component: {giant_size}/{len(concepts)} ({giant_size/len(concepts)*100:.1f}%)")

    # 1. Direct Conceptual Mention (Title/Alias in Entities)
    title_mention_links: List[Tuple[str, str, float, List[str]]] = []
    for title, ents in concept_entities.items():
        for ent in ents:
            if ent in title_lookup:
                matched_target = title_lookup[ent]
                if matched_target != title and not G.has_edge(title, matched_target):
                    title_mention_links.append((title, matched_target, 1.0, [ent]))

    print(f"Discovered {len(title_mention_links)} cross-reference links where a concept's title appears in key entities.")

    # 2. Shared Entities & IDF-Weighted Qualification (F-14)
    # Drop entities with df > max(50, 0.02 * N) to prevent broad hub dominance
    num_concepts = len(concepts)
    df_ceiling = max(50, int(0.02 * num_concepts))
    filtered_entity_to_concepts = {
        ent: titles for ent, titles in entity_to_concepts.items()
        if len(titles) <= df_ceiling
    }

    # Precompute IDF for remaining entities
    idf = {
        ent: math.log(max(1.0, num_concepts / max(1, len(titles))))
        for ent, titles in filtered_entity_to_concepts.items()
    }

    pair_shared: Dict[Tuple[str, str], Set[str]] = defaultdict(set)
    for ent, titles in filtered_entity_to_concepts.items():
        if len(titles) > 1:
            t_list = sorted(list(titles))
            for i in range(len(t_list)):
                for j in range(i + 1, len(t_list)):
                    pair_shared[(t_list[i], t_list[j])].add(ent)

    candidate_links: List[Tuple[str, str, float, List[str], str]] = []
    for (t1, t2), shared in pair_shared.items():
        if G.has_edge(t1, t2):
            continue
        set1 = concept_entities[t1]
        set2 = concept_entities[t2]
        union_ents = set1 | set2

        # IDF-weighted overlap score
        shared_idf = sum(idf.get(e, 0.0) for e in shared)
        union_idf = sum(idf.get(e, 0.0) for e in union_ents) or 1.0
        idf_score = shared_idf / union_idf

        qualifies = False
        # F-14: Qualify if >= 2 shared entities with sufficient IDF score, OR 1 rare entity
        if len(shared) >= 2 and idf_score >= 0.08:
            qualifies = True
        elif len(shared) == 1 and min_shared_entities <= 1:
            only_ent = next(iter(shared))
            if len(entity_to_concepts[only_ent]) <= min(20, specific_entity_threshold):
                qualifies = True

        if qualifies:
            candidate_links.append((t1, t2, idf_score, sorted(list(shared)), "entity_overlap"))

    print(f"Discovered {len(candidate_links)} reciprocal candidate pair(s) meeting IDF semantic criteria.")

    # Convert title_mention_links to 5-tuple
    formatted_title_mentions = [
        (t1, t2, score, shared, "title_mention")
        for t1, t2, score, shared in title_mention_links
    ]
    all_new_links = formatted_title_mentions + candidate_links

    # Establish reciprocal bidirectional links for all qualifying pairs
    added_links_by_source: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
    new_edges_added = 0

    for t1, t2, score, shared_list, l_type in all_new_links:
        if G.has_edge(t1, t2):
            continue
        shared_summary = ", ".join(shared_list[:3])
        if l_type == "title_mention":
            rel_text_1 = "named in this page's key entities"
            rel_text_2 = "named in this page's key entities"
        else:
            rel_text_1 = f"via: {shared_summary}"
            rel_text_2 = f"via: {shared_summary}"

        link_payload_1 = {
            "target_concept": t2,
            "relation": rel_text_1,
            "overlap_score": round(score, 3),
            "jaccard_similarity": round(score, 3),  # backward compatibility
            "shared_entities": shared_list,
            "link_type": l_type,
            "relation_source": "deterministic",
            "source_chunks": []
        }
        link_payload_2 = {
            "target_concept": t1,
            "relation": rel_text_2,
            "overlap_score": round(score, 3),
            "jaccard_similarity": round(score, 3),  # backward compatibility
            "shared_entities": shared_list,
            "link_type": l_type,
            "relation_source": "deterministic",
            "source_chunks": []
        }

        added_links_by_source[t1].append(link_payload_1)
        added_links_by_source[t2].append(link_payload_2)
        G.add_edge(t1, t2)
        new_edges_added += 1

    post_components = list(nx.connected_components(G))
    post_components.sort(key=len, reverse=True)
    post_giant = len(post_components[0]) if post_components else 0

    print(f"\nAdded {new_edges_added} bidirectional semantic edges (guaranteed 100% reciprocal).")
    print(f"Graph After Linking: {len(post_components)} components (down from {len(initial_components)}).")
    print(f"Giant component size: {post_giant}/{len(concepts)} ({post_giant/len(concepts)*100:.1f}%)")

    # Update concepts in memory & sort outbound relations by Jaccard descending
    for c in concepts:
        title = c["title"]
        existing_relations = c.get("outbound_relations", [])
        existing_targets = {r.get("target_concept", "").lower() for r in existing_relations}

        if title in added_links_by_source:
            for new_rel in added_links_by_source[title]:
                if new_rel["target_concept"].lower() not in existing_targets:
                    existing_relations.append(new_rel)
                    existing_targets.add(new_rel["target_concept"].lower())

        # Sort all outbound relations: highest Jaccard similarity first
        existing_relations.sort(
            key=lambda r: (r.get("jaccard_similarity") if r.get("jaccard_similarity") is not None else -1.0),
            reverse=True
        )
        c["outbound_relations"] = existing_relations

    # 3. LLM Relation Description Synthesis
    pairs_to_synthesize: Dict[str, Dict[str, Any]] = {}
    for c in concepts:
        t1 = c["title"]
        b1 = c.get("brief", "") or c.get("description", "")[:250]
        for rel in c.get("outbound_relations", []):
            t2 = rel.get("target_concept", "")
            if not t2 or t2.lower() == t1.lower() or t2 not in concept_by_title:
                continue
            r_text = rel.get("relation", "")
            if not r_text or "Functionally related via" in r_text or r_text.startswith("Functionally related"):
                pk = ":::".join(sorted([t1, t2]))
                if pk not in pairs_to_synthesize:
                    b2 = concept_by_title[t2].get("brief", "") or concept_by_title[t2].get("description", "")[:250]
                    pairs_to_synthesize[pk] = {
                        "concept_1": t1,
                        "brief_1": b1,
                        "concept_2": t2,
                        "brief_2": b2,
                        "shared_entities": rel.get("shared_entities", []),
                        "pair_key": pk
                    }

    if pairs_to_synthesize and not getattr(args, "skip_llm_descriptions", False):
        log_file = wiki_path / "link_relations_log.json"
        checkpoint: Dict[str, Any] = {}
        if getattr(args, "resume", False):
            checkpoint = load_json_log(log_file, corrupt_message="Link relations log corrupted. Initializing fresh.")

        pending_pairs = [p for pk, p in pairs_to_synthesize.items() if pk not in checkpoint or not checkpoint[pk]]
        print(f"\nIdentified {len(pairs_to_synthesize)} unique semantic relationship edge(s) across the graph.")
        print(f"Pending LLM relation synthesis: {len(pending_pairs)} (already completed: {len(pairs_to_synthesize) - len(pending_pairs)})")

        if pending_pairs:
            settings = resolve_settings(args, STAGE)
            print(f"Distributing across {len(settings.servers)} server(s) for LLM relationship synthesis...")
            pool = pool_for_stage(args, settings, STAGE)

            flush_counter = 0
            log_lock = asyncio.Lock()

            async def on_relation_result(item: Dict[str, Any], res: str):
                nonlocal flush_counter
                pk = item["pair_key"]
                async with log_lock:
                    checkpoint[pk] = res
                    flush_counter += 1
                    if flush_counter >= 20:
                        save_log_sync(log_file, checkpoint)
                        flush_counter = 0

            async def generate_relation(client: Any, item: Dict[str, Any], server_index: int) -> str:
                ents_str = ", ".join(item["shared_entities"][:8]) if item["shared_entities"] else "Domain context"
                prompt = (
                    f"CONCEPT 1 (SUBJECT): \"{item['concept_1']}\"\n"
                    f"Brief: {item['brief_1']}\n\n"
                    f"CONCEPT 2 (OBJECT): \"{item['concept_2']}\"\n"
                    f"Brief: {item['brief_2']}\n\n"
                    f"SHARED KEY ENTITIES: {ents_str}\n"
                )
                raw_text = await call_llm(
                    client=client,
                    model=settings.model_name,
                    messages=[
                        {"role": "system", "content": SYSTEM_PROMPT_RELATION_SYNTHESIS},
                        {"role": "user", "content": prompt}
                    ],
                    max_tokens=settings.max_tokens or 256,
                    args=args,
                    task="judge",
                    check_truncation=False,
                )
                parsed = parse_xml_relation(raw_text)
                return parsed if parsed else f"co-occurs with via {ents_str}"

            try:
                async with pool:
                    await pool.map(
                        pending_pairs,
                        generate_relation,
                        attempts=settings.attempts,
                        desc="Synthesizing LLM Relations",
                        key_fn=lambda p: p["pair_key"],
                        on_result=on_relation_result,
                    )
            finally:
                save_log_sync(log_file, checkpoint)

        # Apply rich synthesized relations back to graph
        applied_count = 0
        for c in concepts:
            t1 = c["title"]
            for rel in c.get("outbound_relations", []):
                t2 = rel.get("target_concept", "")
                pk = ":::".join(sorted([t1, t2]))
                if pk in checkpoint and checkpoint[pk]:
                    rel["relation"] = checkpoint[pk]
                    applied_count += 1
        print(f"Applied LLM relationship descriptions to {applied_count} reciprocal directed relations.")

    # Final Link Distribution
    print_link_distribution(concepts, title_label="STAGE 8: AFTER JACCARD LINKING")

    if dry_run:
        print("[DRY RUN] No files modified.")
        return

    print(f"Saving enriched graph to {global_concepts_file}...")
    atomic_write_json(global_concepts_file, {"concepts": concepts})

    if concepts_dir.exists():
        print(f"Updating individual concept JSONs in {concepts_dir}/...")
        for c in tqdm(concepts, desc="Writing concept files"):
            slug = c.get("slug")
            if not slug:
                slug = sanitize_filename(c["title"]) or "unnamed_concept"
            file_path = concepts_dir / f"{slug}.json"
            atomic_write_json(file_path, c, fsync_dir=False)

    write_stage_manifest(
        wiki_path=wiki_path,
        stage=STAGE,
        expected=len(concepts),
        completed=len(concepts),
        failed=0,
        edges_added=new_edges_added,
    )

    print("\n============================================================")
    print("STAGE 8: JACCARD ENTITY LINKING COMPLETE")
    print(f"Total Concepts:              {len(concepts)}")
    print(f"New Bidirectional Edges:     {new_edges_added}")
    print(f"Final Giant Component Size:  {post_giant} ({post_giant/len(concepts)*100:.1f}%)")
    print("============================================================\n")

def main():

"""Stage 9: Final Consolidation, Semantic Link Ranking & Clean Markdown Wiki Export.

Executes final quality consolidation, link ranking, and wiki generation:
1. Macro Alphabetical & Deep LLM Verification: Catches remaining near-duplicates or junk.
2. Semantic Link Ranking: Ranks outbound relations using Jaccard & Asymmetric Containment.
3. Reciprocal Rescue: Connects peripheral nodes with zero inbound links.
4. Clean Markdown Wiki Export: Exports pure markdown concept pages (no YAML frontmatter),
   single-line grep-friendly concept_index.md, and entity_index.md.

Usage:

# ── XML System Prompt for Alphabet Macro Scan (Domain-Agnostic) ──────────────

SYSTEM_PROMPT_MACRO_SCAN = """\
You are an expert knowledge consolidator auditing concepts starting with a specific letter or group.
You are provided with an indexed list of concept titles: "[ID] Title".

TASK:
Examine the list and identify:
1. "merge_pairs": Pairs of IDs that represent the EXACT SAME concept or near-duplicates:
   - Alternative spellings or transliterations.
   - Formatting / hyphenation / spacing splits.
   - High phrasing overlap or minor singular/plural/word-order variations.
   - Redundant sub-split duplicates.
2. "link_pairs": Pairs of IDs that are distinct topics or serialized parts, but MUST have direct reciprocal links:
   - Serialized volumes, versions, or numbered parts (e.g. Part 1 vs Part 2, Type I vs Type II).
   - Direct parent vs sub-category topics.
3. "prune_ids": Candidates that appear to be web scraping junk, cookie/privacy notices, navigation menus, or boilerplate. (Note: prune candidates are verified deeply before deletion; do not prune a concept merely because its title is unfamiliar).

OUTPUT FORMAT:
Output ONLY the <macro_candidates>...</macro_candidates> XML block:
<macro_candidates>
  <merge_pairs>
    <pair><id_a>1</id_a><id_b>5</id_b></pair>
  </merge_pairs>
  <link_pairs>
    <pair><id_a>2</id_a><id_b>8</id_b></pair>
  </link_pairs>
  <prune_ids>
    <id>15</id>
  </prune_ids>
</macro_candidates>
"""

class MacroScanResult(BaseModel):
    merge_pairs: List[Tuple[int, int]] = []
    link_pairs: List[Tuple[int, int]] = []
    prune_ids: List[int] = []


def parse_macro_scan(text: str) -> MacroScanResult:
    if not text:
        return MacroScanResult()

    res = MacroScanResult()

    mp_block = re.search(r"<merge_pairs>(.*?)</merge_pairs>", text, re.DOTALL | re.IGNORECASE)
    if mp_block:
        for pair_m in re.finditer(r"<pair>\s*<id_a>(\d+)</id_a>\s*<id_b>(\d+)</id_b>\s*</pair>", mp_block.group(1), re.IGNORECASE):
            res.merge_pairs.append((int(pair_m.group(1)), int(pair_m.group(2))))

    lp_block = re.search(r"<link_pairs>(.*?)</link_pairs>", text, re.DOTALL | re.IGNORECASE)
    if lp_block:
        for pair_m in re.finditer(r"<pair>\s*<id_a>(\d+)</id_a>\s*<id_b>(\d+)</id_b>\s*</pair>", lp_block.group(1), re.IGNORECASE):
            res.link_pairs.append((int(pair_m.group(1)), int(pair_m.group(2))))

    pr_block = re.search(r"<prune_ids>(.*?)</prune_ids>", text, re.DOTALL | re.IGNORECASE)
    if pr_block:
        for id_m in re.finditer(r"<id>(\d+)</id>", pr_block.group(1), re.IGNORECASE):
            res.prune_ids.append(int(id_m.group(1)))

    if not res.merge_pairs and not res.link_pairs and not res.prune_ids and "{" in text:
        try:
            m = re.search(r"\{.*\}", text, re.DOTALL)
            if m:
                payload = json.loads(m.group(0))
                for p in payload.get("merge_pairs", []):
                    if len(p) == 2:
                        res.merge_pairs.append((int(p[0]), int(p[1])))
                for p in payload.get("link_pairs", []):
                    if len(p) == 2:
                        res.link_pairs.append((int(p[0]), int(p[1])))
                for pid in payload.get("prune_ids", []):
                    res.prune_ids.append(int(pid))
        except Exception:
            pass

    return res

# ── XML System Prompt for Deep Verification (Domain-Agnostic) ────────────────

SYSTEM_PROMPT_DEEP_VERIFY = """\
You are an expert knowledge consolidator performing final quality verification.
You are given a candidate pair or cluster of concepts with their Canonical Titles, Briefs, and Top Entities.

Determine the definitive action for this group:
- MERGE: If both concepts represent the EXACT SAME single subject.
- LINK: If concepts are functionally related, parent/child, version, or part/whole pairs.
- PRUNE: If one is confirmed boilerplate, navigational junk, or scraping residue.
- KEEP_SEPARATE: If they are distinct concepts with no direct relationship.

OUTPUT FORMAT:
Output ONLY the final <verdict>...</verdict> XML block:
<verdict>
<action>MERGE | LINK | PRUNE | KEEP_SEPARATE</action>
<primary_title>Exact Canonical Title to keep for MERGE</primary_title>
<prune_title>Exact Canonical Title of the junk concept for PRUNE</prune_title>
<reason>One clear sentence explaining the verdict.</reason>
</verdict>
"""


class FinalActionVerdict(BaseModel):
    action: str = "KEEP_SEPARATE"
    primary_title: str = ""
    prune_title: str = ""
    reason: str = ""


def parse_xml_final_action(text: str) -> FinalActionVerdict:
    if not text:
        return FinalActionVerdict()

    act_m = re.search(r"<action>(.*?)</action>", text, re.IGNORECASE)
    action = act_m.group(1).strip().upper() if act_m else "KEEP_SEPARATE"

    canon_m = re.search(r"<primary_title>(.*?)</primary_title>", text, re.DOTALL | re.IGNORECASE)
    primary = canon_m.group(1).strip() if canon_m else ""

    prune_m = re.search(r"<prune_title>(.*?)</prune_title>", text, re.DOTALL | re.IGNORECASE)
    prune = prune_m.group(1).strip() if prune_m else ""

    reason_m = re.search(r"<reason>(.*?)</reason>", text, re.DOTALL | re.IGNORECASE)
    reason = reason_m.group(1).strip() if reason_m else ""

    if action == "KEEP_SEPARATE" and "{" in text:
        try:
            m = re.search(r"\{.*\}", text, re.DOTALL)
            if m:
                payload = json.loads(m.group(0))
                action = payload.get("action", "KEEP_SEPARATE").upper()
                primary = payload.get("primary_title", "")
                prune = payload.get("prune_title", "")
                reason = payload.get("reason", "")
        except Exception:
            pass

    return FinalActionVerdict(action=action, primary_title=primary, prune_title=prune, reason=reason)


# ── DisjointSet ───────────────────────────────────────────────────────────────

class DisjointSet:
    def __init__(self):
        self.parent: Dict[str, str] = {}

    def find(self, item: str) -> str:
        curr = item.strip().lower()
        path = []
        while curr in self.parent and self.parent[curr] != curr:
            path.append(curr)
            curr = self.parent[curr]
        for p in path:
            self.parent[p] = curr
        if curr not in self.parent:
            self.parent[curr] = curr
        return curr

    def union(self, a: str, b: str):
        root_a = self.find(a)
        root_b = self.find(b)
        if root_a != root_b:
            self.parent[root_b] = root_a


# ── Pipeline Execution ────────────────────────────────────────────────────────

class FinalConsolidationPipeline:
    def __init__(self, args):
        self.args = args
        wiki_path = getattr(args, "wiki_path", None)
        if not wiki_path:
            raise ValueError("Missing required --wiki_path argument.")

        self.wiki_path = Path(wiki_path)
        self.global_concepts_file = self.wiki_path / "global_concepts.json"
        self.pre_consolidation_backup = self.wiki_path / "global_concepts_pre_consolidation.json"
        self.md_dir = self.wiki_path / "concepts_md"
        self.islands_file = self.wiki_path / "islands_report.json"
        self.macro_settings = resolve_settings(args, STAGE)
        self.deep_settings = resolve_settings(args, DEEP_STAGE)

        if not self.global_concepts_file.exists():
            raise FileNotFoundError(f"Missing {self.global_concepts_file}. Run Stage 7 & 8 first.")

    def _rank_links_and_inject_rescue_backlinks(
        self,
        final_concepts: List[Dict[str, Any]],
        primary_cap: int = 15,
        rescue_slots: int = 2,
    ) -> Dict[str, Any]:
        """
        1. Computes Link Score for every outbound relation using Jaccard & Asymmetric Containment:
           `hybrid_score = 0.55 * Jaccard + 0.45 * Asymmetric_Containment`.
        2. Sorts outbound relations by (`verified_by_llm`, `hybrid_score`, `jaccard_similarity`) descending.
        3. Identifies peripheral/specialized concepts that have `primary_in_degree == 0` under the Top-`primary_cap`
           truncation (i.e. concepts whose outbound links point to hubs, but no hub includes them in its Top-15).
        4. Injects reciprocal rescue backlinks into the target hub's primary window (using up to `rescue_slots` slots
           at the bottom of the Top-15, i.e., slots 14-15) so peripheral concepts become bidirectionally reachable!
        """
        title_to_idx = {c["title"].lower(): idx for idx, c in enumerate(final_concepts)}
        entity_sets = {
            c["title"].lower(): {e.strip().lower() for e in c.get("entities", []) if e.strip()}
            for c in final_concepts
        }

        # Step 1: Compute Score for all outbound relations
        for idx_a, c in enumerate(final_concepts):
            t_a_low = c["title"].lower()
            ents_a = entity_sets.get(t_a_low, set())

            for rel in c.get("outbound_relations", []):
                t_b = rel.get("target_concept", "").strip()
                t_b_low = t_b.lower()
                ents_b = entity_sets.get(t_b_low, set())

                inter_len = len(ents_a & ents_b)
                union_len = len(ents_a | ents_b)
                min_len = min(len(ents_a), len(ents_b))

                recomputed_jacc = inter_len / union_len if union_len > 0 else 0.0
                # F-19: Keep Stage 8's score as a floor so title mentions remain top-ranked
                jacc = max(recomputed_jacc, float(rel.get("jaccard_similarity") or 0.0))

                # F-19: Containment only computed when both sets have >= 4 entities
                if min_len >= 4:
                    containment = inter_len / min_len
                else:
                    containment = jacc

                rel["jaccard_similarity"] = round(float(jacc), 3)
                rel["containment_score"] = round(float(containment), 3)
                hybrid = 0.55 * jacc + 0.45 * containment
                rel["hybrid_score"] = round(float(hybrid), 3)

            # F-19: Final ranking: (verified_by_llm, link_type_rank, hybrid_score)
            type_rank = {"title_mention": 3, "llm_verified": 2, "entity_overlap": 1}
            c["outbound_relations"].sort(
                key=lambda r: (
                    1 if r.get("verified_by_llm") else 0,
                    type_rank.get(r.get("link_type", ""), 1),
                    r.get("hybrid_score", 0.0),
                    r.get("jaccard_similarity", 0.0),
                ),
                reverse=True,
            )

        # Step 2: Measure initial Top-K (`primary_cap`) directed in-degree for every concept
        primary_inbound_from: Dict[str, Set[str]] = defaultdict(set)
        for c in final_concepts:
            src_title = c["title"]
            for rel in c.get("outbound_relations", [])[:primary_cap]:
                tgt_low = rel.get("target_concept", "").strip().lower()
                if tgt_low in title_to_idx:
                    primary_inbound_from[tgt_low].add(src_title)

        initial_zero_in_count = sum(
            1 for c in final_concepts
            if len(primary_inbound_from.get(c["title"].lower(), set())) == 0 and len(c.get("outbound_relations", [])) > 0
        )

        # Step 3: Reciprocal Rescue Backlink Injection (respecting `rescue_slots` per hub)
        rescued_slots_used: Dict[str, int] = Counter()
        rescued_concepts_count = 0

        for c in final_concepts:
            c_title = c["title"]
            c_low = c_title.lower()
            out_rels = c.get("outbound_relations", [])

            # If this concept has outbound links but ZERO inbound primary links (in-degree == 0 in Top-15):
            if out_rels and len(primary_inbound_from.get(c_low, set())) == 0:
                rescued = False
                # Try its top outbound targets in order of hybrid_score
                for rel in out_rels:
                    tgt_title = rel.get("target_concept", "").strip()
                    tgt_low = tgt_title.lower()
                    tgt_idx = title_to_idx.get(tgt_low)
                    if tgt_idx is None:
                        continue
                    tgt_obj = final_concepts[tgt_idx]
                    tgt_out = tgt_obj.get("outbound_relations", [])

                    # Check if target hub still has an available rescue slot (or has < primary_cap links)
                    if len(tgt_out) < primary_cap or rescued_slots_used[tgt_low] < rescue_slots:
                        existing_rel = None
                        for r_idx, r_item in enumerate(tgt_out):
                            if r_item.get("target_concept", "").strip().lower() == c_low:
                                existing_rel = tgt_out.pop(r_idx)
                                break

                        if existing_rel is None:
                            existing_rel = {
                                "target_concept": c_title,
                                "relation": rel.get("relation", "") or f"Reciprocally connected to {c_title}.",
                                "jaccard_similarity": rel.get("jaccard_similarity", 0.0),
                                "containment_score": rel.get("containment_score", 0.0),
                                "cosine_similarity": rel.get("cosine_similarity"),
                                "hybrid_score": rel.get("hybrid_score", 0.0),
                                "shared_entities": rel.get("shared_entities", []),
                                "source_chunks": [],
                            }
                        existing_rel["rescued_backlink"] = True

                        if len(tgt_out) < primary_cap:
                            tgt_out.append(existing_rel)
                        else:
                            # Insert into the rescue window at the end of primary_relations (e.g. slot 14 or 15)
                            insert_pos = max(0, primary_cap - 1 - rescued_slots_used[tgt_low])
                            tgt_out.insert(insert_pos, existing_rel)
                            rescued_slots_used[tgt_low] += 1

                        primary_inbound_from[c_low].add(tgt_title)
                        rescued = True
                        rescued_concepts_count += 1
                        break

                # Fallback: if all top hubs had full rescue slots, force-insert into its #1 neighbor's last primary slot
                if not rescued and out_rels:
                    best_rel = out_rels[0]
                    tgt_low = best_rel.get("target_concept", "").strip().lower()
                    tgt_idx = title_to_idx.get(tgt_low)
                    if tgt_idx is not None:
                        tgt_obj = final_concepts[tgt_idx]
                        tgt_out = tgt_obj.get("outbound_relations", [])
                        existing_rel = None
                        for r_idx, r_item in enumerate(tgt_out):
                            if r_item.get("target_concept", "").strip().lower() == c_low:
                                existing_rel = tgt_out.pop(r_idx)
                                break
                        if existing_rel is None:
                            existing_rel = {
                                "target_concept": c_title,
                                "relation": best_rel.get("relation", "") or f"Reciprocally connected to {c_title}.",
                                "jaccard_similarity": best_rel.get("jaccard_similarity", 0.0),
                                "containment_score": best_rel.get("containment_score", 0.0),
                                "cosine_similarity": best_rel.get("cosine_similarity"),
                                "hybrid_score": best_rel.get("hybrid_score", 0.0),
                                "shared_entities": best_rel.get("shared_entities", []),
                                "source_chunks": [],
                            }
                        existing_rel["rescued_backlink"] = True
                        insert_pos = min(len(tgt_out), max(0, primary_cap - 1))
                        tgt_out.insert(insert_pos, existing_rel)
                        primary_inbound_from[c_low].add(tgt_obj["title"])
                        rescued_concepts_count += 1

        # Recompute final Top-K (`primary_cap`) directed connectivity stats
        final_primary_in: Dict[str, Set[str]] = defaultdict(set)
        for c in final_concepts:
            src_low = c["title"].lower()
            for rel in c.get("outbound_relations", [])[:primary_cap]:
                tgt_low = rel.get("target_concept", "").strip().lower()
                if tgt_low in title_to_idx:
                    final_primary_in[tgt_low].add(src_low)

        bidirectional_cnt = 0
        inbound_connected_cnt = 0
        outbound_only_cnt = 0
        completely_isolated_cnt = 0

        for c in final_concepts:
            c_low = c["title"].lower()
            in_set = final_primary_in.get(c_low, set())
            out_set = {
                r.get("target_concept", "").strip().lower()
                for r in c.get("outbound_relations", [])[:primary_cap]
                if r.get("target_concept", "").strip().lower() in title_to_idx
            }
            has_in = len(in_set) > 0
            has_out = len(out_set) > 0
            if has_in and has_out:
                bidirectional_cnt += 1
                inbound_connected_cnt += 1
            elif has_in and not has_out:
                inbound_connected_cnt += 1
            elif has_out and not has_in:
                outbound_only_cnt += 1
            else:
                completely_isolated_cnt += 1

        stats = {
            "primary_link_cap": primary_cap,
            "rescue_slots_per_hub": rescue_slots,
            "initial_source_only_concepts_before_rescue": initial_zero_in_count,
            "rescued_backlinks_injected": rescued_concepts_count,
            "final_primary_inbound_connected": inbound_connected_cnt,
            "final_primary_bidirectional": bidirectional_cnt,
            "final_primary_outbound_only": outbound_only_cnt,
            "final_completely_isolated": completely_isolated_cnt,
        }
        print(
            f"[Hybrid Ranking & Rescue Backlinks] Rescued {rescued_concepts_count}/{initial_zero_in_count} source-only concepts | "
            f"Final Top-{primary_cap} Inbound-Connected: {inbound_connected_cnt}, Outbound-Only: {outbound_only_cnt}, Isolated: {completely_isolated_cnt}"
        )
        return stats

    def run(self):
        asyncio.run(self.run_async())

    async def run_async(self):
        verify_stage_manifest(self.base_wiki_dir, "link-entities")
        print(f"Loading concepts from {self.global_concepts_file}...")
        with open(self.global_concepts_file, "r", encoding="utf-8") as f:
            data = json.load(f)
            concepts = data.get("concepts", [])

        print(f"Loaded {len(concepts)} concepts for final consolidation.")

        if not self.pre_consolidation_backup.exists():
            print(f"Saving pre-consolidation safety backup to {self.pre_consolidation_backup}...")
            atomic_write_json(self.pre_consolidation_backup, {"concepts": concepts})

        concept_by_name = {c["title"].lower(): c for c in concepts}

        alpha_buckets: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
        for c in concepts:
            t = c.get("title", "").strip()
            if not t:
                continue
            first_char = t[0].upper()
            bucket_key = first_char if first_char.isalpha() else "#"
            alpha_buckets[bucket_key].append(c)

        # ── PASS 1: ALPHABETICAL MACRO SCAN ─────────────────────────────────
        candidate_pair_keys: Set[Tuple[str, str]] = set()
        prune_candidate_titles: Set[str] = set()

        for bucket, b_concepts in alpha_buckets.items():
            sorted_b = sorted(b_concepts, key=lambda x: x["title"].lower())
            for i in range(len(sorted_b) - 1):
                c1, c2 = sorted_b[i], sorted_b[i + 1]
                t1, t2 = c1["title"].lower(), c2["title"].lower()
                words1 = set(t1.split())
                words2 = set(t2.split())
                overlap = words1 & words2
                if len(overlap) >= 1 and (len(overlap) / max(len(words1), len(words2))) >= 0.5:
                    candidate_pair_keys.add(tuple(sorted([c1["title"], c2["title"]])))

        if not getattr(self.args, "skip_macro_scan", False):
            batches = []
            for letter in sorted(alpha_buckets.keys()):
                group = sorted(alpha_buckets[letter], key=lambda c: c["title"].lower().strip())
                if len(group) <= 120:
                    batches.append({"label": f"Letter '{letter}'", "concepts": group})
                else:
                    chunk_size = 100
                    overlap = 20
                    stride = chunk_size - overlap
                    for part_idx, i in enumerate(range(0, len(group), stride), 1):
                        sub_part = group[i:i + chunk_size]
                        batches.append({"label": f"Letter '{letter}' (Part {part_idx})", "concepts": sub_part})
                        if i + chunk_size >= len(group):
                            break

            macro_log_file = self.wiki_path / "macro_scan_log.json"
            macro_checkpoint = load_json_log(macro_log_file) if getattr(self.args, "resume", False) else {}
            pending_batches = [b for b in batches if b["label"] not in macro_checkpoint]

            print(f"\n[PASS 1: MACRO SCAN] {len(batches)} alphabetical batches | Pending LLM scan: {len(pending_batches)}")

            if pending_batches:
                macro_pool = pool_for_stage(self.args, self.macro_settings, STAGE)
                log_lock = asyncio.Lock()

                async def on_macro_result(item: Dict[str, Any], res: Dict[str, Any]):
                    async with log_lock:
                        macro_checkpoint[item["label"]] = res
                        save_log_sync(macro_log_file, macro_checkpoint)

                async def run_macro_batch(client: Any, item: Dict[str, Any], server_index: int) -> Dict[str, Any]:
                    batch_c = item["concepts"]
                    id_to_t = {idx + 1: c["title"] for idx, c in enumerate(batch_c)}
                    indexed_str = "\n".join([f"[{idx + 1}] {c['title']}" for idx, c in enumerate(batch_c)])
                    prompt = (
                        f"AUDITING LETTER GROUP: {item['label']}\n\n"
                        f"Indexed Titles:\n{indexed_str}\n"
                    )
                    extra_body: Dict[str, Any] = {}
                    model_kwargs: Dict[str, Any] = {}
                    if getattr(self.args, "disable_thinking", False):
                        extra_body["chat_template_kwargs"] = {"enable_thinking": False}
                        model_kwargs = {
                            "temperature": 0.2,
                            "top_p": 0.90,
                            "extra_body": extra_body,
                        }
                    else:
                        model_kwargs = {"temperature": 0.0}

                    resp = await client.chat.completions.create(
                        model=self.macro_settings.model_name,
                        messages=[
                            {"role": "system", "content": SYSTEM_PROMPT_MACRO_SCAN},
                            {"role": "user", "content": prompt}
                        ],
                        max_tokens=self.macro_settings.max_tokens or 4096,
                        **model_kwargs
                    )
                    raw_text = resp.choices[0].message.content or ""
                    parsed = parse_macro_scan(raw_text)

                    merge_pairs = []
                    for ida, idb in parsed.merge_pairs:
                        if ida in id_to_t and idb in id_to_t:
                            merge_pairs.append([id_to_t[ida], id_to_t[idb]])
                    link_pairs = []
                    for ida, idb in parsed.link_pairs:
                        if ida in id_to_t and idb in id_to_t:
                            link_pairs.append([id_to_t[ida], id_to_t[idb]])
                    prune_titles = [id_to_t[pid] for pid in parsed.prune_ids if pid in id_to_t]

                    return {
                        "merge_pairs": merge_pairs,
                        "link_pairs": link_pairs,
                        "prune_titles": prune_titles
                    }

                try:
                    async with macro_pool:
                        await macro_pool.map(
                            pending_batches,
                            run_macro_batch,
                            attempts=self.macro_settings.attempts,
                            desc="Macro Scanning Alphabetical Batches",
                            key_fn=lambda b: b["label"],
                            on_result=on_macro_result
                        )
                finally:
                    save_log_sync(macro_log_file, macro_checkpoint)

            for b_label, res in macro_checkpoint.items():
                for t1, t2 in res.get("merge_pairs", []):
                    if t1.lower() in concept_by_name and t2.lower() in concept_by_name and t1.lower() != t2.lower():
                        candidate_pair_keys.add(tuple(sorted([t1, t2])))
                for t1, t2 in res.get("link_pairs", []):
                    if t1.lower() in concept_by_name and t2.lower() in concept_by_name and t1.lower() != t2.lower():
                        candidate_pair_keys.add(tuple(sorted([t1, t2])))
                for pt in res.get("prune_titles", []):
                    if pt.lower() in concept_by_name:
                        prune_candidate_titles.add(pt)

        candidate_pairs: List[Tuple[Dict[str, Any], Dict[str, Any]]] = []
        for t1, t2 in candidate_pair_keys:
            if t1.lower() in concept_by_name and t2.lower() in concept_by_name:
                candidate_pairs.append((concept_by_name[t1.lower()], concept_by_name[t2.lower()]))

        print(f"\nDiscovered {len(candidate_pairs)} candidate pair(s) and {len(prune_candidate_titles)} macro prune candidate(s) for deep verification.")

        # ── PASS 2: DEEP VERIFICATION ─────────────────────────────────────────
        decisions: List[Tuple[Tuple[Dict[str, Any], Dict[str, Any]], FinalActionVerdict]] = []

        if candidate_pairs and not getattr(self.args, "skip_deep_verify", False):
            deep_log_file = self.wiki_path / "deep_verify_log.json"
            deep_checkpoint = load_json_log(deep_log_file) if getattr(self.args, "resume", False) else {}

            pending_pairs = [
                pair for pair in candidate_pairs
                if ":::".join(sorted([pair[0]["title"], pair[1]["title"]])) not in deep_checkpoint
            ]

            print(f"[PASS 2: DEEP VERIFICATION] Total pairs: {len(candidate_pairs)} | Pending: {len(pending_pairs)}")

            if pending_pairs:
                deep_pool = pool_for_stage(self.args, self.deep_settings, DEEP_STAGE)
                deep_lock = asyncio.Lock()

                async def on_verdict(item: Tuple[Dict[str, Any], Dict[str, Any]], verdict: FinalActionVerdict):
                    pk = ":::".join(sorted([item[0]["title"], item[1]["title"]]))
                    async with deep_lock:
                        deep_checkpoint[pk] = verdict.model_dump()
                        save_log_sync(deep_log_file, deep_checkpoint)

                async def verify_pair(client: Any, pair: Tuple[Dict[str, Any], Dict[str, Any]], server_index: int) -> FinalActionVerdict:
                    c1, c2 = pair
                    c1_brief = c1.get("brief", "") or c1.get("description", "")[:200]
                    c2_brief = c2.get("brief", "") or c2.get("description", "")[:200]
                    c1_ents = ", ".join(c1.get("entities", [])[:6]) or "None"
                    c2_ents = ", ".join(c2.get("entities", [])[:6]) or "None"

                    prompt = (
                        f"CONCEPT 1: \"{c1.get('title')}\"\n"
                        f"  Brief: {c1_brief}\n"
                        f"  Key Entities: {c1_ents}\n\n"
                        f"CONCEPT 2: \"{c2.get('title')}\"\n"
                        f"  Brief: {c2_brief}\n"
                        f"  Key Entities: {c2_ents}\n"
                    )

                    extra_body: Dict[str, Any] = {}
                    model_kwargs: Dict[str, Any] = {}
                    if getattr(self.args, "disable_thinking", False):
                        extra_body["chat_template_kwargs"] = {"enable_thinking": False}
                        model_kwargs = {
                            "temperature": 0.7,
                            "top_p": 0.80,
                            "presence_penalty": 1.5,
                            "extra_body": extra_body,
                        }
                    else:
                        model_kwargs = {"temperature": 0.0}

                    response = await client.chat.completions.create(
                        model=self.deep_settings.model_name,
                        messages=[
                            {"role": "system", "content": SYSTEM_PROMPT_DEEP_VERIFY},
                            {"role": "user", "content": prompt}
                        ],
                        max_tokens=self.deep_settings.max_tokens or 1000,
                        **model_kwargs
                    )
                    raw_text = response.choices[0].message.content or ""
                    return parse_xml_final_action(raw_text)

                try:
                    async with deep_pool:
                        await deep_pool.map(
                            pending_pairs,
                            verify_pair,
                            attempts=self.deep_settings.attempts,
                            desc="Deep Verifying Candidate Pairs",
                            key_fn=lambda p: f"{p[0].get('title')} vs {p[1].get('title')}",
                            on_result=on_verdict,
                        )
                finally:
                    save_log_sync(deep_log_file, deep_checkpoint)

            for pair in candidate_pairs:
                pk = ":::".join(sorted([pair[0]["title"], pair[1]["title"]]))
                if pk in deep_checkpoint:
                    decisions.append((pair, FinalActionVerdict(**deep_checkpoint[pk])))

        # 3. Sequential Deterministic Application: PRUNE -> MERGE -> LINK
        pruned_titles: Set[str] = set()
        pruned_concept_records: List[Dict[str, Any]] = []
        merge_uf = DisjointSet()
        for c in concepts:
            merge_uf.find(c["title"])

        additional_links: List[Tuple[str, str, str]] = []

        # (A) Collect PRUNE from deep verification (F-21: explicitly identify target prune_title)
        for (c1, c2), verdict in decisions:
            if verdict.action == "PRUNE":
                victim = verdict.prune_title.strip()
                if not victim:
                    # Fallback to shorter description if not named
                    victim = c2["title"] if len(c2.get("description", "")) <= len(c1.get("description", "")) else c1["title"]
                
                # Check description length: do not prune substantive concepts (>150 words) without confirmation
                victim_c = concept_by_name.get(victim.lower())
                if victim_c:
                    victim_words = len(victim_c.get("description", "").split())
                    if victim_words > 150:
                        print(f"[PRUNE GUARD] Skipped pruning '{victim}': contains {victim_words} words of description.")
                        continue
                    pruned_titles.add(victim)
                    pruned_concept_records.append(victim_c)

        if pruned_concept_records:
            pruned_file = self.wiki_path / "pruned_concepts.json"
            atomic_write_json(pruned_file, {"pruned_concepts": pruned_concept_records})
            print(f"Logged {len(pruned_concept_records)} pruned concept(s) to {pruned_file}")

        # (B) Collect MERGE (F-22: Guard against chaining & oversized merges)
        verified_merges = 0
        for (c1, c2), verdict in decisions:
            if verdict.action == "MERGE" and c1["title"] not in pruned_titles and c2["title"] not in pruned_titles:
                t1, t2 = c1["title"], c2["title"]
                len1 = len(c1.get("description", "").split())
                len2 = len(c2.get("description", "").split())
                combined_words = len1 + len2
                # Reject merge if combined size > 1200 tokens (~900 words) unless one is a stub (<45 words)
                if combined_words > 900 and min(len1, len2) > 45:
                    print(f"[MERGE GUARD] Skipped merging large concepts '{t1}' ({len1}w) and '{t2}' ({len2}w). Converting to LINK.")
                    additional_links.append((t1, t2, "Closely related counterpart in domain."))
                    continue
                merge_uf.union(t1, t2)
                verified_merges += 1

        # (C) Collect LINK
        for (c1, c2), verdict in decisions:
            if verdict.action == "LINK" and c1["title"] not in pruned_titles and c2["title"] not in pruned_titles:
                reason = verdict.reason.strip() if verdict.reason else ""
                additional_links.append((c1["title"], c2["title"], reason))

        print(f"Decisions collected: {len(pruned_titles)} pruned, {verified_merges} merged, {len(additional_links)} linked.")

        # 4. Group by Canonical Root
        grouped_concepts: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
        for c in concepts:
            if c["title"] in pruned_titles:
                continue
            root = merge_uf.find(c["title"])
            grouped_concepts[root].append(c)

        final_concepts: List[Dict[str, Any]] = []
        canonical_title_map: Dict[str, str] = {}

        for root, group in grouped_concepts.items():
            # N-05: Carry the chosen canonical title and its stable slug
            title_counter = Counter(c["title"] for c in group)
            chosen_title = title_counter.most_common(1)[0][0]
            canonical_title_map[root] = chosen_title

            # Find slug of the chosen title member, or of any member with a slug
            chosen_slug = None
            for c in group:
                if c["title"].lower() == chosen_title.lower() and c.get("slug"):
                    chosen_slug = c["slug"]
                    break
            if not chosen_slug:
                for c in group:
                    if c.get("slug"):
                        chosen_slug = c["slug"]
                        break
            if not chosen_slug:
                chosen_slug = sanitize_filename(chosen_title)

            merged_aliases = set()
            merged_subtopics = set()
            merged_entities = set()
            merged_chunks = set()
            seen_sentences = set()
            description_sentences = []
            briefs = []

            for c in group:
                merged_aliases.add(c["title"])
                for a in c.get("aliases", []):
                    if a.strip():
                        merged_aliases.add(a.strip())
                for s in c.get("subtopics", []):
                    if s.strip():
                        merged_subtopics.add(s.strip())
                for ent in c.get("entities", []):
                    if ent.strip():
                        merged_entities.add(ent.strip())
                for sc in c.get("source_chunks", []):
                    if sc.strip():
                        merged_chunks.add(sc.strip())

                if c.get("brief"):
                    briefs.append(c["brief"])

                desc = c.get("description", "").strip()
                if desc:
                    for sent in re.split(r'(?<=[.!?])\s+', desc):
                        s_clean = sent.strip()
                        if s_clean and s_clean.lower() not in seen_sentences:
                            seen_sentences.add(s_clean.lower())
                            description_sentences.append(s_clean)

            merged_aliases.discard(chosen_title)
            merged_subtopics.discard(chosen_title)
            chosen_brief = max(briefs, key=len) if briefs else ""
            fused_desc = clean_chunk_citations(" ".join(description_sentences))

            raw_relations = []
            for c in group:
                raw_relations.extend(c.get("outbound_relations", []))

            final_concepts.append({
                "title": chosen_title,
                "slug": chosen_slug,
                "brief": chosen_brief,
                "aliases": sorted(list(merged_aliases)),
                "subtopics": sorted(list(merged_subtopics)),
                "entities": sorted(list(merged_entities)),
                "description": fused_desc,
                "source_chunks": sorted(list(merged_chunks)),
                "_raw_relations": raw_relations,
            })

        concept_by_name = {c["title"].lower(): c for c in final_concepts}
        # N-05: Ensure every concept has a unique slug
        assert len(set(c["slug"] for c in final_concepts)) == len(final_concepts), "Duplicate slug in final_concepts!"

        # 5. Remap Outbound Relations & REMOVE SELF-LOOPS
        print("Remapping relations and removing self-loops...")
        for c in final_concepts:
            current_title = c["title"]
            fused_relations: Dict[str, Dict[str, Any]] = {}

            for rel in c.get("_raw_relations", []):
                tgt_raw = rel.get("target_concept", "").strip()
                if not tgt_raw:
                    continue
                root_tgt = merge_uf.find(tgt_raw)
                canonical_target = canonical_title_map.get(root_tgt, tgt_raw)

                if canonical_target.lower() == current_title.lower():
                    continue

                if canonical_target in pruned_titles or canonical_target.lower() not in concept_by_name:
                    continue

                if canonical_target not in fused_relations:
                    fused_relations[canonical_target] = {
                        "target_concept": canonical_target,
                        "relation": rel.get("relation", ""),
                        "overlap_score": rel.get("overlap_score") or rel.get("jaccard_similarity"),
                        "jaccard_similarity": rel.get("jaccard_similarity"),
                        "link_type": rel.get("link_type", "entity_overlap"),
                        "verified_by_llm": rel.get("verified_by_llm", False),
                        "shared_entities": rel.get("shared_entities", []),
                        "source_chunks": list(rel.get("source_chunks", []))
                    }
                else:
                    existing_text = fused_relations[canonical_target]["relation"]
                    new_text = rel.get("relation", "")
                    if new_text and new_text not in existing_text:
                        fused_relations[canonical_target]["relation"] = f"{existing_text} {new_text}".strip()
                    j1 = fused_relations[canonical_target].get("jaccard_similarity") or 0.0
                    j2 = rel.get("jaccard_similarity") or 0.0
                    if j2 > j1:
                        fused_relations[canonical_target]["jaccard_similarity"] = j2
                        fused_relations[canonical_target]["overlap_score"] = j2

            c["outbound_relations"] = list(fused_relations.values())
            del c["_raw_relations"]

        # Apply Additional Verified Links
        for t1, t2, reason in additional_links:
            root1 = canonical_title_map.get(merge_uf.find(t1), t1)
            root2 = canonical_title_map.get(merge_uf.find(t2), t2)
            if root1.lower() != root2.lower() and root1.lower() in concept_by_name and root2.lower() in concept_by_name:
                c1_obj = concept_by_name[root1.lower()]
                c2_obj = concept_by_name[root2.lower()]
                targets1 = {r["target_concept"].lower() for r in c1_obj["outbound_relations"]}
                targets2 = {r["target_concept"].lower() for r in c2_obj["outbound_relations"]}

                ents1 = set(c1_obj.get("entities", []))
                ents2 = set(c2_obj.get("entities", []))
                shared_ents = sorted(list(ents1 & ents2))
                union_len = len(ents1 | ents2)
                calc_jacc = round(len(shared_ents) / union_len, 3) if union_len > 0 else 0.0

                rel_text = f"Verified: {reason}" if reason else "Directly related in context."

                if root2.lower() not in targets1:
                    c1_obj["outbound_relations"].append({
                        "target_concept": root2,
                        "relation": rel_text,
                        "overlap_score": calc_jacc,
                        "jaccard_similarity": calc_jacc,
                        "shared_entities": shared_ents,
                        "link_type": "llm_verified",
                        "source_chunks": [],
                        "verified_by_llm": True
                    })
                else:
                    for r in c1_obj["outbound_relations"]:
                        if r["target_concept"].lower() == root2.lower():
                            r["verified_by_llm"] = True
                            r["link_type"] = "llm_verified"
                            if reason and reason not in r.get("relation", ""):
                                r["relation"] = f"{r.get('relation', '')} ({reason})".strip()

                if root1.lower() not in targets2:
                    c2_obj["outbound_relations"].append({
                        "target_concept": root1,
                        "relation": rel_text,
                        "overlap_score": calc_jacc,
                        "jaccard_similarity": calc_jacc,
                        "shared_entities": shared_ents,
                        "link_type": "llm_verified",
                        "source_chunks": [],
                        "verified_by_llm": True
                    })
                else:
                    for r in c2_obj["outbound_relations"]:
                        if r["target_concept"].lower() == root1.lower():
                            r["verified_by_llm"] = True
                            r["link_type"] = "llm_verified"
                            if reason and reason not in r.get("relation", ""):
                                r["relation"] = f"{r.get('relation', '')} ({reason})".strip()

        # 5b. Score Ranking + Reciprocal Rescue Backlinks
        primary_cap = getattr(self.args, "max_primary_links", 15)
        rescue_slots = getattr(self.args, "rescue_slots", 2)
        directed_link_stats = self._rank_links_and_inject_rescue_backlinks(
            final_concepts,
            primary_cap=primary_cap,
            rescue_slots=rescue_slots,
        )

        print(f"Total surviving concepts after final merge & pruning: {len(final_concepts)}")
        atomic_write_json(self.global_concepts_file, {"concepts": final_concepts})

        print_link_distribution(final_concepts, title_label="STAGE 9: SURVIVING CONCEPTS GRAPH")

        concepts_dir = self.wiki_path / "concepts"
        if concepts_dir.is_dir():
            print("Syncing final consolidated concepts into concepts/ by unique slug...")
            surviving_files = set()
            for c in final_concepts:
                slug = c.get("slug")
                if not slug:
                    slug = sanitize_filename(c['title']) or "unnamed_concept"
                c_fname = f"{slug}.json"
                surviving_files.add(c_fname)
                atomic_write_json(concepts_dir / c_fname, c, fsync_dir=False)
            for old_file in concepts_dir.glob("*.json"):
                if old_file.name not in surviving_files:
                    try:
                        old_file.unlink()
                    except OSError:
                        pass

        # 6. Island Health Audit (Fast O(1) Lookup for Connected Components & Giant Component)
        print("Analyzing knowledge graph topology & islands...")
        G = nx.Graph()
        valid_titles_set = {con["title"] for con in final_concepts}
        for c in final_concepts:
            G.add_node(c["title"])
            for rel in c.get("outbound_relations", []):
                tgt = rel.get("target_concept", "").strip()
                if tgt and tgt in valid_titles_set:
                    G.add_edge(c["title"], tgt)

        components = list(nx.connected_components(G))
        components.sort(key=len, reverse=True)
        giant_size = len(components[0]) if components else 0
        coverage = (giant_size / len(final_concepts) * 100) if final_concepts else 0.0

        islands_summary = {
            "total_concepts": len(final_concepts),
            "total_islands": len(components),
            "giant_component_size": giant_size,
            "giant_component_coverage_pct": round(coverage, 2),
            "isolated_concepts_count": sum(1 for comp in components if len(comp) == 1),
            "island_sizes": [len(comp) for comp in components[:20]],
            "directed_top_k_reachability": directed_link_stats,
        }
        atomic_write_json(self.islands_file, islands_summary)

        # 7. Compile and Export Markdown Wiki, concept_index.md & entity_index.md (with Stale File Cleanup)
        print(f"Exporting Markdown Wiki to {self.md_dir}...")
        self.md_dir.mkdir(parents=True, exist_ok=True)
        surviving_md_names: Set[str] = {"concept_index.md", "entity_index.md"}

        concept_index_entries: List[Tuple[str, str, str]] = []
        entity_to_concepts: Dict[str, Set[Tuple[str, str]]] = defaultdict(set)

        # Build reverse index for compact "Referenced by" section (Section 4.3)
        referenced_by_map: Dict[str, List[Tuple[str, str]]] = defaultdict(list)
        for c in final_concepts:
            src_t = c["title"]
            src_s = c.get("slug") or generate_stable_slug(src_t)
            for rel in c.get("outbound_relations", [])[:getattr(self.args, "max_primary_links", 15)]:
                tgt_t = rel.get("target_concept", "")
                if tgt_t:
                    referenced_by_map[tgt_t.lower()].append((src_s, src_t))

        for c in tqdm(final_concepts, desc="Exporting Markdown Pages"):
            title = c["title"]
            brief = c.get("brief", "").strip()
            desc = clean_chunk_citations(c.get("description", ""))
            relations = c.get("outbound_relations", [])
            aliases = [a.strip() for a in c.get("aliases", []) if a.strip()]
            subtopics = [s.strip() for s in c.get("subtopics", []) if s.strip()]
            entities = [e.strip() for e in c.get("entities", []) if e.strip()]

            slug = c.get("slug") or generate_stable_slug(title)
            md_filename = f"{slug}.md"
            surviving_md_names.add(md_filename)
            md_path = self.md_dir / md_filename

            # Track for concept_index.md
            concept_index_entries.append((slug, title, brief))

            # Track for entity_index.md (key entities, aliases, and subtopics)
            for ent in entities:
                entity_to_concepts[ent].add((slug, title))
            for al in aliases:
                entity_to_concepts[al].add((slug, title))
            for st in subtopics:
                entity_to_concepts[st].add((slug, title))

            md_content = [f"# {title}\n"]
            if aliases:
                md_content.append(f"**Aliases**: {', '.join(aliases)}\n")
            if subtopics:
                md_content.append(f"**Covers**: {', '.join(subtopics)}\n")
            if entities:
                md_content.append(f"**Key Entities**: {', '.join(entities[:15])}\n")

            if brief:
                md_content.append("## Overview\n")
                md_content.append(f"{brief}\n")

            md_content.append("## Description\n")
            md_content.append(f"{desc}\n")

            if relations:
                md_content.append("## Related\n")
                primary_cap = getattr(self.args, "max_primary_links", 15)
                primary_relations = relations[:primary_cap]
                extended_relations = relations[primary_cap:]

                for rel in primary_relations:
                    tgt = rel.get("target_concept", "")
                    r_text = rel.get("relation", "").strip()
                    tgt_obj = concept_by_name.get(tgt.lower())
                    tgt_slug = tgt_obj.get("slug") if tgt_obj else generate_stable_slug(tgt)
                    if r_text:
                        rel_desc = r_text.rstrip(".")
                        md_content.append(f"- [[{tgt_slug}|{tgt}]] ({rel_desc})")
                    else:
                        md_content.append(f"- [[{tgt_slug}|{tgt}]]")

                # Section 4.3: Compact "Referenced by" list guaranteeing reachability from spoke to hub
                refs = referenced_by_map.get(title.lower(), [])
                if refs:
                    md_content.append("\n## Referenced by\n")
                    ref_links = [f"[[{r_slug}|{r_t}]]" for r_slug, r_t in refs]
                    md_content.append(", ".join(ref_links))

                if extended_relations:
                    md_content.append("\n## Mentions\n")
                    compact_links = []
                    for r in extended_relations:
                        tgt = r.get("target_concept", "")
                        if tgt:
                            tgt_obj = concept_by_name.get(tgt.lower())
                            tgt_slug = tgt_obj.get("slug") if tgt_obj else generate_stable_slug(tgt)
                            compact_links.append(f"[[{tgt_slug}|{tgt}]]")
                    md_content.append(", ".join(compact_links))
                md_content.append("")

            atomic_write_text(md_path, "\n".join(md_content))

        # 8. Export Single-Line concept_index.md (optimized for grep)
        print("Generating single-line grep-friendly concept_index.md...")
        concept_index_entries.sort(key=lambda x: x[1].lower())
        c_lines = ["# Concept Index\n"]
        for slug, c_title, c_brief in concept_index_entries:
            # Single-line format: - [[slug|Title]]: Brief description
            if c_brief:
                # Collapse any multiline brief to a single clean line
                clean_brief_line = " ".join(c_brief.split())
                c_lines.append(f"- [[{slug}|{c_title}]]: {clean_brief_line}")
            else:
                c_lines.append(f"- [[{slug}|{c_title}]]")
        c_lines.append("")
        atomic_write_text(self.md_dir / "concept_index.md", "\n".join(c_lines))

        # 9. Export entity_index.md (mapping entities and aliases to concepts)
        print("Generating entity_index.md...")
        sorted_entities = sorted(entity_to_concepts.keys(), key=lambda x: x.lower())
        e_lines = ["# Entity Index\n"]
        for ent in sorted_entities:
            target_links = sorted(entity_to_concepts[ent], key=lambda x: x[1].lower())
            links_str = ", ".join([f"[[{slug}|{t}]]" for slug, t in target_links])
            e_lines.append(f"- **{ent}**: {links_str}")
        e_lines.append("")
        atomic_write_text(self.md_dir / "entity_index.md", "\n".join(e_lines))

        # Remove any stale/pruned/merged Markdown files from `concepts_md/`
        removed_stale = 0
        for old_md in self.md_dir.glob("*.md"):
            if old_md.name not in surviving_md_names:
                try:
                    old_md.unlink()
                    removed_stale += 1
                except OSError:
                    pass
        if removed_stale > 0:
            print(f"Cleaned up {removed_stale} stale/merged/pruned Markdown file(s) from {self.md_dir}.")

        write_stage_manifest(
            wiki_path=self.wiki_path,
            stage="final-consolidation",
            expected=len(concepts),
            completed=len(final_concepts),
            failed=0,
            islands=len(components),
            giant_component_coverage=round(coverage, 2),
        )

        print("\n============================================================")
        print("STAGE 9: FINAL CONSOLIDATION & WIKI EXPORT COMPLETE")
        print(f"Total Concepts in Wiki:      {len(final_concepts)}")
        print(f"Total Islands:               {len(components)}")
        print(f"Giant Component Coverage:    {giant_size}/{len(final_concepts)} ({coverage:.1f}%)")
        print(f"Rescued Source-Only Links:   {directed_link_stats['rescued_backlinks_injected']}")
        print(f"Primary Inbound Connected:   {directed_link_stats['final_primary_inbound_connected']}/{len(final_concepts)}")
        print(f"Markdown Wiki Exported:      {self.md_dir}")
        print(f"Island Report Written:       {self.islands_file}")
        print("============================================================\n")


def main():
    parser = build_parser(stage=STAGE, description="Final knowledge graph consolidation, link ranking, island audit and markdown wiki export", include_resume=True)
    parser.add_argument("--skip_macro_scan", action="store_true",
                        help="Skip LLM alphabetical macro scan and use fast in-memory candidate scan.")
    parser.add_argument("--skip_deep_verify", action="store_true",
                        help="Skip LLM deep verification and export current graph.")
    parser.add_argument("--max_primary_links", type=int, default=15,
                        help="Maximum primary related links with full explanations on Markdown pages (remainder listed compactly).")
    parser.add_argument("--rescue_slots", type=int, default=2,
                        help="Number of slots within --max_primary_links reserved for reciprocal rescue backlinks to source-only nodes (default: 2).")
    args = parser.parse_args()

    pipeline = FinalConsolidationPipeline(args)
    pipeline.run()


if __name__ == "__main__":
    main()
